 46def album_add(num: int) -> None:
 47    if num is None or num < 0:
 48        return
 49    import tuipet.data.loaders.data as data
 50    num = data.canonical_num(num)    # store the name-canonical identity
 51    if num in _ALBUM_SEEN:
 52        return
 53    _ALBUM_SEEN.add(num)
 54    from .settings_io import load_settings, save_settings
 55    d = load_settings()
 56    prog = d.setdefault("progress", {})
 57    album = set(prog.get("album", []))
 58    if num in album:
 59        return                       # already registered -> no write
 60    album.add(num)
 61    prog["album"] = sorted(album)
 62    save_settings(d)

304def add_pending_bug(rec: Any) -> Any:
305    """Stash a bug that could not be sent (offline) to retry next launch.
306    True when it is safely on disk -- the caller PROMISES the player it will
307    send later, so a failed stash must not be reported as a save (swallowed-
308    failure sweep 2026-07-13)."""
309    import os as _os
310    import json as _json
311    try:
312        _os.makedirs(SAVE_DIR, exist_ok=True)
313        with open(_os.path.join(SAVE_DIR, "pending_bugs.jsonl"), "a", encoding="utf-8") as f:
314            f.write(_json.dumps(rec) + "\n")
315        return True
316    except OSError:
317        return False

319def peek_pending_bugs() -> Any:
320    """Read the stashed bugs WITHOUT deleting them (bug audit 2026-07-19:
321    the old take-then-send cleared the stash up front, so quitting mid-
322    flush lost every unsent report -- the round-5 PM-flush lesson).  The
323    flush rewrites the survivors when it is done."""
324    import os as _os
325    import json as _json
326    p = _os.path.join(SAVE_DIR, "pending_bugs.jsonl")
327    try:
328        return [_json.loads(l) for l in open(p, encoding="utf-8") if l.strip()]
329    except (OSError, ValueError):
330        return []

332def write_pending_bugs(recs: Any) -> Any:
333    """Atomically rewrite the stash to exactly `recs` ([] removes the file).
334    True when it landed on disk."""
335    import os as _os
336    import json as _json
337    p = _os.path.join(SAVE_DIR, "pending_bugs.jsonl")
338    try:
339        if not recs:
340            if _os.path.exists(p):
341                _os.remove(p)
342            return True
343        tmp = p + ".tmp"
344        with open(tmp, "w", encoding="utf-8") as f:
345            for r in recs:
346                f.write(_json.dumps(r) + "\n")
347        _os.replace(tmp, p)
348        return True
349    except OSError:
350        return False
