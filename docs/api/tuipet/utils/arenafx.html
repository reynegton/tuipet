  72def _filth_pts(pet: Any, tick: Any, count: Optional[Any]=None, sizes: Optional[Any]=None, push: int=0, px_h: Optional[Any]=None) -> Any:
  73    """DVPet drawFilthLevel + animFilth: per-pile SIZED sprites (the real filth.png
  74    sizes 1-4, two anim frames each, from pet.poop_sizes) laid in fixed 2-high
  75    columns stepping right from the grid's left edge; the frame swaps every
  76    7 ticks awake / 10 asleep."""
  77    E = data.load_effects()
  78    n = min((pet.poop if count is None else count) or 0, POOP_MAX_PILES)
  79    if n <= 0:
  80        return []
  81    if px_h is None:
  82        px_h = SCREEN_ROWS * 2
  83    sz = list(pet.poop_sizes if sizes is None else (sizes or []))
  84    fi = (tick // (10 if getattr(pet, "asleep", False) else 7)) % 2
  85    pts = []
  86    for i in range(n):
  87        s = max(1, min(4, sz[i] if i < len(sz) else 2))
  88        frames = E.get("poop_s%d" % s) or E.get("poop") or []
  89        if not frames:
  90            continue
  91        pm = frames[fi % len(frames)]
  92        if len(pm[0]) > POOP_W:                     # safety: only the unused size-4 art exceeds the slot
  93            pm = grid.fit_w(pm, POOP_W)
  94        if len(pm) > POOP_W:
  95            pm = grid.fit_band(pm, (POOP_W + 2) * 2)
  96        ph_ = len(pm)
  97        col, up = i // 2, i % 2
  98        x = grid.X0 + col * (POOP_W + POOP_PAD) - push
  99        if up:      # top slot sits on the 8-tall bottom slot -> the 2x2 block tops out at the band (y6)
 100            y = px_h - 2 - POOP_W - ph_
 101        else:       # bottom slot grounds 2px above the border
 102            y = px_h - 2 - ph_
 103        pts += _blit(pm, x, y)
 104    return pts

 205def _holiday_decor(today: Optional[Any]=None) -> Any:
 206    """TODAY's festival decoration sprite (cropped rows), or None.
 207
 208    Food icons are stored at 3x (24px cells over an 8px native sprite -- a
 209    clean integer upscale), so a food-keyed decoration is downsampled back

 224        sprite = [r[::3] for r in sprite[::3]]
 225    return grid._crop(sprite)

 228def _effect_overlay(pet: Any, frame_i: Any, cols: Any, px_h: Any, tick: int=0) -> Any:
 229    """The scene's overlay actors: filth piles, the sleep Zzz, the sick skull.
 230    Nothing else -- badges belong to the HUD, not the play field."""
 231    E = data.load_effects()
 232    pts = []  # type: ignore
 233    if pet.dead:
 234        return pts
 235    # a festival prop in the TOP-LEFT corner (2026-07-24): ambient, shown for
 236    # a live pet OR an egg.  It sits where a resting pet (centred, x~12..28)
 237    # does not reach; a roamer may briefly pass in front, exactly like poop.
 238    deco = _holiday_decor()
 239    if deco:
 240        pts += _blit(deco, grid.X0, grid.TOP)
 241    # sized piles in the slot grid -- BUT NOT IN THE DARK (Joel 2026-07-25:
 242    # "is the poop supposed to be visible during lights off?").  It was:
 243    # the lights-off branch blanks the PET rows and let the overlay through,
 244    # so the piles (flies and all) sat lit on a field that is meant to be
 245    # DVPet's fully-opaque lightsOff -- the same cover that makes a
 246    # dark-room fx "show NOTHING".  The sick skull already reads the switch
 247    # this way (`_sick_mark_up`: sick AND `pet.lights`), and the Zzz below
 248    # is deliberately the ONE mark that survives the blackout.  The piles
 249    # do not vanish, they are unlit: flip the lights and they are all still
 250    # there, waiting to be cleaned.
 251    if pet.poop and pet.lights:
 252        pts += _filth_pts(pet, tick, px_h=px_h)
 253    if pet.num == -1:
 254        return pts
 255    asleep = bool(getattr(pet, "asleep", False))
 256    # sleep Zzz: floats in the sky corner, above the 32x16 world (the 6px
 257    # glyph fills the sky strip exactly; a nap's own glyph is 7px and dips
 258    # onto the band's top row -- sleeping poses lie low, sweep-verified).
 259    # It shows lights-off too: the dark room keeps its Zzz (DVPet
 260    # sleepLightsOff), our one mark on a black field.
 261    zkey = "zzz_nap" if (getattr(pet, "nap", False) and E.get("zzz_nap")) else "zzz"
 262    # the Zzz belongs to the SLEEP SCENE, INSIDE the window (LAW: nothing
 263    # above the band).  It hangs at the band's top-right; a sleeper with <=2
 264    # piles is clamped to CENTRE (x12..27), so the corner columns are its own.
 265    # 3-4 piles push the sleeper right under the corner -- the Zzz yields to
 266    # the HUD (whose status word already says asleep), poop always wins.
 267    # It rides the sleep pose: an asleep pet still wearing another anim (a
 268    # disturbed sleeper's startle beat) skips a beat; a dark room keeps its
 269    # Zzz over the black field.
 270    in_sleep_scene = asleep and (pet.anim == "sleep" or not pet.lights)
 271    z_bot = 0
 272    if (in_sleep_scene and pet.poop < 3 and E.get(zkey)
 273            and zkey not in _HIDDEN_STATUS_ICONS):
 274        z = grid._crop(E[zkey][(tick // 10) % len(E[zkey])])
 275        pts += _blit(z, grid.X1 - len(z[0]), grid.TOP)
 276        z_bot = grid.TOP + len(z)
 277    # sick skull: HIGH at the band's top-right, floating at head height the
 278    # way the device marks status (Joel 2026-07-12 -- the grounded floor
 279    # placement was tuipet's own miss), blinking its 2-frame pair on the
 280    # stateNumTic beat (7 ticks awake / 10 asleep).  A sick SLEEPER's skull
 281    # tucks in under the Zzz -- the corner belongs to the sleep mark.
 282    if _sick_mark_up(pet) and E.get("st_sick") and "st_sick" not in _HIDDEN_STATUS_ICONS:
 283        sf = (tick // (10 if asleep else 7)) % 2
 284        sk = grid._crop(E["st_sick"][sf])       # crop: the cell pads its INK
 285        pts += _blit(sk, grid.X1 - len(sk[0]),
 286                     (z_bot + 1) if z_bot else grid.TOP)
 287    return pts

 296    def start_fx(self, kind: Any, icon: Optional[Any]=None, poop: int=0, old_num: Optional[Any]=None, pet: Optional[Any]=None, starving: bool=False, good: bool=True, script: Optional[Any]=None) -> None:
 297        steps = {"eat": 35, "cheer": 31, "jeer": 31, "clean": 22, "spit": 25, "evolve": 41, "dying": 50, "dna_charge": 44, "play": 48, "poop": 25, "poopdance": 21, "yawn": 22, "losing": 50,
 298                 "gift": GIFT_OUT + GIFT_BACK + GIFT_HOLD, "assist": 28, "inherit": 50}.get(kind, 12)
 299        self.fx = {"kind": kind, "step": 0, "steps": steps, "icon": icon, "poop": poop,
 300                   "old_num": old_num, "good": good}
 301        if kind == "item":
 302            # a scripted item-use (itemfx: the per-AnimationType canon tables)
 303            import tuipet.utils.itemfx as itemfx
 304            sc = itemfx.SCRIPTS[script]  # type: ignore
 305            self.fx["script"] = script
 306            self.fx["steps"] = sc["steps"]
 307            self.fx["snds"] = dict(sc["snds"])
 308            self.fx["end"] = sc["end"]
 309        if kind == "eat":
 310            # DVPet eat(): each chew beat is scaled by pow(N, mod) -- a starving pet or
 311            # a glutton wolfs food down (mod 0.9, ends ~beat 23), a picky eater dawdles
 312            # (mod 1.1, ~48); food descent (beats 0/2/4/6) is NOT scaled.  Disliked
 313            # food -> +9 grimace.  A heavy species (baseWeight>=40) skips a chew cycle
 314            # (DVPet's frame jump 18->26): two bites instead of three, ends ~beat 26.
 315            glut = getattr(pet, "glutton", 0) if pet else 0
 316            mod = 0.9 if (glut > 0 or starving) else (1.1 if glut < 0 else 1.0)
 317            # eat(): the grimace bite (+9) fires on DISLIKED food OR an overeating
 318            # stomach -- canon re-audit 2026-07 (the pill rides this same eat
 319            # fx on its own bite strip; pill-anim fix 2026-07-18)
 320            from tuipet.core.pet import OVEREAT_LIMIT as _OVL
 321            bite = 9 if (pet is not None and (getattr(pet, "_last_meal_disliked", False)
 322                                              or pet.hunger >= _OVL)) else 7
 323            heavy = pet is not None and pet.num != -1 and pet._base_weight() >= 40
 324            leftover = pet is not None and getattr(pet, "_last_meal_leftover", False)
 325            if leftover:
 326                # applyFood: modifier <= DisposeLeftoversMinModifier(0.5) -> the
 327                # STUFFED pet's meal is State.Munching -- two beats of chewing,
 328                # then disposeFood: it turns away and DROPS the rest off-screen
 329                beats = [int(b ** mod) for b in (10, 14, 18)]
 330                self.fx["chew"] = {beats[0]: 8, beats[1]: bite, beats[2]: 8}
 331                fb = (beats[1], 999, 999)                        # only bite one lands
 332                self.fx["bite_snds"] = {beats[1]: "eat"}
 333                self.fx["munch_at"] = beats[2]
 334                self.fx["steps"] = beats[2] + 22                 # the dispose tail
 335            elif heavy:
 336                beats = [int(b ** mod) for b in (10, 14, 18, 22)]
 337                self.fx["chew"] = {beats[0]: 8, beats[1]: bite, beats[2]: 8, beats[3]: bite}
 338                fb = (beats[1], beats[3], beats[3])         # food frames 0 -> 1 -> 3 (skips 2)
 339                self.fx["bite_snds"] = {beats[1]: "eat", beats[3]: "lastBite"}
 340                self.fx["steps"] = int(26 ** mod) + 1
 341            else:
 342                beats = [int(b ** mod) for b in (10, 14, 18, 22, 26, 30)]
 343                self.fx["chew"] = {b: (8 if i % 2 == 0 else bite) for i, b in enumerate(beats)}
 344                fb = (beats[1], beats[3], beats[5])
 345                # DVPet eat(): _eat on the first two bites, _lastBite on the third.
 346                self.fx["bite_snds"] = {fb[0]: "eat", fb[1]: "eat", fb[2]: "lastBite"}
 347                self.fx["steps"] = int(34 ** mod) + 1
 348            self.fx["food_beats"] = fb
 349        elif kind == "spit":
 350            # DVPet refuse(): _refuse fires on EVERY head-shake flip (t0/6/12/18).
 351            # (t0 sounds key as 1 -- the drain runs after the first advance.)
 352            self.fx["snds"] = {1: "refuse", 6: "refuse", 12: "refuse", 18: "refuse"}
 353        elif kind == "cheer":
 354            # DVPet cheer(): its sound (praise/_happy) plays at the anim's t0 --
 355            # keyed here so chained cheers (wash/evolve/heal tails) sound too.
 356            self.fx["snds"] = {1: "happy"}
 357        elif kind == "losing":
 358            # DVPet losing(): jeer(disposition, _lose) -- the sound at the
 359            # first UP beat, like every jeer
 360            self.fx["snds"] = {6: "lose"}
 361        elif kind == "jeer":
 362            # DVPet jeer(): the sound fires at the first UP beat (t6).  Canon
 363            # routes Bad_Scold through the _unhappy cue, but soundConfig.csv
 364            # maps unhappy -> angry.wav -- the same bark either way, so only
 365            # the POSES distinguish the variants here.
 366            self.fx["snds"] = {6: "angry"}
 367        elif kind == "evolve":
 368            # DVPet evolveAnim(): _evolve sounds at the first burst beat (t5);
 369            # evolve() runs the strobe to evolFinish at 41 (was cut at 37).
 370            # An ITEM evolution (Relic) prepends canon itemEvolve's first
 371            # act: the pet parades with the item cycling its own anim frames
 372            # (itemEvolveLoop -> jogress.wav), THEN the strobe fires.
 373            # Pre-ritual (anim hardening 2026-07-14): every reference fronts
 374            # the strobe with a first act -- the old form HOLDS, then its
 375            # SILHOUETTE blinks -- so the change reads as a ritual, not a cut.
 376            # An ITEM evolution keeps canon itemEvolve's parade as that act.
 377            off = 14 if icon else 12
 378            self.fx["off"] = off
 379            self.fx["steps"] = 41 + off
 380            self.fx["snds"] = {5 + off: "evolve"}
 381            if icon:                      # the parade opens on the jogress sting
 382                self.fx["snds"][1] = "jogress"
 383        elif kind == "inherit":
 384            # DVPet inheriting(): chip-shrink t11 / parent-grow t17 / parent-shrink
 385            # t37 all key the attackHit cue; the flight home pips inheritMove
 386            # (=alarm) every 4; the collide lands inheritCollide (=strongHit).
 387            # The t-values above are CANON's 65-beat timeline; these keys are
 388            # the same beats on _fxk_inherit's remapped 50-step timeline
 389            # (shrink 10 / grow 14 / visit-end 32 / collide 46) -- aligned
 390            # with the painter, do NOT "fix" them to the canon numbers
 391            # (sound/fx audit 2026-07-18).
 392            self.fx["snds"] = {10: "attackHit", 14: "attackHit", 32: "attackHit",
 393                               36: "alarm", 40: "alarm", 44: "alarm", 46: "strongHit"}
 394        elif kind == "dna_charge":
 395            # dnaCharge(): _dnaWash as the sweep enters (t21).
 396            self.fx["snds"] = {21: "wash"}

 398    def advance_fx(self) -> Any:
 399        if not self.fx:
 400            return False
 401        self.frame_i += 1        # type: ignore
 402        self.fx["step"] += 1
 403        if self.fx["step"] < self.fx["steps"]:
 404            return True
 405        kind = self.fx["kind"]
 406        had_filth = self.fx.get("poop", 0) > 0
 407        chain_eat = self.fx.get("chain_eat")
 408        pet_ref = self.fx.get("pet_ref")
 409        item_end = self.fx.get("end")
 410        self.fx = None  # type: ignore
 411        if kind == "item" and item_end:
 412            # canon: playing()/interact/study/... resolve into Cheering;
 413            # angrySurprise resolves into Jeering (itemfx script tables)
 414            self.start_fx(item_end, good=True)
 415        elif kind == "clean" and had_filth:
 416            # DVPet clean(): the cheer chains ONLY when filth was actually washed

 489    def _paint_fx(self, pet: Any) -> None:
 490        """The care-fx painter, decomposed (audit 2026-07): a shared context is
 491        pre-loaded (default pose, the cross-kind FILTH prelude),
 492        then the kind's own painter (_fxk_<kind>) mutates it.  Bodies verbatim
 493        from the old 210-line chain; behavior pinned by the fx golden."""
 494        fx = self.fx
 495        on, bg = LCD_ON, LCD_BG
 496        bgimg = self._background(pet)  # type: ignore
 497        if bgimg:
 498            on = SIL_SCENE   # dark silhouette day OR night, same rule as paint() --
 499            #                the pet is never white (the old SIL_LIGHTSOFF branch washed
 500            #                every night-time care fx white)
 501        step = fx["step"]
 502        # an Assistant_Lights visit is the one anim that CAUSES the darkness --
 503        # and DVPet toggles the room at the anim's FINAL beat, so the whole
 504        # visit (switch AND the helper's exit) plays lit; the old cut at beat
 505        # 18 left the exit playing white in the dark (bug report 2026-07-13)
 506        lit_visit = fx["kind"] == "assist" and fx.get("act") == "lights"
 507        dark = not pet.lights and fx["kind"] != "evolve" and not lit_visit
 508        if dark:
 509            # the dark room stays dark through a care fx -- and DVPet's care
 510            # anims KEEP the fully-opaque lightsOff cover up (SpriteAnim sets
 511            # lightsOff inside the anims), so a dark-room fx shows NOTHING:
 512            # no pet, no props, no white poses (bug report 2026-07-13, "mon in
 513            # white poses during lights out sequence"); the sprite blank-out
 514            # happens just before update() once the kind painter has run
 515            bgimg, bg, on = None, VOID, SIL_LIGHTSOFF
 516        c = _FxCtx()
 517        c.px_h = SCREEN_ROWS * 2  # type: ignore
 518        c.bg, c.bgimg = bg, bgimg  # type: ignore
 519        # only clean/dying rely on this default pose; the kind painters override
 520        # `rows` unconditionally otherwise
 521        pose = {"clean": "idle", "dying": "exhausted"}.get(fx["kind"], "idle")
 522        c.rows = self._pose_rows(pet, pose, step // 2)  # type: ignore
 523        c.overlay = []  # type: ignore
 524        c.free = []          # type: ignore
 525        c.xshift = 0  # type: ignore
 526        c.yshift = 0  # type: ignore
 527        c.mirror = False  # type: ignore
 528        if fx["kind"] in ("eat", "cheer", "jeer", "spit"):
 529            # DVPet checkFilth runs inside these anims: piles stay visible and the
 530            # pet (and its food) stands clear of them.
 531            filth_pts, filth_clear = self._fx_filth(pet, self.frame_i)  # type: ignore
 532            c.overlay += filth_pts  # type: ignore
 533            c.xshift = filth_clear  # type: ignore
 534        elif fx["kind"] == "poop":
 535            # DVPet poop(): squat (+4, MIRRORED) clear of the old piles, net-zero
 536            # sway every 3 ticks; the new pile lands at t18 with the size-keyed
 537            # sound (fx snds) and the relieved pose (+5); ends 24.
 538            new = fx["step"] >= 18
 539            filth_pts, filth_clear = self._fx_filth(pet, self.frame_i,  # type: ignore
 540                                                    count=fx.get("poop", 0) + (1 if new else 0))
 541            c.overlay += filth_pts  # type: ignore
 542            sway = -1 if (3 <= step < 18 and (step // 3) % 2 == 1) else 0
 543            c.xshift = filth_clear + sway  # type: ignore
 544            c.rows = self._pose_rows_idx(pet, 5 if new else 4)  # type: ignore
 545        elif fx["kind"] in ("yawn", "poopdance"):
 546            # canon idle(): the tells are IDLE-family specials -- yawning()/
 547            # poopDance() move RELATIVE to the pet's LocX with its current
 548            # facing; only care anims resetScreen() to the anchor.  Seed the
 549            # roamer so the pose plays where the pet stands (walk-pose audit
 550            # 2026-07-08: it teleported to centre for every tell).
 551            c.xshift = self.roamer.xshift  # type: ignore
 552            c.mirror = self.roamer.mirror  # type: ignore
 553        painter = getattr(self, "_fxk_" + fx["kind"], None)
 554        if painter is not None:
 555            painter(pet, fx, step, c)
 556        if fx["kind"] in ("yawn", "poopdance"):
 557            # canon idle() keeps running stateAnims (checkStates/checkFilth/
 558            # stateNumTic) while a special idle plays: the scene actors (piles,
 559            # skull, Zzz) stay up and blinking through the pose.  Care anims
 560            # keep their resetScreen() blackout -- canon hides those labels.
 561            # The pose sways around the roamer's spot, so it gets the same
 562            # zone clamp as paint() (sickness striking MID-yawn would
 563            # otherwise stand the skull on the pet for the pose's last beats).
 564            lo = (_filth_right(pet.poop) if pet.poop else grid.X0) - PET_BASE_X
 565            cap = ((grid.X1 - SICK_ZONE if _sick_mark_up(pet) else grid.X1)
 566                   - SPRITE_W) - PET_BASE_X
 567            c.xshift = min(max(c.xshift, lo), max(cap, lo))  # type: ignore
 568            c.overlay += _effect_overlay(pet, self.frame_i // 4, SCREEN_COLS, c.px_h,  # type: ignore
 569                                         tick=self.frame_i)  # type: ignore
 570        if dark:                     # the opaque cover: black over everything
 571            c.rows, c.overlay, c.free, c.xshift, c.yshift = [], [], [], 0, 0  # type: ignore
 572        mirror = (c.mirror or fx["kind"] in ("dying", "poop")  # type: ignore
 573                  or (fx["kind"] == "gift" and GIFT_OUT <= step < GIFT_OUT + GIFT_BACK)  # facing right, ambling back
 574                  or (fx["kind"] == "spit" and (step // 6) % 2 == 0))   # refuse(): head-shake flips
 575        if c.yshift and c.rows:  # type: ignore
 576            # a hop may lift the sprite only as far as ITS clearance under the
 577            # band top -- nothing exits the window upward (LAW: off-screen is
 578            # left/right only).  A full-16px mon has no headroom, exactly like
 579            # the real 16px matrix: its excitement is the pose, not the air.
 580            ink = grid._crop(c.rows)  # type: ignore
 581            c.yshift = max(0, min(c.yshift, grid.BAND - len(ink)))  # type: ignore
 582        import tuipet.core.arena as _arena
 583        # render_screen resolves in ARENA's namespace: tuipet.arena.render_screen
 584        # is the documented spy/patch point, and it must catch fx frames too
 585        self.update(_arena.render_screen(c.rows, SCREEN_COLS, SCREEN_ROWS, on, c.bg,  # type: ignore
 586                                  xshift=c.xshift, yshift=c.yshift,  # type: ignore
 587                                  overlay=_clip_win(c.overlay),  # type: ignore
 588                                  bgimg=c.bgimg, mirror=mirror, clip=_WINDOW))  # type: ignore

 590    def _fxk_eat(self, pet: Any, fx: Any, step: Any, c: Any) -> None:
 591        # DVPet eat(): 24px food descends in 4 stages (beats 0/2/4/6) toward the
 592        # mouth, then a chew triad alternates open-mouth(+8)/chew(+7) at beats
 593        # 10/14/18/22/26/30 while the food is consumed frame-by-frame; ends ~34.
 594        food0 = self._food_frames(fx.get("icon") or "f:0")
 595        fw0 = len(food0[0][0]) if (food0 and food0[0] and food0[0][0]) else 8
 596        if getattr(pet, "poop", 0):
 597            # canon eat(): pad = _filthLabel.getSizeX() -- BOTH the food
 598            # (x31+pad) and the char (x55+pad) slide right by the FULL filth
 599            # width whenever piles exist, however narrow the block.  The old
 600            # clear-shift was 0 for 1-2 piles (their edge == the anchor), so
 601            # the food descended ONTO the slots (feeding audit 2026-07-08).
 602            # Piles stay visible: checkFilth runs inside eat().
 603            n = min(pet.poop, POOP_MAX_PILES)
 604            c.xshift = (_filth_right(n) - PET_BASE_X) + fw0
 605        elif c.xshift == 0:
 606            c.xshift = -1                                  # no filth: DVPet char x29 of 104 (~28%)
 607        # the food is BORN inside the window (Joel 2026-07-12: the descent's
 608        # left columns were getting cut at the matrix edge): slide the whole
 609        # canon-abutted pair (food right edge == char left edge) right until
 610        # the food's left edge sits at the window's -- x4 -- never past it
 611        c.xshift = max(c.xshift, (grid.X0 + fw0) - PET_BASE_X)
 612        ma = fx.get("munch_at")
 613        if ma is not None and step >= ma:
 614            # disposeFood: the pet turns away (pose 1, flipping at beat 5) and
 615            # the half-eaten food FALLS off the bottom of the screen
 616            d = step - ma
 617            c.rows = self._pose_rows_idx(pet, 1)
 618            c.mirror = d >= 5
 619            food = self._food_frames(fx.get("icon") or "f:0")
 620            if food:
 621                fr = food[min(1, len(food) - 1)]
 622                if fr:
 623                    fw = len(fr[0])
 624                    fy = 13 + max(0, d - 10)
 625                    if fy < grid.FLOOR:                    # crumbs stop AT the floor (no bottom exit)
 626                        c.overlay += _blit(fr, max(grid.X0, PET_BASE_X + c.xshift - fw), fy)
 627            return
 628        chew = fx.get("chew") or {10: 8, 14: 7, 18: 8, 22: 7, 26: 8, 30: 7}
 629        pose_i = 0
 630        for b in sorted(chew):
 631            if step >= b:
 632                pose_i = chew[b]
 633        c.rows = self._pose_rows_idx(pet, pose_i)
 634        food = food0
 635        if food:
 636            fw = fw0
 637            # DVPet: the food's RIGHT edge meets the pet's LEFT edge (foodLabel x31+24 == char x55),
 638            # so it descends right into the mouth -- abut it instead of stranding it on the far left.
 639            # (The filth pad above already moved BOTH food and char clear of the piles.)
 640            fx_x = max(grid.X0, PET_BASE_X + c.xshift - fw)
 641            stage = 0 if step < 2 else 1 if step < 4 else 2 if step < 6 else 3
 642            fy = (grid.TOP, 8, 11, 13)[stage]              # DVPet descent, mapped INTO the window: the
 643            #                                                  food drops from the band top to the mouth
 644            #                                                  (it used to enter from over the matrix --
 645            #                                                  LAW 2026-07-11: nothing crosses the top edge)
 646            fb = fx.get("food_beats") or (14, 22, 30)
 647            fi = 0 if step < fb[0] else 1 if step < fb[1] else 2 if step < fb[2] else 3
 648            c.overlay += _blit(food[min(fi, len(food) - 1)], fx_x, fy)

 399    def advance_fx(self) -> Any:
 400        if not self.fx:
 401            return False
 402        self.frame_i += 1        # type: ignore
 403        self.fx["step"] += 1
 404        if self.fx["step"] < self.fx["steps"]:
 405            return True
 406        kind = self.fx["kind"]
 407        had_filth = self.fx.get("poop", 0) > 0
 408        chain_eat = self.fx.get("chain_eat")
 409        pet_ref = self.fx.get("pet_ref")
 410        item_end = self.fx.get("end")
 411        self.fx = None  # type: ignore
 412        if kind == "item" and item_end:
 413            # canon: playing()/interact/study/... resolve into Cheering;
 414            # angrySurprise resolves into Jeering (itemfx script tables)
 415            self.start_fx(item_end, good=True)
 416        elif kind == "clean" and had_filth:
 417            # DVPet clean(): the cheer chains ONLY when filth was actually washed
 418            # (an empty-room wash just ends -- no celebration).
 419            self.start_fx("cheer")
 420        elif kind in ("evolve", "gift", "play", "inherit"):

 422            # giftEnd, jumping() frame 48, inheriting()'s strobe tail (branches
 423            # collapsed 2026-07-05; poopToilet left with the staple props,
 424            # strict-DSprite items 2026-07-17; the heal/bandage fx left with
 425            # the pill-anim fix 2026-07-18 -- the pill is eaten, meal-style)
 426            self.start_fx("cheer")
 427        elif kind == "assist" and chain_eat:
 428            # assistantFeed runs the STANDARD eat underneath (canon
 429            # eat(Assistant_Feed)); the meal is already on the floor, so the
 430            # chained eat skips its own descent stages
 431            self.start_fx("eat", chain_eat, pet=pet_ref,
 432                          starving=getattr(pet_ref, "_last_meal_starving", False))

 490    def _paint_fx(self, pet: Any) -> None:
 491        """The care-fx painter, decomposed (audit 2026-07): a shared context is
 492        pre-loaded (default pose, the cross-kind FILTH prelude),
 493        then the kind's own painter (_fxk_<kind>) mutates it.  Bodies verbatim
 494        from the old 210-line chain; behavior pinned by the fx golden."""
 495        fx = self.fx
 496        on, bg = LCD_ON, LCD_BG
 497        bgimg = self._background(pet)  # type: ignore
 498        if bgimg:
 499            on = SIL_SCENE   # dark silhouette day OR night, same rule as paint() --
 500            #                the pet is never white (the old SIL_LIGHTSOFF branch washed
 501            #                every night-time care fx white)
 502        step = fx["step"]
 503        # an Assistant_Lights visit is the one anim that CAUSES the darkness --
 504        # and DVPet toggles the room at the anim's FINAL beat, so the whole
 505        # visit (switch AND the helper's exit) plays lit; the old cut at beat
 506        # 18 left the exit playing white in the dark (bug report 2026-07-13)
 507        lit_visit = fx["kind"] == "assist" and fx.get("act") == "lights"
 508        dark = not pet.lights and fx["kind"] != "evolve" and not lit_visit
 509        if dark:
 510            # the dark room stays dark through a care fx -- and DVPet's care
 511            # anims KEEP the fully-opaque lightsOff cover up (SpriteAnim sets
 512            # lightsOff inside the anims), so a dark-room fx shows NOTHING:
 513            # no pet, no props, no white poses (bug report 2026-07-13, "mon in
 514            # white poses during lights out sequence"); the sprite blank-out
 515            # happens just before update() once the kind painter has run
 516            bgimg, bg, on = None, VOID, SIL_LIGHTSOFF
 517        c = _FxCtx()
 518        c.px_h = SCREEN_ROWS * 2  # type: ignore
 519        c.bg, c.bgimg = bg, bgimg  # type: ignore
 520        # only clean/dying rely on this default pose; the kind painters override
 521        # `rows` unconditionally otherwise
 522        pose = {"clean": "idle", "dying": "exhausted"}.get(fx["kind"], "idle")
 523        c.rows = self._pose_rows(pet, pose, step // 2)  # type: ignore
 524        c.overlay = []  # type: ignore
 525        c.free = []          # type: ignore
 526        c.xshift = 0  # type: ignore
 527        c.yshift = 0  # type: ignore
 528        c.mirror = False  # type: ignore
 529        if fx["kind"] in ("eat", "cheer", "jeer", "spit"):
 530            # DVPet checkFilth runs inside these anims: piles stay visible and the
 531            # pet (and its food) stands clear of them.
 532            filth_pts, filth_clear = self._fx_filth(pet, self.frame_i)  # type: ignore
 533            c.overlay += filth_pts  # type: ignore
 534            c.xshift = filth_clear  # type: ignore
 535        elif fx["kind"] == "poop":
 536            # DVPet poop(): squat (+4, MIRRORED) clear of the old piles, net-zero
 537            # sway every 3 ticks; the new pile lands at t18 with the size-keyed
 538            # sound (fx snds) and the relieved pose (+5); ends 24.
 539            new = fx["step"] >= 18
 540            filth_pts, filth_clear = self._fx_filth(pet, self.frame_i,  # type: ignore
 541                                                    count=fx.get("poop", 0) + (1 if new else 0))
 542            c.overlay += filth_pts  # type: ignore
 543            sway = -1 if (3 <= step < 18 and (step // 3) % 2 == 1) else 0
 544            c.xshift = filth_clear + sway  # type: ignore
 545            c.rows = self._pose_rows_idx(pet, 5 if new else 4)  # type: ignore
 546        elif fx["kind"] in ("yawn", "poopdance"):
 547            # canon idle(): the tells are IDLE-family specials -- yawning()/
 548            # poopDance() move RELATIVE to the pet's LocX with its current
 549            # facing; only care anims resetScreen() to the anchor.  Seed the
 550            # roamer so the pose plays where the pet stands (walk-pose audit
 551            # 2026-07-08: it teleported to centre for every tell).
 552            c.xshift = self.roamer.xshift  # type: ignore
 553            c.mirror = self.roamer.mirror  # type: ignore
 554        painter = getattr(self, "_fxk_" + fx["kind"], None)
 555        if painter is not None:
 556            painter(pet, fx, step, c)
 557        if fx["kind"] in ("yawn", "poopdance"):
 558            # canon idle() keeps running stateAnims (checkStates/checkFilth/
 559            # stateNumTic) while a special idle plays: the scene actors (piles,
 560            # skull, Zzz) stay up and blinking through the pose.  Care anims
 561            # keep their resetScreen() blackout -- canon hides those labels.
 562            # The pose sways around the roamer's spot, so it gets the same
 563            # zone clamp as paint() (sickness striking MID-yawn would
 564            # otherwise stand the skull on the pet for the pose's last beats).
 565            lo = (_filth_right(pet.poop) if pet.poop else grid.X0) - PET_BASE_X
 566            cap = ((grid.X1 - SICK_ZONE if _sick_mark_up(pet) else grid.X1)
 567                   - SPRITE_W) - PET_BASE_X
 568            c.xshift = min(max(c.xshift, lo), max(cap, lo))  # type: ignore
 569            c.overlay += _effect_overlay(pet, self.frame_i // 4, SCREEN_COLS, c.px_h,  # type: ignore
 570                                         tick=self.frame_i)  # type: ignore
 571        if dark:                     # the opaque cover: black over everything
 572            c.rows, c.overlay, c.free, c.xshift, c.yshift = [], [], [], 0, 0  # type: ignore
 573        mirror = (c.mirror or fx["kind"] in ("dying", "poop")  # type: ignore
 574                  or (fx["kind"] == "gift" and GIFT_OUT <= step < GIFT_OUT + GIFT_BACK)  # facing right, ambling back
 575                  or (fx["kind"] == "spit" and (step // 6) % 2 == 0))   # refuse(): head-shake flips
 576        if c.yshift and c.rows:  # type: ignore
 577            # a hop may lift the sprite only as far as ITS clearance under the
 578            # band top -- nothing exits the window upward (LAW: off-screen is
 579            # left/right only).  A full-16px mon has no headroom, exactly like
 580            # the real 16px matrix: its excitement is the pose, not the air.
 581            ink = grid._crop(c.rows)  # type: ignore
 582            c.yshift = max(0, min(c.yshift, grid.BAND - len(ink)))  # type: ignore
 583        import tuipet.core.arena as _arena
 584        # render_screen resolves in ARENA's namespace: tuipet.arena.render_screen
 585        # is the documented spy/patch point, and it must catch fx frames too
 586        self.update(_arena.render_screen(c.rows, SCREEN_COLS, SCREEN_ROWS, on, c.bg,  # type: ignore
 587                                  xshift=c.xshift, yshift=c.yshift,  # type: ignore
 588                                  overlay=_clip_win(c.overlay),  # type: ignore
 589                                  bgimg=c.bgimg, mirror=mirror, clip=_WINDOW))  # type: ignore

 591    def _fxk_eat(self, pet: Any, fx: Any, step: Any, c: Any) -> None:
 592        # DVPet eat(): 24px food descends in 4 stages (beats 0/2/4/6) toward the
 593        # mouth, then a chew triad alternates open-mouth(+8)/chew(+7) at beats
 594        # 10/14/18/22/26/30 while the food is consumed frame-by-frame; ends ~34.
 595        food0 = self._food_frames(fx.get("icon") or "f:0")
 596        fw0 = len(food0[0][0]) if (food0 and food0[0] and food0[0][0]) else 8
 597        if getattr(pet, "poop", 0):
 598            # canon eat(): pad = _filthLabel.getSizeX() -- BOTH the food
 599            # (x31+pad) and the char (x55+pad) slide right by the FULL filth
 600            # width whenever piles exist, however narrow the block.  The old
 601            # clear-shift was 0 for 1-2 piles (their edge == the anchor), so
 602            # the food descended ONTO the slots (feeding audit 2026-07-08).
 603            # Piles stay visible: checkFilth runs inside eat().
 604            n = min(pet.poop, POOP_MAX_PILES)
 605            c.xshift = (_filth_right(n) - PET_BASE_X) + fw0
 606        elif c.xshift == 0:
 607            c.xshift = -1                                  # no filth: DVPet char x29 of 104 (~28%)
 608        # the food is BORN inside the window (Joel 2026-07-12: the descent's
 609        # left columns were getting cut at the matrix edge): slide the whole
 610        # canon-abutted pair (food right edge == char left edge) right until
 611        # the food's left edge sits at the window's -- x4 -- never past it
 612        c.xshift = max(c.xshift, (grid.X0 + fw0) - PET_BASE_X)
 613        ma = fx.get("munch_at")
 614        if ma is not None and step >= ma:
 615            # disposeFood: the pet turns away (pose 1, flipping at beat 5) and
 616            # the half-eaten food FALLS off the bottom of the screen
 617            d = step - ma
 618            c.rows = self._pose_rows_idx(pet, 1)
 619            c.mirror = d >= 5
 620            food = self._food_frames(fx.get("icon") or "f:0")
 621            if food:
 622                fr = food[min(1, len(food) - 1)]
 623                if fr:
 624                    fw = len(fr[0])
 625                    fy = 13 + max(0, d - 10)
 626                    if fy < grid.FLOOR:                    # crumbs stop AT the floor (no bottom exit)
 627                        c.overlay += _blit(fr, max(grid.X0, PET_BASE_X + c.xshift - fw), fy)
 628            return
 629        chew = fx.get("chew") or {10: 8, 14: 7, 18: 8, 22: 7, 26: 8, 30: 7}
 630        pose_i = 0
 631        for b in sorted(chew):
 632            if step >= b:
 633                pose_i = chew[b]
 634        c.rows = self._pose_rows_idx(pet, pose_i)
 635        food = food0
 636        if food:
 637            fw = fw0
 638            # DVPet: the food's RIGHT edge meets the pet's LEFT edge (foodLabel x31+24 == char x55),
 639            # so it descends right into the mouth -- abut it instead of stranding it on the far left.
 640            # (The filth pad above already moved BOTH food and char clear of the piles.)
 641            fx_x = max(grid.X0, PET_BASE_X + c.xshift - fw)
 642            stage = 0 if step < 2 else 1 if step < 4 else 2 if step < 6 else 3
 643            fy = (grid.TOP, 8, 11, 13)[stage]              # DVPet descent, mapped INTO the window: the
 644            #                                                  food drops from the band top to the mouth
 645            #                                                  (it used to enter from over the matrix --
 646            #                                                  LAW 2026-07-11: nothing crosses the top edge)
 647            fb = fx.get("food_beats") or (14, 22, 30)
 648            fi = 0 if step < fb[0] else 1 if step < fb[1] else 2 if step < fb[2] else 3
 649            c.overlay += _blit(food[min(fi, len(food) - 1)], fx_x, fy)

 671    def _fxk_assist(self, pet: Any, fx: Any, step: Any, c: Any) -> None:

 673        # descends from the top on the LEFT (locX 6, icon flipped to face the
 674        # pet), does its round, and rises away (moveUp beats 18/19).  Mapped to
 675        # 28 steps: descend 0-7, act 8-19 (with the wiggle), leave 20+.  During
 676        # a clean the piles sweep RIGHT off-screen (filthLabel.moveRight(4) each
 677        # descent beat) -- the OPPOSITE of your wash, which shoves them left.

 680        cleaning = act == "clean" and fx.get("poop")

 682            # canon assistantClean: filthLabel.moveRight(4) sweeps the mess RIGHT
 683            # while adjustCharacterForFilth() clamps the pet to the filth's right
 684            # edge every beat -- so the pet is shoved right AHEAD of the piles and

 686            # give-ground xshift (max 8) ignored the filth entirely, so a full
 687            # 4-pile block (16px) sat UNDER a sleeping pet auto-care was cleaning
 688            # -- the poop-overlap glitch (audit 2026-07-08).
 689            push = -step * 3                                   # filth marches right
 690            c.xshift = _filth_right(fx["poop"]) - push - PET_BASE_X   # stay clear of its right edge
 691            c.overlay += _filth_pts(pet, self.frame_i, count=fx["poop"],  # type: ignore
 692                                    sizes=fx.get("sizes"), push=push, px_h=c.px_h)

 696            # 4+16 | 20+16 fills the grid band x[4,36) with both sprites abutted
 697            if step < 8:
 698                c.xshift = min(8, step * 2)
 699            elif step < 20:
 700                c.xshift = 8
 701            else:
 702                c.xshift = max(0, 8 - (step - 19) * 2)
 703        if pet.asleep:
 704            c.rows = self._pose_rows(pet, "sleep", step // 2)
 705        _, by_num = data.load_sprites()
 706        rec = by_num.get(fx.get("helper", -1))
 707        fr = rec["frames"] if rec else None
 708        hf = (fr[0] if fr and fr[0] else next((f for f in fr if f), None)) if fr else None
 709        hh = len(hf) if hf else 16
 710        ground = c.px_h - hh - 2
 711        if feed:
 712            # canon assistantFeed: the helper walks the MEAL in -- descends
 713            # WITH the food, sets it down at the eat spot, and EXITS LEFT by
 714            # beat 10 (moveLeft 24 x2); the standard eat anim (chained) takes
 715            # it from there
 716            food = self._food_frames(fx.get("icon") or "f:44")
 717            f0 = food[0] if (food and food[0]) else None
 718            fw = len(f0[0]) if f0 else 8
 719            fx_x = max(0, PET_BASE_X - 1 - fw)             # the eat anim's food spot
 720            hff = [row[::-1] for row in hf] if hf else None   # faces the pet
 721            if step < 7:                                   # the walk-in
 722                hy = -hh + (step + 1) * (ground + hh) // 7
 723                hx = 2
 724                if hff and hy > -hh:
 725                    c.overlay += _blit(hff, hx, hy)
 726                if f0:
 727                    c.overlay += _blit(f0, hx + 10, max(0, hy + hh // 2))
 728            else:                                          # set down + exit left
 729                hx = 2 - (step - 6) * 8
 730                if hf and hx > -20:                        # native facing: walking LEFT out
 731                    c.overlay += _blit(hf, hx, ground)
 732                if f0:
 733                    c.overlay += _blit(f0, fx_x, c.px_h - 2 - len(f0))

 745                c.overlay += _blit(hf, hx, hy)

 747    def _fxk_cheer(self, pet: Any, fx: Any, step: Any, c: Any) -> None:
 748        # DVPet cheer(): pose alternates up(+5)/down(+7) every 6 intervals with a
 749        # "happy" emote bubble pulsing on the up-beats; ends ~beat 30.  A
 750        # spoiling Bad_Praise (cheer(false)) bounces on 6/4 instead of 5/7.
