 46def _clamp_card(card: Any) -> Any:
 47    """Bound an UNTRUSTED battle card to what the game can actually produce.
 48    Module-level because seeded PvP feeds BOTH engines identically-clamped
 49    cards -- including one's own, clamped exactly as the peer clamps it."""
 50    def _n(v: Any, d: int=0, lo: int=0, hi: int=999) -> Any:
 51        if not isinstance(v, (int, float)) or isinstance(v, bool):
 52            v = d
 53        return max(lo, min(int(v), hi))
 54    card = dict(card)
 55    card["strength_max"] = _n(card.get("strength_max"), 4, 1, 9)
 56    card["strength"] = _n(card.get("strength"), 4, 0, card["strength_max"])
 57    card["hunger_max"] = _n(card.get("hunger_max"), 4, 1, 9)
 58    card["hunger"] = _n(card.get("hunger"), 4, 0, card["hunger_max"])
 59    card["energy_max"] = _n(card.get("energy_max"), 5, 1, 2000)
 60    card["energy"] = _n(card.get("energy"), 0, 0, card["energy_max"])
 61    card["weight"] = _n(card.get("weight"), 10, 1, 999)
 62    card["base_weight"] = _n(card.get("base_weight"), 10, 1, 999)
 63    card["trainings_cur"] = _n(card.get("trainings_cur"), 0, 0, 999)
 64    card["trainings_total"] = _n(card.get("trainings_total"), 0, 0, 9999)
 65    card["battles"] = _n(card.get("battles"), 0, 0, 10 ** 6)
 66    card["wins"] = _n(card.get("wins"), 0, 0, card["battles"])
 67    if card.get("hit_type") not in ("mega", "normal", "miss"):
 68        card["hit_type"] = "miss"
 69    card["num"] = _n(card.get("num"), 0, 0, 10 ** 6)
 70    # stage/attribute come from the SPECIES RECORD of the claimed num, never
 71    # the wire: a forged {"stage": "Special"} ranked 14.5 and auto-won every
 72    # online bout while the sprite still showed a Child (audit 2026-07-15).
 73    # Unknown nums (a newer build's dex) keep a KNOWN claimed stage so the
 74    # two engines stay symmetric; garbage stages already ranked as "Child".
 75    rec = data.record_for(card["num"])
 76    if not rec.get("_placeholder"):
 77        card["stage"] = rec.get("stage", "Rookie")
 78        card["attribute"] = rec.get("attribute", "Free")
 79    else:
 80        # unknown num (a newer build's dex): keep a KNOWN claimed stage so the
 81        # two engines stay symmetric.  Anything unrankable -- "Special"
 82        # included, whose old re-map elif was dead behind this membership
 83        # check -- lands on "Rookie": the same rank 3 _RANK's default already
 84        # gave it, now wearing the tuipet word instead of "Child"
 85        if card.get("stage") not in battle._RANK:
 86            card["stage"] = "Rookie"
 87        if card.get("attribute") not in ("Vaccine", "Virus", "Data", "Free"):
 88            card["attribute"] = "Free"
 89    card["hp"] = 5
 90    return card

 93class BoutMixin:
 94    def _battle_begin(self, opp_card: Any, commit: Optional[Any]=None) -> None:
 95        """Cards crossed: clamp the untrusted card, verify the proto, and
 96        reveal my nonce.  The fight itself is SEEDED SYMMETRIC: both clients
 97        run the identical precomputed engine on identically-clamped cards, so
 98        no result ever crosses the wire."""
 99        opp_card = _clamp_card(opp_card)
100        self.opp_card = opp_card
101        try:
102            proto_ok = int(opp_card.get("proto") or 0) >= 3 and bool(commit)
103        except (TypeError, ValueError):
104            proto_ok = False
105        if not proto_ok:
106            self.bt_outcome = "Battle void — version mismatch."
107            self.bt_payload = ("battle_msg", "Battle void — the other side "
108                              "runs an older tuipet.")
109            self.bphase = "over"
110            if self.partner:  # type: ignore
111                self.client.relay(self.partner[0], {"kind": "battle", "abort": True})  # type: ignore
112            return
113        self.bt_peer_commit = commit
114        self.my_max = self.my_hp = 5
115        self.opp_max = self.opp_hp = 5
116        self.bphase = "wait"
117        # the nonce reveal is its own message now (no picks in this world)
118        self.client.relay(self.partner[0],  # type: ignore
119                          {"kind": "battle", "t": "pick", "nonce": self.bt_nonce})  # type: ignore
120        self._maybe_build()
121    def _maybe_build(self) -> None:
122        """Both nonces in -> verify the commit, seed the shared engine, and
123        precompute the whole 5-round fight (both sides independently)."""
124        if (self.battle is not None or self.bt_peer_nonce is None  # type: ignore
125                or self.opp_card is None):  # type: ignore  # type: ignore
126            return
127        pn = self.bt_peer_nonce  # type: ignore
128        if hashlib.sha256(str(pn).encode()).hexdigest() != (self.bt_peer_commit or ""):
129            self.bt_outcome = "Battle void — bad checksum."
130            self.bt_payload = ("battle_msg", "Battle void — bad checksum.")
131            self.bphase = "over"
132            if self.partner:  # type: ignore
133                self.client.relay(self.partner[0], {"kind": "battle", "abort": True})  # type: ignore
134            return
135        hn, gn = (self.bt_nonce, pn) if self.is_host else (pn, self.bt_nonce)  # type: ignore
136        seed = int.from_bytes(hashlib.sha256(f"{hn}:{gn}".encode()).digest()[:8], "big")
137        host_card, guest_card = ((self.bt_my_card, self.opp_card) if self.is_host  # type: ignore
138                                 else (self.opp_card, self.bt_my_card))  # type: ignore
139        rng = random.Random(seed).random
140        host = battle.Side.of_card(dict(host_card))
141        guest = battle.Side.of_card(dict(guest_card))
142        seq, hhp, ghp = battle.generate(host, guest, rounds=battle.ROUNDS_ONLINE,
143                                        rng=rng)
144        self.battle = {"seq": seq, "host_hp": 5, "guest_hp": 5, "i": 0}
145        self.bphase = "fight"
146        self._play_next_round()
147    def _play_next_round(self) -> None:
148        """Advance the precomputed fight one round and stage its volley."""
149        b = self.battle
150        if b is None:
151            return
152        if b["i"] >= len(b["seq"]) or b["host_hp"] <= 0 or b["guest_hp"] <= 0:
153            self._battle_over()
154            return
155        h_hit, h_dmg, g_hit, g_dmg = b["seq"][b["i"]]
156        b["i"] += 1
157        if h_hit:
158            b["guest_hp"] = max(0, b["guest_hp"] - h_dmg)
159        if g_hit:
160            b["host_hp"] = max(0, b["host_hp"] - g_dmg)
161        if self.is_host:  # type: ignore
162            dealt = h_dmg if h_hit else 0
163            taken = g_dmg if g_hit else 0
164        else:
165            dealt = g_dmg if g_hit else 0
166            taken = h_dmg if h_hit else 0
167        my0, opp0 = self.my_hp, self.opp_hp
168        self.my_hp = b["host_hp"] if self.is_host else b["guest_hp"]  # type: ignore
169        self.opp_hp = b["guest_hp"] if self.is_host else b["host_hp"]  # type: ignore
170        self._stage_volley(my0, opp0, dealt, taken)
171        self.bt_log = f"you \u2192 {dealt} dmg\n  them \u2192 {taken} dmg"
172    def _battle_over(self) -> None:
173        b = self.battle
174        my_hp = b["host_hp"] if self.is_host else b["guest_hp"]  # type: ignore
175        opp_hp = b["guest_hp"] if self.is_host else b["host_hp"]  # type: ignore
176        won = opp_hp <= 0 and my_hp > 0
177        draw = (my_hp > 0 and opp_hp > 0 and my_hp == opp_hp) \
178            or (my_hp <= 0 and opp_hp <= 0)
179        if not won and not draw and my_hp > opp_hp:
180            won = True                     # rounds ran dry: higher HP stands
181        self.bphase = "over"
182        self.sfx = "attack"
183        if self.partner:                   # type: ignore
184            persistence.record_connection(self.partner[1])  # type: ignore
185        # the ladder needs BOTH stories: the winner's claim only credits when
186        # the LOSER's agreeing report lands too, keyed by ACCOUNT name -- the
187        # old code filed only on won (nobody ever confirmed) and led with the
188        # PET name (which the server doesn't know), so the monthly ladder
189        # never credited a single win (audit 2026-07-15)
190        opp_nm = (self.partner or (0, ""))[1] or (self.opp_card or {}).get("name")  # type: ignore
191        report = getattr(self.client, "ladder_report", None)  # type: ignore
192        if report and opp_nm and not draw:
193            report(won, opp_nm)
194        from tuipet.core.pet import online_reward
195        purse = online_reward(won, draw=draw)
196        self.pet.record_battle(won and not draw, online=True)  # type: ignore
197        # pet.add_bits died with the classic revert (v0.5.0) -- every online
198        # payout has crashed on BOTH sides since; caught by the live two-bot
199        # smoke 2026-07-17.  The raid claim's idiom is the house style.
200        self.pet.bits += int(purse)  # type: ignore
201        self.bt_outcome = ("DRAW" if draw
202                           else "\u2605 YOU WIN! \u2605" if won else "YOU LOSE\u2026")
203        # ask the calendar, not the amount: the weekend loss purse (150)
204        # equals the plain draw purse, which hid the tag on weekend losses
205        from tuipet.core.pet import weekend_bonus
206        self.bt_reward = f"+{purse}b" + ("  (weekend bonus!)" if weekend_bonus() > 1 else "") + _evo_note(self.pet)  # type: ignore
207        self.bt_payload = ("battle_msg", self.bt_outcome)
208    def _stage_volley(self, my0: Any, opp0: Any, dealt: Any, taken: Any) -> None:
209        """A presentation-only BattlePanel replays the round: my pet RIGHT,
210        the opponent's LEFT, orbs/hit/dodge from the engine's numbers."""
211        try:
212            card = dict(self.opp_card or {})
213            if not card.get("num"):
214                return
215            show = battlescreen.BattlePanel(self.pet, enemy=card)  # type: ignore
216            show.foe_attr = card.get("attribute", "Free")
217            show.timeline = battlescreen.round_timeline(my0, opp0, dealt, taken, True)
218            show.i = 0
219            show.phase = "anim"
220            show._last_m = None
221            self.bshow = show
222        except Exception:
223            self.bshow = None               # type: ignore
224    def _key_battle(self, k: Any) -> Any:
225        if self.bshow is not None:              # the round is replaying
226            if k in ("space", "enter", "escape"):
227                self.bshow = None               # type: ignore
228            return None
229        if self.bphase == "over":
230            if k in ("enter", "space", "escape"):
231                self._return_to_lobby(self.bt_outcome)
232            return None
233        if self.bphase == "fight":
234            if k in ("space", "enter"):
235                self._play_next_round()
236            elif k == "escape":
237                self._forfeit()
238            return None
239        if self.bphase in ("card", "wait") and k == "escape":
240            self._forfeit()
241        return None
242    def _forfeit(self) -> None:
243        """Leave a battle in progress -> tell the opponent, then back to the lobby.
244        Once the seeded fight is RUNNING the bout is committed: walking out is
245        a forfeit LOSS, filed with the ladder like any decided bout -- ESC in
246        "fight" used to relay only the abort, so a losing player could void
247        the result at will while the winner's half expired unconfirmed
248        (MED audit 2026-07-19).  Pre-commit (card/wait) stays a free back-out."""
249        counted = self.bphase == "fight" and self.battle is not None
250        if counted:
251            opp_nm = (self.partner or (0, ""))[1] or (self.opp_card or {}).get("name")  # type: ignore
252            report = getattr(self.client, "ladder_report", None)  # type: ignore
253            if report and opp_nm:
254                report(False, opp_nm)
255            self.pet.record_battle(False, online=True)  # type: ignore
256        if self.partner:  # type: ignore
257            self.client.relay(self.partner[0], {"kind": "battle", "abort": True})  # type: ignore
258        self._return_to_lobby("Forfeit — counted as a loss." if counted  # type: ignore
259                              else "You forfeited.")
260    def _opp_fled(self) -> None:
261        """The partner walked out of a RUNNING seeded fight (abort relay or
262        roster vanish): that is THEIR forfeit.  File the winner's half so the
263        pair credits when their agreeing loss lands, and pay the win -- both
264        sites used to call it void, so the ladder never heard of it (MED
265        audit 2026-07-19)."""
266        opp_nm = (self.partner or (0, ""))[1] or (self.opp_card or {}).get("name")  # type: ignore
267        report = getattr(self.client, "ladder_report", None)  # type: ignore
268        if report and opp_nm:
269            report(True, opp_nm)
270        from tuipet.core.pet import online_reward
271        purse = online_reward(True)
272        self.pet.record_battle(True, online=True)  # type: ignore
273        self.pet.bits += int(purse)  # type: ignore
274        self.bt_outcome = "Opponent fled — you win!"
275        from tuipet.core.pet import weekend_bonus
276        self.bt_reward = f"+{purse}b" + ("  (weekend bonus!)"      # calendar, not amount
277                                         if weekend_bonus() > 1 else "") + _evo_note(self.pet)  # type: ignore
278        self.bt_payload = ("battle_msg", self.bt_outcome)
279        self.bphase = "over"

296    def _key_jogress(self, k: Any) -> Any:
297        if self.jphase == "result":  # type: ignore
298            if self.jshow is not None and self.jshow.phase == "fusing":  # type: ignore
299                self.jshow.key(k)                       # type: ignore
300                return None
301            if not self.j_peer_two_phase:  # type: ignore
302                # LEGACY peer (pre-v0.2.350): its client commits on any key with
303                # no decline -- mirror it, or a mixed-version pair goes one-sided
304                if k in ("enter", "space", "escape"):
305                    self._commit_fusion()
306                return None
307            # two-phase commit (consent audit 2026-07-07): the fusion is
308            # PERMANENT -- both players must say yes at the result screen
309            if k in ("enter", "space") and not self.j_confirmed:  # type: ignore
310                self.j_confirmed = True
311                self.client.relay(self.partner[0], {"kind": "jogress", "t": "confirm"})  # type: ignore
312                if self.j_partner_confirmed:  # type: ignore
313                    self._commit_fusion()
314                else:
315                    self.status = "Waiting for the partner…"
316            elif k == "escape":
317                # a real DECLINE: nobody fuses, both sides told
318                self.client.relay(self.partner[0], {"kind": "jogress", "t": "decline"})  # type: ignore
319                self._return_to_lobby("Fusion declined — no one fused.")  # type: ignore
320            return None
321        if self.jphase == "failed":  # type: ignore
322            if k in ("enter", "space", "escape"):
323                self._return_to_lobby(self.fail_reason or "No resonance.")  # type: ignore
324            return None
325        if k == "escape":                                  # cancel mid-fusion -> free the partner
326            if self.partner:  # type: ignore
327                self.client.relay(self.partner[0], {"kind": "jogress", "abort": True})  # type: ignore
328            self._return_to_lobby("Fusion cancelled.")  # type: ignore
329        return None

114        self.bt_peer_commit = commit
115        self.my_max = self.my_hp = 5
116        self.opp_max = self.opp_hp = 5
117        self.bphase = "wait"
118        # the nonce reveal is its own message now (no picks in this world)
119        self.client.relay(self.partner[0],  # type: ignore
120                          {"kind": "battle", "t": "pick", "nonce": self.bt_nonce})  # type: ignore
121        self._maybe_build()
122    def _maybe_build(self) -> None:
123        """Both nonces in -> verify the commit, seed the shared engine, and
124        precompute the whole 5-round fight (both sides independently)."""
125        if (self.battle is not None or self.bt_peer_nonce is None  # type: ignore
126                or self.opp_card is None):  # type: ignore  # type: ignore
127            return
128        pn = self.bt_peer_nonce  # type: ignore
129        if hashlib.sha256(str(pn).encode()).hexdigest() != (self.bt_peer_commit or ""):
130            self.bt_outcome = "Battle void — bad checksum."
131            self.bt_payload = ("battle_msg", "Battle void — bad checksum.")
132            self.bphase = "over"
133            if self.partner:  # type: ignore
134                self.client.relay(self.partner[0], {"kind": "battle", "abort": True})  # type: ignore
135            return
136        hn, gn = (self.bt_nonce, pn) if self.is_host else (pn, self.bt_nonce)  # type: ignore
137        seed = int.from_bytes(hashlib.sha256(f"{hn}:{gn}".encode()).digest()[:8], "big")
138        host_card, guest_card = ((self.bt_my_card, self.opp_card) if self.is_host  # type: ignore
139                                 else (self.opp_card, self.bt_my_card))  # type: ignore
140        rng = random.Random(seed).random
141        host = battle.Side.of_card(dict(host_card))
142        guest = battle.Side.of_card(dict(guest_card))
143        seq, hhp, ghp = battle.generate(host, guest, rounds=battle.ROUNDS_ONLINE,
144                                        rng=rng)
145        self.battle = {"seq": seq, "host_hp": 5, "guest_hp": 5, "i": 0}
146        self.bphase = "fight"
147        self._play_next_round()
148    def _play_next_round(self) -> None:
149        """Advance the precomputed fight one round and stage its volley."""
150        b = self.battle
151        if b is None:
152            return
153        if b["i"] >= len(b["seq"]) or b["host_hp"] <= 0 or b["guest_hp"] <= 0:
154            self._battle_over()
155            return
156        h_hit, h_dmg, g_hit, g_dmg = b["seq"][b["i"]]
157        b["i"] += 1
158        if h_hit:
159            b["guest_hp"] = max(0, b["guest_hp"] - h_dmg)
160        if g_hit:
161            b["host_hp"] = max(0, b["host_hp"] - g_dmg)
162        if self.is_host:  # type: ignore
163            dealt = h_dmg if h_hit else 0
164            taken = g_dmg if g_hit else 0
165        else:
166            dealt = g_dmg if g_hit else 0
167            taken = h_dmg if h_hit else 0
168        my0, opp0 = self.my_hp, self.opp_hp
169        self.my_hp = b["host_hp"] if self.is_host else b["guest_hp"]  # type: ignore
170        self.opp_hp = b["guest_hp"] if self.is_host else b["host_hp"]  # type: ignore
171        self._stage_volley(my0, opp0, dealt, taken)
172        self.bt_log = f"you \u2192 {dealt} dmg\n  them \u2192 {taken} dmg"
173    def _battle_over(self) -> None:
174        b = self.battle
175        my_hp = b["host_hp"] if self.is_host else b["guest_hp"]  # type: ignore
176        opp_hp = b["guest_hp"] if self.is_host else b["host_hp"]  # type: ignore
177        won = opp_hp <= 0 and my_hp > 0
178        draw = (my_hp > 0 and opp_hp > 0 and my_hp == opp_hp) \
179            or (my_hp <= 0 and opp_hp <= 0)
180        if not won and not draw and my_hp > opp_hp:
181            won = True                     # rounds ran dry: higher HP stands
182        self.bphase = "over"
183        self.sfx = "attack"
184        if self.partner:                   # type: ignore
185            persistence.record_connection(self.partner[1])  # type: ignore
186        # the ladder needs BOTH stories: the winner's claim only credits when
187        # the LOSER's agreeing report lands too, keyed by ACCOUNT name -- the
188        # old code filed only on won (nobody ever confirmed) and led with the
189        # PET name (which the server doesn't know), so the monthly ladder
190        # never credited a single win (audit 2026-07-15)
191        opp_nm = (self.partner or (0, ""))[1] or (self.opp_card or {}).get("name")  # type: ignore
192        report = getattr(self.client, "ladder_report", None)  # type: ignore
193        if report and opp_nm and not draw:
194            report(won, opp_nm)
195        from tuipet.core.pet import online_reward
196        purse = online_reward(won, draw=draw)
197        self.pet.record_battle(won and not draw, online=True)  # type: ignore
198        # pet.add_bits died with the classic revert (v0.5.0) -- every online
199        # payout has crashed on BOTH sides since; caught by the live two-bot
200        # smoke 2026-07-17.  The raid claim's idiom is the house style.
201        self.pet.bits += int(purse)  # type: ignore
202        self.bt_outcome = ("DRAW" if draw
203                           else "\u2605 YOU WIN! \u2605" if won else "YOU LOSE\u2026")
204        # ask the calendar, not the amount: the weekend loss purse (150)
205        # equals the plain draw purse, which hid the tag on weekend losses
206        from tuipet.core.pet import weekend_bonus
207        self.bt_reward = f"+{purse}b" + ("  (weekend bonus!)" if weekend_bonus() > 1 else "") + _evo_note(self.pet)  # type: ignore
208        self.bt_payload = ("battle_msg", self.bt_outcome)
