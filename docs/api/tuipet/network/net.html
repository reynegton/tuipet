tuipet.network.net

Lobby network client — the tuipet side of the multiplayer relay.

A LobbyClient runs as one asyncio task inside Textual's event loop. It owns a LobbyState snapshot that the lobby Panel reads on each render tick (everything runs on the one loop, so no locking). Outgoing messages are fire-and-forget via a queue; incoming messages update the snapshot or land in inbox for the session logic (jogress/battle) to drain.

 38class _WsClient:
 39    """The transport loop BOTH clients ride: connect -> login -> pump send +
 40    recv -> on a drop, retry with exponential backoff; a rejected login sets
 41    _stop and ends the loop for good.  This loop lived in two drifting copies
 42    (the send-loop `return`-kills-autosave bug lived in exactly this drift;
 43    refactor 2026-07-05).  Subclasses supply the login payload, their own
 44    _send_loop/_handle, and the connection-state bookkeeping hooks."""
 45
 46    _backoff0 = 2.0
 47    _backoff_cap = 30.0
 48    # a dead host must fail FAST into the retry path: the library default
 49    # (~10s) froze "Connecting…" for the whole wait.  submit_bug and
 50    # cloudsync always capped their opens; the LIVE loop was the only one
 51    # left uncapped (QOL sweep 2026-07-23).
 52    _open_timeout = 6.0
 53
 54    def retry_now(self) -> None:
 55        """Hurry the backoff: the player knows the wifi is back."""
 56        self._hurry = True
 57
 58    async def run(self) -> None:
 59        backoff = self._backoff0
 60        self._hurry = False
 61        while not self._stop:  # type: ignore
 62            try:
 63                async with websockets.connect(self.uri, max_size=WIRE_READ_MAX,  # type: ignore
 64                                              open_timeout=self._open_timeout) as ws:
 65                    self._ws = ws
 66                    await ws.send(json.dumps(self._login_msg()))  # type: ignore
 67                    self._on_connect()  # type: ignore
 68                    backoff = self._backoff0
 69                    sender = asyncio.create_task(self._send_loop())  # type: ignore
 70                    try:
 71                        async for raw in ws:
 72                            self._handle(raw)  # type: ignore
 73                    finally:
 74                        sender.cancel()
 75            except Exception as e:
 76                self._on_error(e)
 77            finally:
 78                self._on_disconnect()  # type: ignore
 79                self._ws = None  # type: ignore
 80            if self._stop:  # type: ignore
 81                break
 82            self._on_retry_wait()
 83            # sliced sleep so retry_now() (a UI-thread bool flip) can cut a
 84            # 30s backoff short -- the player watching "reconnecting…" had
 85            # no key to hurry it (QOL sweep 2026-07-23)
 86            waited = 0.0
 87            while waited < backoff and not self._hurry and not self._stop:  # type: ignore
 88                await asyncio.sleep(0.2)
 89                waited += 0.2
 90            if self._hurry:
 91                self._hurry = False
 92                backoff = self._backoff0       # a deliberate retry resets the ramp
 93            else:
 94                backoff = min(backoff * 2, self._backoff_cap)
 95        self._on_stopped()
 96
 97    # hooks -- default no-ops
 98    def _on_error(self, e: Any) -> None:
 99        pass
100
101    def _on_retry_wait(self) -> None:
102        pass
103
104    def _on_stopped(self) -> None:
105        pass

131class SyncClient(_WsClient):
132    """Background cloud-save sync over the same lobby server.
133
134    Logs in with `sync_only` so it never joins the live roster (it can coexist with
135    a lobby login on the same account). On connect it receives the account's stored
136    save and hands it to `on_pull` once; thereafter `push_save` ships the latest
137    local save up (only the newest pending push is kept). Reconnects with backoff;
138    a bad password stops the loop. Fully fail-soft — offline just means no sync.
139    """
140
141    _backoff_cap = 60.0
142
143    def __init__(self, uri: Any, name: str, pw: str="", on_pull: Optional[Any]=None) -> None:
144        # The server ACKs every save with {"t":"saved","ok":...} and answers
145        # ok=False when it DROPPED it (a stale lease: a newer session of this
146        # account owns the saves).  We never read that ack -- _handle had no
147        # "saved" branch -- so a device whose lease was taken went on believing
148        # its pet was syncing while the server binned every push.  Cross-device
149        # progress vanished, silently.  (Swallowed-failure sweep 2026-07-13:
150        # the same shape as the iOS silent save and the dead Termux bridge.)
151        self.cloud_dropped = False
152        self.save_invalid = False             # server rejected the save FORMAT
153        self.save_too_big = False             # our own pre-send size refusal
154        self.last_error = ""                  # last server error frame (e.g. full)
155        self.uri = uri
156        self.name = name
157        self.pw = pw
158        self.on_pull = on_pull
159        self.connected = False
160        self.inbox: list = []                 # (from_name, text) PMs -> the home-screen alert
161        self._pending = None                  # latest save dict awaiting upload
162        self._ws = None  # type: ignore
163        self._wake: asyncio.Event = asyncio.Event()
164        self._stop = False                    # set on auth failure -> don't retry
165
166    def push_save(self, save: Any) -> None:
167        try:
168            if len(json.dumps(save)) > SAVE_WIRE_MAX:
169                self.save_too_big = True      # surfaced once by the app's warn pass
170                return
171        except (TypeError, ValueError):
172            return                            # an unserializable save can't ship anyway
173        self.save_too_big = False
174        self._pending = save                  # keep only the newest; server is last-write-wins
175        self._wake.set()
176
177    def _login_msg(self) -> Any:
178        from tuipet.network.cloudsync import BOOT    # one launch stamp per process
179        return {"t": "login", "name": self.name, "pw": self.pw,
180                "sync_only": True, "boot": BOOT}
181
182    def _on_connect(self) -> None:
183        self.connected = True
184        if self._pending is not None:
185            self._wake.set()                  # flush anything queued while we were down
186
187    def _on_disconnect(self) -> None:
188        self.connected = False
189
190    async def _send_loop(self) -> None:
191        while True:
192            await self._wake.wait()
193            self._wake.clear()
194            save = self._pending
195            if save is not None and self._ws is not None:
196                try:
197                    await self._ws.send(json.dumps({"t": "save", "save": save}))
198                except Exception:
199                    # a failed send must NOT kill the loop (it used to `return`,
200                    # silently ending autosave uploads for the process lifetime);
201                    # keep the save pending -- the reconnect loop re-wakes us
202                    self._wake.set()
203                    await asyncio.sleep(1.0)
204
205    def _handle(self, raw: Any) -> None:
206        m, t = parse_msg(raw)
207        if t == "welcome":
208            if self.on_pull is not None:
209                self.on_pull(m.get("save"))    # server's stored save (or None)
210        elif t == "saved":
211            # the verdict on our last push: ok=False means the server BINNED
212            # it -- and `why` says for WHICH reason (lease vs format; the old
213            # single flag blamed "a newer session" for everything)
214            ok = bool(m.get("ok"))
215            why = m.get("why")
216            self.cloud_dropped = (not ok) and why in (None, "lease")
217            self.save_invalid = (not ok) and why == "invalid"
218        elif t == "pm":
219            # the sync ghost is the HOME-SCREEN alert channel: the app drains
220            # this into the message box (presence 2026-07-05)
221            self.inbox.append((m.get("from_name", "?"), m.get("text", "")))
222            del self.inbox[:-20]
223        elif t == "announce":
224            # a server announcement reaches the home screen the same way a PM
225            # does -- the 📢 name marks it as the dev's line
226            self.inbox.append(("📢", m.get("text", "")))
227            del self.inbox[:-20]
228        elif t == "login_failed":
229            self._stop = True                  # wrong password -> stop retrying
230        elif t == "error":
231            # e.g. "Lobby is full." -- silently ignoring it meant an unending
232            # blind reconnect loop (audit 2026-07-18); the app warns once
233            code = m.get("code")
234            msg = m.get("msg") or ""
235            self.last_error = t(code, msg) if code else msg

346    async def _send_loop(self) -> None:
347        while True:
348            obj = await self._q.get()
349            try:
350                await self._ws.send(json.dumps(obj))
351            except Exception:
352                # don't lose the frame we already dequeued: requeue it so
353                # run()'s reconnect spawns a fresh sender that delivers it.  A
354                # bare `return` silently dropped that obj -- a move/invite frame
355                # could vanish and hang a handshake (hardening 2026-07-12).
356                try:
357                    self._q.put_nowait(obj)
358                except Exception:
359                    pass
360                return
361
362    def _handle(self, raw: Any) -> None:
363        m, t = parse_msg(raw)
364        s = self.state
365        if t == "welcome":
366            self._had_welcome = True
367            s.me_id = m.get("id")
368            s.me_name = m.get("name")
369        elif t == "ladder":
370            self.ladder = m               # the rankings page renders from this
371        elif t == "roster":
372            s.roster = m.get("players") or []
373        elif t == "room_ok":
374            s.room = m.get("room") or None
375            s.chat.append(("", f"— room: {s.room} —" if s.room else "— main lobby —"))
376            del s.chat[:-CHAT_CAP]
377        elif t == "chat":
378            nm = m.get("from_name", "?")
379            if nm not in s.blocked and not self._replayed(m, nm):
380                s.chat.append((nm, m.get("text", "")))
381                del s.chat[:-CHAT_CAP]
382        elif t == "announce":
383            # a server announcement rides the public feed under the 📢 speaker
384            # (unblockable -- it's the dev's line, not a peer's)
385            if not self._replayed(m, ANNOUNCE):
386                s.chat.append((ANNOUNCE, m.get("text", "")))
387                del s.chat[:-CHAT_CAP]
388        elif t == "pm":
389            nm = m.get("from_name", "?")
390            if nm not in s.blocked:
391                s.dms.setdefault(nm, []).append((nm, m.get("text", "")))
392                del s.dms[nm][:-CHAT_CAP]
393                s.unread.add(nm)
394        elif t == "pm_ok":
395            to = m.get("to_name", "?")
396            s.dms.setdefault(to, []).append((s.me_name or "you", m.get("text", "")))
397            del s.dms[to][:-CHAT_CAP]
398        elif t == "raid":
399            self.raid = m                 # the raid page renders from this
400        elif t == "raid_hit":
401            self.raid = None              # stale view: the pool moved; refetch
402            self.last_hit = m
403        elif t == "raid_reward":
404            self.raid_reward = m          # the claim flow consumes this once
405        elif t == "ladder_reward":
406            self.ladder_reward = m        # the season-award flow consumes this once
407        elif t in ("invite", "invite_resp", "relay"):
408            s.inbox.append(m)
409        elif t == "login_failed":
410            msg = m.get("msg") or ""
411            if self._had_welcome and "newer session" in msg.lower():
412                # (matches the server's real conflict line "Signed in on a
413                # newer session." -- the previous match string was one the
414                # server never sent, a drifted dead branch; audit 2026-07-18)
415                # reconnect race: the server hasn't reaped our dead session yet --
416                # NOT a credentials problem, so keep retrying instead of ejecting
417                # the player to the login screen
418                pass
419            else:
420                code = m.get("code")
421                msg = m.get("msg") or ""
422                s.login_failed = t(code, msg) if code else msg
423                self._stop = True
424        elif t == "error":
425            code = m.get("code")
426            msg = m.get("msg") or ""
427            s.error = t(code, msg) if code else msg
428
429    def _replayed(self, m: Any, nm: Any) -> Any:
430        """True for a server-marked backlog `replay` line the pane already
431        shows: reconnects and room→main returns re-send the same window, and
432        with no client dedup every line printed twice.  Live repeats are
433        untagged and always append."""
434        return bool(m.get("replay")) and (nm, m.get("text", "")) in self.state.chat[-30:]

437async def submit_bug(uri: Any, text: str, meta: Optional[Any]=None, name: str="", timeout: float=8.0) -> Any:
438    """One-shot bug submit: connect, send a bug envelope (NO login needed),
439    await the server ack, close.  True only on a server-confirmed store."""
440    payload = {"t": "bug", "text": text, "name": name}
441    if meta:
442        payload.update(meta)
443    try:
444        async with websockets.connect(uri, max_size=64 * 1024, open_timeout=timeout) as ws:
445            await ws.send(json.dumps(payload))
446            raw = await asyncio.wait_for(ws.recv(), timeout=timeout)
447            m, t = parse_msg(raw)
448            return bool(t == "bug_ok" and m and m.get("ok"))
449    except Exception:
450        return False
