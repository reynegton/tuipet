tuipet.core.evolution

Care-quality evolution engine — a faithful port of DVPet's Model/Evolution.

A candidate evolves into the form whose requirement gates it best satisfies: gather the current form's evolution targets, keep those that pass every gate (checkEvolReq), then pick the highest fulfilledReq, breaking ties by the smallest deviation, then at random. Comparison semantics match testCondition: GreaterThan -> actual > threshold, LessThan -> actual < threshold, EqualTo -> ==.

 29def _cmp(cond: Any, threshold: Any, actual: Any) -> Any:
 30    if cond == "None":
 31        return True
 32    if cond == "GreaterThan":
 33        return actual > threshold
 34    if cond == "LessThan":
 35        return actual < threshold
 36    if cond == "EqualTo":
 37        return actual == threshold
 38    return False

 41def _attr(gate: Any, actual: Any, total: Any) -> Any:
 42    """Attribute-power gate; fractional thresholds (0<v<1) compare a ratio."""
 43    cond, val = gate
 44    if cond == "None":
 45        return True
 46    if 0.0 < val < 1.0:
 47        return _cmp(cond, val, actual / total) if total > 0 else False
 48    return _cmp(cond, val, actual)

 51def _scale_rate(cond: Any, first: Any, second: Any) -> Any:
 52    """Config.scaleRate: weight a met attribute gate by the target threshold's
 53    distance from the current form's threshold (Java switch fall-through)."""
 54    if cond in ("GreaterThan", "EqualTo") and first > second:
 55        return (first - second) / first if first else 1.0
 56    if second > first:
 57        return (second - first) / second if second else 1.0
 58    return 1.0

 61def weight_category(weight: Any, base: Any) -> Any:

 90def check(pet: Any, num: int, item: Any=-1, food: Any=-1, connecting: bool=False) -> Any:
 91    """checkEvolReq, verbatim semantics (canon re-audit 2026-07):
 92
 93    * NO DNA gate-forgiveness (the canon consume-once excuse left with the
 94      DNA slim, BASIC VPET 2026-07-16): a failed care gate is a failed gate.
 95      The charge still BENDS selection -- the fulfilled-score bonus and the
 96      divergence arm -- but it never excuses a gate here (the body's own
 97      note; this docstring taught the removed rule until the LOW audit).
 98    * special types follow checkSpecialCondition + the optional flags:
 99      JogressOptional=TRUE (classic) -- a Jogress form IS reachable by normal
100      timed care (its heavy gates still apply); FusionOptional=FALSE -- Fusion
101      needs the handshake; Failed forms compete like anyone; Mode/Death/Xros
102      have their own triggers.  `connecting` waives the whole type gate (the
103      jogress/mode/death helpers all pass it, as before).
104    * only an INDUCED X requirement demands the antibody (canon gates Natural
105      forms by care alone -- 180 corpus forms were wrongly locked); the
106      probability roll applies to X forms like everyone (the old skip was not
107      canon).  The probability roll is never DNA-bypassed.  (checkStatTotal
108      and the per-attr stat gates left 2026-07-17 -- DVPet-only power walls
109      the humulos guides never speak.)"""
110    req = data.load_requirements().get(num)
111    if req is None:
112        return False
113    special = req.get("special", "None")
114    if not connecting:
115        if special in ("Mode", "Death", "Xros"):
116            return False          # checkSpecialCondition: their own triggers only
117        if special == "Fusion":
118            return False          # FusionOptional=FALSE: the handshake only
119        # "Jogress" (JogressOptional=TRUE), "Failed" and "None" compete normally
120    ev_item = req.get("evol_item", -1)
121    if item == -1:
122        if ev_item != -1:
123            return False     # item-locked form: unreachable by timed care (need the item)
124    elif ev_item != item:
125        return False         # using an item only validates the forms that require it
126    # the FOOD lock is UNLOCKED (Joel 2026-07-16, with the item-system
127    # clone): the food catalog is gone, so Citramon -- the corpus' one
128    # food-locked form (EvolFood 42, the Orange) -- competes by timed care
129    # like everyone; its remaining requirement gates still apply.
130    if req.get("xantibody", "None") == "Induced" and getattr(pet, "x_antibody", "None") == "None":
131        return False  # Induced X-forms are unreachable without the antibody (canon: Induced ONLY)
132    vac, dat, vir = _stats(pet)
133    total = vac + dat + vir
134    # (checkStatTotal DROPPED with the six per-attr stat gates, 2026-07-17:
135    # the same DVPet-only power wall under the same +1-per-win economy)
136    gates = [
137        _cmp(*req["battles"], pet.battles),  # type: ignore
138        # (the six vaccine/data/virus stat gates DROPPED 2026-07-17: they
139        # were DVPet-app truth -- the humulos guides gate on mistakes/
140        # trainings/battles/wins, never power numbers -- and the 0.5 economy
141        # (+1 power per win) put their monster.csv thresholds out of reach.
142        # The fulfilled-score/deviation legs still read the live stats.)
143        # (the "trains at Morning/Noon/Night" TIME gate DROPPED with the
144        # day/night system -- BASIC VPET 2026-07-17: an hour nothing can
145        # reach would wall 374 corpus forms, the temp_req/habitat_req call)
146        req["weight"] == "None" or req["weight"] == weight_category(pet.weight, pet._base_weight()),
147        _cmp(*req["disturb"], pet.disturb),  # type: ignore
148        _cmp(*req["overeat"], pet.overeat),  # type: ignore
149        # (the sick_count gates DROPPED with the sickness system (BASIC VPET 2026-07-17) --
150        # GreaterThan rows would wall forever on a count nothing can move;
151        # the injured gates already read a pinned 0 -- and 41 GreaterThan
152        # rows, Omnimon's Fusion among them, were WALLS on a counter nothing
153        # can move.  DROPPED like every other dead-meter family; Fusion/Mode
154        # canon audit 2026-07-18)
155        # checkMoodReq: getCurrentMood -- the STICKY tier (Depressed holds until
156        # checkDepressed's exit roll; a threshold recompute is never Depressed).
157        # The old mood_category invented a Depressed tier at <= -250, failing
158        # the 70 Mood=Unhappy requirement rows for any deeply-sad pet.
159        True,   # (mood gates left with the mood system; BASIC VPET 2026-07-16)
160        True,   # (obedience gates left with the discipline system)
161        _cmp(*req["wins"], _win_rate(pet)),  # type: ignore
162        _cmp(*req["mistakes"], pet.care_mistakes),  # type: ignore
163        True,   # (major_food gates left with the nutrition/taste system)
164        _cmp(*req.get("incarnations", ("None", 0)), getattr(pet, "generation", 1)),  # type: ignore
165    ]
166    # LevelFought: enough opponents of at least MinLevelFought power beaten this stage
167    lf_min = req.get("level_fought_min", 0)
168    if lf_min:
169        cnt = sum(1 for lv in getattr(pet, "levels_fought", ()) if lv >= lf_min)
170        gates.append(_cmp(*req["level_fought"], cnt))  # type: ignore
171    # (temp_req gates left with the weather system -- BASIC VPET 2026-07-16:
172    # a temperature band nothing can move would permanently wall those forms)
173    # (habitat_req gates DROPPED with the habitat system -- BASIC VPET
174    # 2026-07-16: a home nothing can move to would permanently wall those
175    # forms, the same call the temp_req gates got)
176    failed = sum(1 for g in gates if not g)
177    # (the DNA gate-forgiveness -- canon getDNA()'s consume-once excuse --
178    # left with the DNA slim, BASIC VPET 2026-07-16: the charge still BENDS
179    # selection through the fulfilled-score bonus and arms divergence, but a
180    # failed care gate is a failed gate)
181    if failed > 0:
182        return False
183    # probability: prob >= probBound -> always; else prob must beat a roll.
184    # DVPet boosts prob by the care bonus + winRate*coefficient (checkEvolReq).
185    boost = getattr(pet, "evol_bonus", 0) + int(_win_rate(pet) * WIN_RATE_PROB_COEF)
186    prob, bound = req["prob"] + boost, req["probBound"]
187    if prob < bound:
188        return prob > random.randint(0, bound - 1)
189    return True

197def fulfilled(pet: Any, num: int) -> Any:
198    """getFulfilledReq: 1 + priority + summed rates of every met gate."""
199    req = data.load_requirements()[num]
200    vac, dat, vir = _stats(pet)
201    total = vac + dat + vir
202    score = 1.0 + req["priority"]
203    cur = data.load_requirements().get(pet.num, {})   # current form, for scaleRate
204    if _met(req["battles"], pet.battles):
205        score += R["battle"]
206    for k, actual in (("data", dat), ("vaccine", vac), ("virus", vir)):
207        cg = cur.get(k) or [("None", 0), ("None", 0)]
208        for i in (0, 1):
209            g = req[k][i]
210            if g[0] != "None" and _attr(g, actual, total):
211                cur_thr = cg[i][1] if i < len(cg) else 0
212                score += R[k if k in R else "data"] * _scale_rate(g[0], g[1], cur_thr)
213    if req["weight"] != "None" and req["weight"] == weight_category(pet.weight, pet._base_weight()):
214        score += R["weight"]
215    for k, actual in (("disturb", pet.disturb), ("overeat", pet.overeat)):
216        if _met(req[k], actual):
217            score += R.get(k, 1)
218    if False:   # (the mood score left with the mood system; BASIC VPET)
219        score += R["mood"]
220    if False:   # (the major_food score left with the taste system)
221        score += 1
222    if _met(req["wins"], _win_rate(pet)):
223        score += R["winRate"]
224    cond, val = req["mistakes"]
225    if cond != "None" and _cmp(cond, val, pet.care_mistakes):
226        score += {"LessThan": R["mistakeLess"], "GreaterThan": R["mistakeGreater"],
227                  "EqualTo": R["mistakeEqual"]}.get(cond, R["mistakeNone"])
228    lf_min = req.get("level_fought_min", 0)
229    if lf_min and req["level_fought"][0] != "None":
230        cnt = sum(1 for lv in getattr(pet, "levels_fought", ()) if lv >= lf_min)
231        if _cmp(*req["level_fought"], cnt):  # type: ignore
232            score += 1  # Config._levelFoughtRate
233    # canon scores ONLY an Induced X requirement (evolution audit 2026-07-06:
234    # the old "Natural" arm over-scored 180 corpus forms)
235    if req.get("xantibody", "None") == "Induced" and getattr(pet, "x_antibody", "None") != "None":
236        score += X_ANTIBODY_RATE
237    for f, g in (req.get("dna") or {}).items():     # getDNAReq: priority per met Field gate
238        if g[0] != "None" and _cmp(g[0], g[1], pet.dna_percent(f)):
239            score += DNA_FULFILLED_RATE
240    if _dna_ok(pet, req):                            # getDNAReq: full-match dnaFulfilledRate bonus
241        score += DNA_FULFILLED_RATE
242    # (the habitat_req score leg and the major-habitat element/field
243    # affinity shade left with the habitat system -- BASIC VPET 2026-07-16.
244    # A dangling element fragment survived that cut INSIDE the _dna_ok
245    # branch above -- a latent NameError on `h` for any DNA-charged pet --
246    # and left with the element system, 2026-07-18.)
247    return score

250def deviation(pet: Any, num: int) -> Any:
251    """tieBreaker: total absolute distance from the met gates' thresholds."""
252    req = data.load_requirements()[num]
253    vac, dat, vir = _stats(pet)
254    total = vac + dat + vir
255    dev = 0.0
256    if _met(req["battles"], pet.battles):
257        dev += abs(req["battles"][1] - pet.battles)
258    for k, actual in (("data", dat), ("vaccine", vac), ("virus", vir)):
259        for i in (0, 1):
260            g = req[k][i]
261            if g[0] != "None" and _attr(g, actual, total):
262                dev += abs(g[1] - actual)
263    for k, actual in (("disturb", pet.disturb), ("overeat", pet.overeat),
264                      ("mistakes", pet.care_mistakes)):
265        if _met(req[k], actual):
266            dev += abs(req[k][1] - actual)
267    if _met(req["wins"], _win_rate(pet)):
268        dev += abs(req["wins"][1] - _win_rate(pet))
269    return dev

272def _failed_form(pet: Any, by_num: Any) -> Any:
273    """DVPet's safety net: a pet that meets no requirement still evolves -- into

277              and not data.is_placeholder(t)
278              and data.load_requirements().get(t, {}).get("special") == "Failed"]
279    return random.choice(failed) if failed else None

282def item_select(pet: Any, item_id: Any) -> Any:
283    """Forms reachable by USING item `item_id`: graph targets whose EvolItemID == item_id
284    and whose care gates pass (the item is an extra gate, not a bypass). Best by fulfilled."""
285    _, by_num = data.load_sprites()
286    targets = [t for t in data.load_evolutions().get(pet.num, [])
287               if t in by_num and not data.is_placeholder(t)]
288    valid = [t for t in targets if check(pet, t, item=item_id)]
289    if not valid:
290        return None
291    best = max(fulfilled(pet, t) for t in valid)
292    top = [t for t in valid if abs(fulfilled(pet, t) - best) < 1e-9]
293    if len(top) > 1:
294        mind = min(deviation(pet, t) for t in top)
295        top = [t for t in top if deviation(pet, t) == mind]
296    return random.choice(top)

327def select(pet: Any) -> Any:
328    """Return the chosen evolution target num, or None.
329
330    Normal evolution climbs a stage.  (The X-Antibody steering and same-stage
331    X reformat are retired -- LINES_SPEC §4: X-forms belong to X egg lines;
332    an antibody-carrying corpus pet still passes check()'s Induced gate, but
333    the antibody no longer hijacks the choice.)"""
334    _, by_num = data.load_sprites()
335    targets = []
336    for t in data.load_evolutions().get(pet.num, []):
337        if t not in by_num or t == pet.num:
338            continue
339        if data.is_placeholder(t):
340            continue
341        if by_num[t]["stage"] != pet.stage:
342            targets.append(t)
343    valid = [t for t in targets if check(pet, t)]
344    if not valid:
345        # nothing fully qualifies -> Failed form if the species has one, else the
346        # best-matched stage-up target, so a pet NEVER gets stuck below Mega
347        ff = _failed_form(pet, by_num)
348        if ff is not None:
349            return ff
350        stage_up = [t for t in targets if by_num[t]["stage"] != pet.stage
351                    and data.load_requirements().get(t, {}).get("special", "None") == "None"
352                    and data.load_requirements().get(t, {}).get("evol_item", -1) == -1
353                    # the Induced gate check()/divergence enforce: without it
354                    # the fallback pushed an antibody-less pet into an X form
355                    # and evolve_to locked x_antibody="Permanent" -- a free X
356                    # with no X egg or chip (gameplay audit 2026-07-19)
357                    and (data.load_requirements().get(t, {}).get("xantibody", "None") != "Induced"
358                         or getattr(pet, "x_antibody", "None") != "None")]
359        if stage_up:
360            return max(stage_up, key=lambda t: fulfilled(pet, t))
361        return None
362    best = max(fulfilled(pet, t) for t in valid)
363    top = [t for t in valid if abs(fulfilled(pet, t) - best) < 1e-9]
364    if len(top) > 1:
365        mind = min(deviation(pet, t) for t in top)
366        top = [t for t in top if deviation(pet, t) == mind]
367    return random.choice(top)

370# ---- DNA divergence: the wild road ("ultimate v-pet" arc, 2026-07-07) -------
371# The lines are the raising spine (529 forms); the corpus graph reaches 1,454.
372# Divergence is the door between them: a pet whose CHARGED DNA has a
373# strict-max Field at/over its stage's threshold steers evolution off the
374# chart -- the graph's NEXT-STAGE children of the current form in that Field
375# become the candidates, judged by this engine's own gates (check ->
376# fulfilled -> deviation).  The charge is consumed by evolve_to's reset_dna
377# like any evolution; the caller re-anchors via lines.adopt_line (a chart
378# that claims the target keeps the pet on lines; otherwise it rides this
379# corpus engine from then on -- a road it PAID for).
380#
381# Thresholds are the balance knob: the charge bill is ENERGY, 1/unit on
382# your own Field and 2/unit off-Field over the -24..24 meter (the old
383# spirit bill left with its system), so each figure below is still N charge
384# SESSIONS with recovery between, not one action -- plus the wager bits
385# that banked the DNA.
386DIVERGE_NEED = {"Fresh": 2, "InTraining": 4, "Rookie": 8,
387                "Champion": 12, "Ultimate": 16}
388_STAGE_ORDER = ["Fresh", "InTraining", "Rookie", "Champion", "Ultimate", "Mega"]

391def divergence_target(pet: Any) -> Any:
392    """The armed steer's destination, or None while unarmed (no strict-max
393    Field, an under-threshold charge, or no graph edge in that Field)."""
394    field = pet.highest_dna()
395    if not field or field == "None":       # the None bin is wasted DNA, not a road
396        return None
397    need = DIVERGE_NEED.get(pet.stage)
398    if need is None or pet.dna_applied.get(field, 0) < need:
399        return None
400    try:
401        nxt = _STAGE_ORDER[_STAGE_ORDER.index(pet.stage) + 1]
402    except (ValueError, IndexError):
403        return None
404    _, by_num = data.load_sprites()
405    reqs = data.load_requirements()
406    out = []
407    for t in data.load_evolutions().get(pet.num, []):
408        rec = by_num.get(t)
409        if rec is None or t == pet.num or data.is_placeholder(t):
410            continue
411        if rec["stage"] != nxt or rec["field"] != field:
412            continue
413        req = reqs.get(t, {})
414        if req.get("special", "None") != "None" or req.get("evol_item", -1) != -1:
415            continue        # jogress/fusion/mode/item roads keep their own doors
416        if req.get("xantibody", "None") == "Induced" \
417                and getattr(pet, "x_antibody", "None") == "None":
418            continue        # Induced X-forms stay X-egg territory (LINES_SPEC §4)
419        out.append(t)
420    if not out:
421        return None
422    # the paid steer always fires: gates pick among qualifiers, else the
423    # best-fulfilled edge (select()'s never-stuck fallback idiom)
424    valid = [t for t in out if check(pet, t)]
425    pool = valid or out
426    best = max(fulfilled(pet, t) for t in pool)
427    top = [t for t in pool if abs(fulfilled(pet, t) - best) < 1e-9]
428    if len(top) > 1:
429        mind = min(deviation(pet, t) for t in top)
430        top = [t for t in top if deviation(pet, t) == mind]
431    return random.choice(top)

396    if not field or field == "None":       # the None bin is wasted DNA, not a road
397        return None
398    need = DIVERGE_NEED.get(pet.stage)
399    if need is None or pet.dna_applied.get(field, 0) < need:
400        return None
401    try:
402        nxt = _STAGE_ORDER[_STAGE_ORDER.index(pet.stage) + 1]
403    except (ValueError, IndexError):
404        return None
405    _, by_num = data.load_sprites()
406    reqs = data.load_requirements()
407    out = []
408    for t in data.load_evolutions().get(pet.num, []):
409        rec = by_num.get(t)
410        if rec is None or t == pet.num or data.is_placeholder(t):
411            continue
412        if rec["stage"] != nxt or rec["field"] != field:
413            continue
414        req = reqs.get(t, {})
415        if req.get("special", "None") != "None" or req.get("evol_item", -1) != -1:
416            continue        # jogress/fusion/mode/item roads keep their own doors
417        if req.get("xantibody", "None") == "Induced" \
418                and getattr(pet, "x_antibody", "None") == "None":
419            continue        # Induced X-forms stay X-egg territory (LINES_SPEC §4)
420        out.append(t)
421    if not out:
422        return None
423    # the paid steer always fires: gates pick among qualifiers, else the
424    # best-fulfilled edge (select()'s never-stuck fallback idiom)
425    valid = [t for t in out if check(pet, t)]
426    pool = valid or out
427    best = max(fulfilled(pet, t) for t in pool)
428    top = [t for t in pool if abs(fulfilled(pet, t) - best) < 1e-9]
429    if len(top) > 1:
430        mind = min(deviation(pet, t) for t in top)
431        top = [t for t in top if deviation(pet, t) == mind]
432    return random.choice(top)

506    (met, text) rows, one per CONSTRAINED gate, mirroring check()'s reads exactly
507    (unconstrained "None" gates are skipped).  met is True/False, or None for an
508    informational row (the probability line, the DNA-bypass note)."""
509    req = data.load_requirements().get(num)
510    if req is None:
511        return [(False, "unknown form")]
512    rows = []
513
514    def cmp_row(label: Any, gate: Any, actual: Any, pct: bool=False) -> None:
515        cond, val = gate
516        if cond == "None":
517            return
518        unit = "%" if pct else ""
519        rows.append((_cmp(cond, val, actual),
520                     f"{label} {_SYM.get(cond, '?')}{val:g}{unit}  (now {actual:g}{unit})"))
521
522    # hard locks first: forms normal timed care can never reach
523    special = req.get("special", "None")
524    if special == "Fusion":

526    elif special == "Mode":

528    elif special == "Death":

530    elif special == "Jogress":
531        rows.append((None, "a Jogress line \u2014 care can reach it"))
532    ev_item = req.get("evol_item", -1)
533    if ev_item != -1:
534        item = data.consumable_by_key(f"i:{ev_item}")

538        from tuipet.core.pet import Pet as _Pet
