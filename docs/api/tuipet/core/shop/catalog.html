 205def tier_for_price(price: Any) -> Any:
 206    """The band a price falls in, or None for a grant-only item."""
 207    if price is None:
 208        return None
 209    for ceiling, name in TIER_BANDS:
 210        if price <= ceiling:
 211            return name
 212    return TIER_TOP

 215def tier_weight(key: str) -> Any:
 216    """Roll weight for `key` -- the find pools and any weighted shelf pick."""
 217    if key in _WEIGHT_OVERRIDE:
 218        return _WEIGHT_OVERRIDE[key]
 219    v = CATALOG.get(key)
 220    return TIER_WEIGHT.get((v.tier if v else None) or "common", 1)

 223def tier_stock(key: str) -> Any:
 224    v = CATALOG.get(key)

 228def adventure_open(key: str, prog: Optional[Any]=None) -> Any:
 229    """Is this road-shelf item unlocked (enough maps cleared)?  Non-gated keys
 230    are always open."""
 231    need = ADVENTURE_GATES.get(key)
 232    if need is None:
 233        return True
 234    if prog is None:
 235        import tuipet.utils.persistence as persistence
 236        prog = persistence.get_progress()
 237    return len(prog.get("maps", ()) or ()) >= need

 240def key_for_icon(icon: Any) -> Any:

 242    A RETIRED key's icon resolves to its heir -- authored loot rows, cup

 245    if k is not None:

 401def wave_status(prog: Optional[Any]=None) -> Any:
 402    """(sealed_count, closest-wave tease) from live RELIC_GATES
 403    progress -- (0, '') once every relic is on the shelf."""
 404    if prog is None:
 405        import tuipet.utils.persistence as persistence
 406        prog = persistence.get_progress()
 407    sealed = [g for k, g in RELIC_GATES.items()
 408              if g is not None and not relic_open(k, prog)]
 409    if not sealed:
 410        return 0, ""
 411
 412    def ratio(g: Any) -> Any:
 413        sig, need = g
 414        return min(1.0, int(prog.get(sig, 0)) / need)
 415    # DETERMINISTIC TIE-BREAK (shops audit 2026-07-25).  `max(set(...))`
 416    # walked a SET, so when two sealed waves tied on ratio the winner fell
 417    # out of string-hash iteration order -- which Python randomises PER
 418    # PROCESS.  Measured: the same save printed "wins 0/25 wake Light &
 419    # Kindness" or "generation 0/5 wakes Destiny" depending on the launch,
 420    # and 7% of sampled progress states tie (any player at gen 5+ with no
 421    # armor evo and no wins sits in one).  A shop's character is supposed
 422    # to be STABLE -- the same law that keeps a town's guest good crc32-
 423    # ordered and its deal fixed for the day.  Ties now break toward the
 424    # NEAREST goal in absolute terms (smallest `need`), which is also the
 425    # more useful tease: "your 1st armor evo" over "wins 0/25".
 426    sig, need = max(sorted(set(sealed)), key=lambda g: (ratio(g), -g[1]))
 427    have = min(int(prog.get(sig, 0)), need)
 428    tease = _WAVE_TEASE.get((sig, need), "mais relíquias surgem por aí")
 429    return len(sealed), tease.format(have=have, need=need)

 932def town_egg_stock(town_id: Any, count: Any=EGG_STOCK_PER_TOWN) -> Any:
 933    """The DISTINCT set of eggs THIS town sells -- a stable band over the
 934    earnable egg, rotated by town so no two town shops feel the same."""
 935    pool = _sellable_eggs()
 936    if not pool:
 937        return []
 938    count = min(count, len(pool))
 939    # a stride CO-PRIME with the pool keeps every town's band distinct --
 940    # stepping by the band width alone collapsed to len(pool)/count bands
 941    # the moment the width divided the pool (egg audit 2026-07-25: cutting
 942    # the 5 lineage eggs left a 36-egg pool, and 26 towns fell into 6 bands)
 943    stride = count
 944    while len(pool) > 1 and _gcd(stride, len(pool)) != 1:  # type: ignore
 945        stride += 1
 946    start = (int(town_id) * stride) % len(pool)
 947    return [pool[(start + i) % len(pool)] for i in range(count)]

1347def _deal_index(seed: Any, count: int, today: Optional[Any]=None) -> Any:
1348    """A daily rotating index in [0, count), crc32-seeded on (seed, day):
1349    stable all day, different tomorrow.  DEDUPED (2026-07-24, Joel: "dedup
1350    the town deal") -- it never repeats YESTERDAY's pick, so no shelf shows
1351    the same deal two days running.  (With a single item there is nothing
1352    to rotate to; it stays put.)
1353
1354    The dedup compares against yesterday's FINAL (post-bump) pick, not its
1355    raw index -- if it compared raws, a day that was itself bumped could be
1356    silently repeated by today.  So the picks are walked FORWARD from a
1357    fixed lookback: each day bumps off the previous day's final.  Any error
1358    in the window's first day washes out long before today."""
1359    if count <= 0:
1360        return None
1361    if count == 1:
1362        return 0
1363    import zlib
1364    def raw(day: Any) -> Any:
1365        return zlib.crc32(f"{seed}:{day}".encode()) % count
1366    day = _today_ordinal(today)
1367    final = raw(day - _DEAL_LOOKBACK)
1368    for d in range(day - _DEAL_LOOKBACK + 1, day + 1):
1369        r = raw(d)
1370        if r == final:                       # would repeat yesterday's pick
1371            r = (r + 1) % count
1372        final = r
1373    return final
