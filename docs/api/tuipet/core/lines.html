tuipet.core.lines

Evolution lines — the LINES_SPEC.md engine (arc 1).

An egg is a line: a curated, DM20-shaped subtree of the corpus. A pet hatched from a line egg evolves ONLY inside its line, judged by first-match bracket rules instead of the corpus fulfilled-score engine. Rules read the per-stage care counters (care_mistakes / stage_trainings / overeat / stage_battles) and the rolling battle_log, so every evolution is a legible verdict on this stage's care.

Rule grammar (LINES_SPEC §2) — comma = AND, | = OR, first matching row wins: CM 0-2 | CM 3+ care mistakes bracket TR 5-15 trainings OF 3+ overfeeds BTL 15+ battles this stage WIN 12/15 >=12 wins in last 15 LV 5-6 DMX level (battle exp, humulos table; fix 2026-07-17) KO6 5+ Mega-class foes beaten AREA n+1 raid bosses felled (was "map n cleared"; the raid re-gate, BASIC VPET 2026-07-16) TIME no requirement — the stage timer alone

 34def _bracket(arg: Any) -> Any:
 35    """'0-2' -> (0, 2); '3+' -> (3, None); '7' -> (7, 7)."""
 36    arg = arg.strip()
 37    if arg.endswith("+"):
 38        return int(arg[:-1]), None
 39    if "-" in arg:
 40        lo, hi = arg.split("-", 1)
 41        return int(lo), int(hi)
 42    v = int(arg)
 43    return v, v

 46def parse_rule(text: str) -> Any:
 47    """Rule text -> list of OR-alternatives, each a list of (kind, a, b) atoms.
 48    'TIME' (or blank) parses to one unconstrained alternative."""
 49    text = (text or "").strip()
 50    alts = []
 51    for alt in text.split("|"):
 52        atoms = []
 53        for part in alt.split(","):
 54            part = part.strip()
 55            if not part or part.upper() == "TIME":
 56                continue
 57            kind, _, arg = part.partition(" ")
 58            kind, arg = kind.upper(), arg.strip()
 59            if kind == "WIN":
 60                k, n = arg.split("/", 1)
 61                atoms.append(("win", int(k), int(n)))
 62            elif kind == "JOGRESS":
 63                # a jogress DOOR (LINES_SPEC §6): arg = the exact partner dex,
 64                # or a Pendulum attribute spec ("Data/Virus") -- any same-stage
 65                # partner with a listed attribute resonates (pen20 manual).
 66                # Never fires on the stage timer -- the lobby fusion opens it.
 67                if arg.isdigit():
 68                    atoms.append(("jogress", int(arg), None))  # type: ignore
 69                else:
 70                    attrs = tuple(a.strip() for a in arg.split("/") if a.strip())
 71                    if not attrs or any(a not in ("Vaccine", "Data", "Virus",
 72                                                  "None") for a in attrs):
 73                        raise ValueError(f"bad JOGRESS partner: {part!r}")
 74                    atoms.append(("jogress", attrs, None))  # type: ignore
 75            elif kind == "AREA":
 76                if not arg:
 77                    raise ValueError(f"AREA atom needs a map: {part!r}")
 78                atoms.append(("area", arg, None))  # type: ignore
 79            elif kind in ("CM", "TR", "OF", "BTL", "LV", "KO6"):
 80                lo, hi = _bracket(arg)
 81                atoms.append((kind.lower(), lo, hi))
 82            else:
 83                raise ValueError(f"unknown rule atom: {part!r}")
 84        alts.append(atoms)
 85    return alts

 88@lru_cache(maxsize=1)
 89def load_lines() -> Any:
 90    """lines.csv -> {line_id: {"root": dex, "members": {dex: row},
 91    "children": {parent_dex: [row, ...] in CSV (first-match) order}}}.
 92    A row is {"num", "stage", "parents", "rule" (parsed), "rule_text", "bedtime"}."""
 93    out = {}  # type: ignore
 94    with open(os.path.join(_DATA, "lines.csv"), newline="") as fh:
 95        for raw in csv.DictReader(fh):
 96            lid = raw["LineID"].strip()
 97            line = out.setdefault(lid, {"root": None, "members": {}, "children": {}})
 98            num = int(raw["DexNum"])
 99            parents = [p.strip() for p in raw["Parents"].split(";") if p.strip()]
100            rule = parse_rule(raw["Rule"])
101            row = {"num": num, "stage": raw["Stage"].strip(),
102                   "rule": rule, "rule_text": raw["Rule"].strip(),
103                   "bedtime": raw["Bedtime"].strip(),
104                   "parents": [int(p) for p in parents if p != "egg"],
105                   # jogress door: the exact partner dex, or None for a
106                   # normal timed-care row (LINES_SPEC §6)
107                   "jogress": next((a for alt in rule for k, a, _ in alt
108                                    if k == "jogress"), None)}
109            mem = line["members"].get(num)
110            if mem is None:
111                line["members"][num] = row
112            else:
113                # a dual-road target (ver1 Devimon: Agumon needs TR 0-15,
114                # Betamon TR 16+ -- DM20 canon) keeps ONE row per parent so
115                # each road judges by its own rule; members unions the
116                # parentage so the curation invariants see every edge
117                mem["parents"] = sorted(set(mem["parents"]) | set(row["parents"]))
118            if "egg" in parents:
119                line["root"] = num
120            for p in row["parents"]:
121                line["children"].setdefault(p, []).append(row)
122    return out

133def canonical_root(dex: Any) -> Any:
134    """(root_dex, line_id) a hatching `dex` should become.  A dex that IS a
135    root maps to itself; a duplicate twin (the corpus keeps 2-3 dexes per Baby
136    name; the mystery-egg pools hatch the sub-1410 twins) maps to the root
137    sharing its NAME.  16 Baby names root SEVERAL lines (four Petitmon roots:
138    the classic petitmon line plus the slayerdra/breakdra/draco device charts),
139    so a name-hatched twin always becomes the CLASSIC line -- the lowest root
140    dex (141x < 16xx by construction) -- never csv-order luck; the device
141    lines are entered only through their own eggs.  (None, '') when no line
142    claims the name -- with every egg curated (arc 5) that means legacy data
143    only."""
144    lid = line_for_hatch(dex)
145    if lid:
146        return dex, lid
147    _, by_num = data.load_sprites()
148    rec = by_num.get(dex)
149    if rec:
150        owners = [(line["root"], lid) for lid, line in load_lines().items()
151                  if (by_num.get(line["root"]) or {}).get("name") == rec["name"]]
152        if owners:
153            return min(owners)
154    return None, ""

172def bedtime_minutes(pet: Any) -> Any:
173    """The bedtime as a minute-of-day (0..1439), or None for non-line forms.
174    '24:00' wraps to 0 (a midnight sleeper like Devimon)."""
175    bt = bedtime(pet)
176    if not bt:

179    return (int(h) % 24) * 60 + int(m)

215def _atom_met(pet: Any, atom: Any) -> Any:
216    kind, a, b = atom
217    if kind == "jogress":
218        return False       # a door, not a timer rule: only the lobby fusion opens it
219    if kind == "win":
220        return sum(pet.battle_log[-b:]) >= a
221    if kind == "area":
222        # AREA n = adventure area n CLEARED -- the authored meaning, RESTORED
223        # 2026-07-21 (run care/evo arc) now that adventure is back.  Area
224        # indices are 0-based like the eggUnlock MapComplete rows (area n ==
225        # zones map n+1), and the raid re-gate (BASIC VPET 2026-07-16: n+1
226        # felled community bosses) stays as the FALLBACK -- the same dual
227        # gate the eggs took in the rebuild; nobody who earned the gate
228        # through raids loses the road they walked.
229        n = int(a) if str(a).lstrip("-").isdigit() else 0
230        # DEVICE-LIFETIME first (audit 2026-07-25): the eggs and the shop
231        # read the profile `maps` set, and this atom's own raid arm reads
232        # the lifetime raid tally -- but the map arm read the per-life
233        # adv_progress, so Alphamon's road had to be re-walked every
234        # generation while its raid door stayed open.  The per-life check
235        # stays as the OR (pre-profile saves earned their road too).
236        import tuipet.core.adventure as adventure
237        import tuipet.utils.persistence as persistence
238        try:
239            prog = persistence.get_progress()
240            cleared = (n in (prog.get("maps", ()) or ())
241                       or adventure.is_map_cleared(pet, n + 1))
242        except Exception:
243            cleared = False
244        return cleared or _felled_raids() >= n + 1
245    v = _actual(pet, kind)
246    return v >= a and (b is None or v <= b)

305def adopt_line(pet: Any, prev: Optional[Any]=None) -> Any:
306    """Re-anchor the pet to a line whose chart contains its CURRENT form -- a
307    jogress/mode fusion keeps the pet in the line system whenever ANY line
308    claims the target (its own line preferred).  Shared nodes sit in several
309    charts (Babydmon 955 is in slayerdra, breakdra AND draco), so among
310    foreign claimants a line containing the road actually travelled -- `prev`,
311    the dex worn BEFORE the special evolution, AND the current form -- is the
312    pet's true chart; then HATCHABLE lines outrank the dormant legacy charts
313    (hatchable_roots: never capture a new pet into a line no egg hatches);
314    remaining ties break on the lowest root so the binding is data-stable,
315    never csv-order luck.  '' = truly off-chart: the legacy corpus engine
316    takes over, as before."""
317    cur = load_lines().get(getattr(pet, "line_id", ""))
318    if cur and pet.num in cur["members"]:
319        return pet.line_id
320    owners = [(lid, line) for lid, line in load_lines().items()
321              if pet.num in line["members"]]
322    if prev is not None:
323        travelled = [o for o in owners if prev in o[1]["members"]]
324        if travelled:
325            owners = travelled
326    live = [o for o in owners if o[1]["root"] in hatchable_roots()]
327    if live:
328        owners = live
329    if owners:
330        lid = min(owners, key=lambda o: o[1]["root"])[0]
331        pet.line_id = lid
332        return lid
333    pet.line_id = ""
334    return ""

363def straight_needed(pet: Any, a: Any, b: Any) -> Any:
364    """How many CONSECUTIVE wins from here reach `a` of the last `b`.
365
366    THE WINDOW ROLLS, and that is the whole point of this number (Joel
367    2026-07-23: "i won 2 fucking battles and it didnt count as wins, i
368    been on 11 for the last 2 battles").  A win that displaces an older
369    WIN is a no-op -- the counter sits perfectly still while you keep
370    winning -- so `now/b` alone cannot tell a player how far off they
371    are.  His real log was 11/15 with its three OLDEST entries wins:
372    victories 1-3 each swapped a 1 for a 1, and only the 4th pushed a
373    LOSS off the end and scored.  Returns 0 when the gate is already
374    met (b consecutive wins always fills the window, so this
375    terminates)."""
376    w = list(getattr(pet, "battle_log", None) or [])[-b:]
377    for n in range(b + 1):
378        if sum(w[-b:]) >= a:
379            return n
380        w = (w + [1])[-b:]
381    return b

384def _atom_row(pet: Any, atom: Any) -> Any:
385    kind, a, b = atom
386    if kind == "win":
387        now = sum(pet.battle_log[-b:])
388        # "training isnt filling the evo requirements in my metal greymon,
389        # unless thats supposed to be battles? is it training or battles,
390        # because i dont think its filling either way" (Joel, 2026-07-21).
391        # VERDICT MATCHES -- the counters are canon: checkEvolReq has no
392        # trainings gate (drills feed TR gates + effort), and the L17
393        # ruling keeps lobby duels progression-neutral.  The ROW was the
394        # gap: it never said WHICH battles fill the window, right after a
395        # stage whose gate WAS trainings (Agumon->Greymon TR 16+).  The
396        # answer lives in a WIN_FEED_NOTE sub-row (requirement_report):
397        # "local bouts" inlined here overflowed the LCD's 35-char row clip
398        # and amputated the (now ...) counter (Joel's screenshot, same day).
399        return _atom_met(pet, atom), f"wins {a} of last {b}  (now {now}/{min(len(pet.battle_log), b)})"
400    if kind == "jogress":
401        if isinstance(a, tuple):                         # Pendulum attribute door
402            spec = "/".join("Free" if x == "None" else x for x in a)
403            return None, f"jogress with a {spec} partner"
404        _, by_num = data.load_sprites()
405        pname = (by_num.get(a) or {}).get("name", f"#{a}")
406        return None, f"jogress with {pname} (lobby)"     # informational: a door, not a counter
407    if kind == "area":
408        # BOTH doors speak (audit 2026-07-25: "the departed adventure maps"
409        # came back 2026-07-21 -- the row still described only the raid arm,
410        # so a met gate read "(now 0)"); same phrasing as the egg guide
411        n = int(a) if str(a).lstrip("-").isdigit() else 0
412        return _atom_met(pet, atom), (f"clear map {n + 1} (or fell {n + 1} "
413                                      f"raid boss{'es' if n else ''})")
414    span = f"{a}+" if b is None else (f"{a}" if a == b else f"{a}-{b}")
415    return _atom_met(pet, atom), f"{_TXT[kind]} {span}  (now {_actual(pet, kind)})"

418def requirement_report(pet: Any, num: int) -> Any:
419    """The bracket checklist for one line target, same row shape as
420    evolution.requirement_report: (met, text). With OR alternatives, shows the
421    closest one (fewest unmet atoms)."""
422    line = load_lines().get(pet.line_id)
423    row = None
424    if line:
425        # a dual-road target has one row per parent -- report the rule for
426        # THIS pet's road (its child row), not whichever row loaded first
427        row = next((r for r in line["children"].get(pet.num, [])
428                    if r["num"] == num), None) or line["members"].get(num)
429    if row is None:
430        return [(False, "not in this line")]
431    best, best_alt, best_unmet = None, None, None
432    for alt in row["rule"]:
433        rows = [_atom_row(pet, atom) for atom in alt]
434        unmet = sum(1 for met, _ in rows if not met)
435        if best is None or unmet < best_unmet:  # type: ignore
436            best, best_alt, best_unmet = rows, alt, unmet
437    win_atom = next((at for at in best_alt if at[0] == "win"), None) if best_alt else None
438    if best and win_atom:
439        # the WHICH-battles answer rides under the WIN row as a dim
440        # informational line (met=None, the jogress-door shape) -- inlining
441        # it in the row clipped the counter off the 35-char LCD (Joel's
442        # screenshot 2026-07-21, the "is it training or battles" report).
443        # While the gate is UNMET the same sub-row answers the more urgent
444        # question instead -- how far off you actually are.  `now/b` alone
445        # is a liar on a rolling window: wins that displace older wins
446        # move nothing, so the counter can freeze mid-winning-streak
447        # (Joel 2026-07-23).  Both strings stay inside the 35-char clip.
448        # BOTH sub-rows while short: the feed note answers WHICH battles
449        # count (and a grinder needs that most while unmet), the streak
450        # row answers HOW FAR.  The detail page scrolls, so the extra row
451        # costs nothing; each stays inside the clip.
452        _, wa, wb = win_atom
453        need = straight_needed(pet, wa, wb)
454        best = best + [(None, WIN_FEED_NOTE)]
455        if need:
456            best = best + [(None, f"needs {need} straight win"
457                                  f"{'s' if need > 1 else ''} from here")]
458    return best or [(True, "time alone — the clock decides")]

461def evo_rows(pet: Any) -> Any:
462    """(num, name, ready, unmet) per child of the current form, in chart
463    (first-match) order — the line data book page."""
464    line = load_lines().get(pet.line_id)
465    if not line:
466        return []
467    _, by_num = data.load_sprites()
468    out = []
469    for row in line["children"].get(pet.num, []):
470        rec = by_num.get(row["num"])
471        report = requirement_report(pet, row["num"])
472        unmet = sum(1 for met, _ in report if met is False)
473        out.append((row["num"], rec["name"] if rec else f"#{row['num']}",
474                    check_rule(pet, row["rule"]), unmet))
475    return out

49    'TIME' (or blank) parses to one unconstrained alternative."""
50    text = (text or "").strip()
51    alts = []
52    for alt in text.split("|"):
53        atoms = []
54        for part in alt.split(","):
55            part = part.strip()
56            if not part or part.upper() == "TIME":
57                continue
58            kind, _, arg = part.partition(" ")
59            kind, arg = kind.upper(), arg.strip()
60            if kind == "WIN":
61                k, n = arg.split("/", 1)
62                atoms.append(("win", int(k), int(n)))
63            elif kind == "JOGRESS":
64                # a jogress DOOR (LINES_SPEC §6): arg = the exact partner dex,
65                # or a Pendulum attribute spec ("Data/Virus") -- any same-stage
66                # partner with a listed attribute resonates (pen20 manual).
67                # Never fires on the stage timer -- the lobby fusion opens it.
68                if arg.isdigit():
69                    atoms.append(("jogress", int(arg), None))  # type: ignore
70                else:
71                    attrs = tuple(a.strip() for a in arg.split("/") if a.strip())
72                    if not attrs or any(a not in ("Vaccine", "Data", "Virus",
73                                                  "None") for a in attrs):
74                        raise ValueError(f"bad JOGRESS partner: {part!r}")
75                    atoms.append(("jogress", attrs, None))  # type: ignore
76            elif kind == "AREA":
77                if not arg:
78                    raise ValueError(f"AREA atom needs a map: {part!r}")
79                atoms.append(("area", arg, None))  # type: ignore
80            elif kind in ("CM", "TR", "OF", "BTL", "LV", "KO6"):
81                lo, hi = _bracket(arg)
82                atoms.append((kind.lower(), lo, hi))
83            else:
84                raise ValueError(f"unknown rule atom: {part!r}")
85        alts.append(atoms)
86    return alts
