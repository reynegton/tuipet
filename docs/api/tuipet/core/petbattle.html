 35    def _change_rank(self, cat: Any) -> None:
 36        """Taste.changeRank: bump the eaten category's rank (+/- species pref bias); eating
 37        your current favourite/disliked pulls the OTHER ranks back toward 0; clamp to
 38        +/-RankLimit; a rank that reaches the cap becomes the new favourite/disliked."""
 39        pref, aver, _ = self._species_food()  # type: ignore
 40        delta = RANK_CHANGE_FOOD
 41        if cat == pref:
 42            delta += RANK_PREF_INC
 43        elif cat == aver:
 44            delta -= RANK_PREF_INC
 45        if cat == self.disliked_food:  # type: ignore
 46            delta += RANK_DISLIKED
 47            for c in data.FOOD_CATEGORIES:                 # incRankExcept toward 0
 48                if c != cat and self.food_ranks[c] < 0:  # type: ignore
 49                    self.food_ranks[c] = min(0, self.food_ranks[c] + RANK_AFTER_FAV)  # type: ignore
 50        if cat == self.favorite_food:  # type: ignore
 51            for c in data.FOOD_CATEGORIES:                 # decRankExcept toward 0
 52                if c != cat and self.food_ranks[c] > 0:  # type: ignore
 53                    self.food_ranks[c] = max(0, self.food_ranks[c] - RANK_AFTER_FAV)  # type: ignore
 54        self.food_ranks[cat] = _clamp(self.food_ranks[cat] + delta, RANK_MIN, RANK_LIMIT)  # type: ignore
 55        for c in data.FOOD_CATEGORIES:
 56            if self.food_ranks[c] >= RANK_LIMIT:  # type: ignore
 57                self.favorite_food = c
 58            elif self.food_ranks[c] <= RANK_MIN:  # type: ignore
 59                self.disliked_food = c

 65    def _promote_attr_ranks(self) -> None:
 66        """A rank at +-RankLimit becomes the emergent favourite/disliked
 67        (simplified from Taste.setNewFavDislike like the food port: no
 68        repeat-collision reroll -- all three attributes are valid both ways)."""
 69        for a in self._ATTR3:  # type: ignore
 70            if self.attr_ranks[a] >= RANK_LIMIT and self.favorite_attr != a:  # type: ignore
 71                self.favorite_attr = a
 72                if self.disliked_attr == a:  # type: ignore
 73                    self.disliked_attr = ""
 74            elif self.attr_ranks[a] <= RANK_MIN and self.disliked_attr != a:  # type: ignore
 75                self.disliked_attr = a
 76                if self.favorite_attr == a:
 77                    self.favorite_attr = ""

 79    def _change_attr_rank(self, attr: Any) -> None:
 80        """Taste.changeRank for the attribute ledger: a drill warms the pet to
 81        its attribute (stage-scaled base, +-2 species preference/aversion
 82        bias); drilling the current favourite decays the others toward 0."""
 83        if attr not in self._ATTR3:  # type: ignore
 84            return
 85        req = self._phys()  # type: ignore
 86        delta = RANK_CHANGE_ATTR + self._rank_stage_inc()
 87        if attr == req.get("attr_pref", "None"):
 88            delta += RANK_PREF_INC
 89        elif attr == req.get("attr_aversion", "None"):
 90            delta -= RANK_PREF_INC
 91        if attr == self.disliked_attr:
 92            delta += RANK_DISLIKED
 93            for a in self._ATTR3:                          # type: ignore
 94                if a != attr and self.attr_ranks[a] < 0:  # type: ignore
 95                    self.attr_ranks[a] = min(0, self.attr_ranks[a] + RANK_AFTER_FAV)  # type: ignore
 96        if attr == self.favorite_attr:
 97            for a in self._ATTR3:                          # type: ignore
 98                if a != attr and self.attr_ranks[a] > 0:  # type: ignore
 99                    self.attr_ranks[a] = max(0, self.attr_ranks[a] - RANK_AFTER_FAV)  # type: ignore
100        self.attr_ranks[attr] = _clamp(self.attr_ranks[attr] + delta, RANK_MIN, RANK_LIMIT)  # type: ignore
101        self._promote_attr_ranks()

304        (Pen20 WIN gates), stage_battles (BTL gates), lifetime wins + the

307        power directly).  ONLINE PvP IS PROGRESSION-NEUTRAL (L17 ruling,
308        Joel 2026-07-20, option a): exp and KO6 always excluded it as
309        untrusted -- colluding tamers, forged cards -- and the same
310        collusion farmed the channels that stayed fed; an online bout now
311        bills the BODY only (energy/weight; its purse and ladder standing
312        live server-side).  battles+wins move together so win_rate stays
313        coherent.  The old free_style/low_health params are
314        accepted-and-ignored for stragglers.  (Mood/compliance/contagion

355                self.rival_wins += 1  # type: ignore
356            else:
357                self.rival_losses += 1  # type: ignore
358        # A BOUT TRAINS, ON BOTH CLOCKS (Joel 2026-07-25: "feed the
359        # total_trainings thing too, flip it").  The +2 is the clone rule;
360        # what the battle audit found was that only the STAGE counter got
361        # it, so fighting fed the TR evolution gate and the hit formula's
362        # +10% term while the lifetime +20% term -- the bigger of the two

391        self._open_praise()          # type: ignore
