tuipet.core.battle

Battle — the 0.5 HP race (clone rebuild 2026-07-15; ported onto the classic tree 2026-07-17, Joel: "replace the battle system" — DSprite is the ultimate truth for animations and mechanics).

Both sides start at 5 HP. The whole fight is PRECOMPUTED round by round (the screen replays it): each round both sides roll to hit; damage per landed hit is the side's PROJECTILE count, which comes from its trained hit-type (mega: 90% double / normal: 50% / miss: 20%). The per-round hit chance is care's scoreboard — win rate, stage rank, this-stage training, lifetime training, strength/hunger/energy meters, weight closeness and the attribute triangle all feed it. Transcribed EXACTLY from the source rule set (ripped from multiple fan games).

 34def _tri(mine: Any, theirs: Any) -> Any:
 35    """The attribute triangle: ±0.05 hit chance."""
 36    a, b = (mine or "").lower(), (theirs or "").lower()
 37    if (a, b) in (("vaccine", "virus"), ("virus", "data"), ("data", "vaccine")):
 38        return 0.05
 39    if (a, b) in (("virus", "vaccine"), ("data", "virus"), ("vaccine", "data")):
 40        return -0.05
 41    return 0

114    def _condition(self) -> Any:
115        """The meter terms of the hit formula (each 0..0.1 - 0.05)."""
116        st = (self.strength / self.strength_max) * 0.1 - 0.05
117        hu = (self.hunger / self.hunger_max) * 0.1 - 0.05
118        en = (self.energy / self.energy_max) * 0.1 - 0.05
119        if self.base_weight > 0:
120            w = (1 - min(1, abs(self.weight - self.base_weight) / 15)) * 0.2 - 0.1
121        else:
122            w = 0
123        return st + hu + en + w

125    def hit_chance(self, other: Any) -> Any:
126        wr = self.wins / self.battles if self.battles > 0 else 0.5
127        p = (0.3 + (wr - 0.5) * 0.1
128             + (self._rank() - other._rank()) * 0.1
129             + min(self.trainings_cur / 999 * 0.1, 0.1)
130             + min(self.trainings_total / 9999 * 0.2, 0.2)
131             + self._condition()
132             + _tri(self.attribute, other.attribute)
133             # THE PEN20 SHAKE RULING (Joel 2026-07-23 "ok do it"): the
134             # lock is HEART, and heart can only help -- the Pendulum's
135             # shake charge, not a reflex exam.  A clean mega lock adds
136             # aim (+0.10) and steadies the guard (the other side's term
137             # below); normal and miss both just fight AT YOUR STATS --
138             # no lock can ever make a pet fight worse than it was
139             # raised.  Attentive duelists both lock mega, the terms
140             # cancel, and the raising decides -- exactly the device.
141             # (Supersedes the 2026-07-23 ±0.10 punish-tier stakes, same
142             # day: a terminal is a bad reflex judge, and the miss tier
143             # was the one mechanic left that could betray a player who
144             # did everything right.)
145             + (0.10 if self.hit_type == "mega" else 0.0)
146             + (-0.10 if other.hit_type == "mega" else 0.0))
147        return min(1.0, max(0.0, p))
148
149    def roll_damage(self, rng: Any) -> Any:
150        # Pen20 shake ruling: a miss deals the plain 50% tier, same as
151        # normal -- the shanked-lock damage PENALTY (20%) died with the
152        # miss punish tier.  Only the mega bonus remains.
153        if self.hit_type == "mega":
154            return 2 if rng() < 0.9 else 1
155        return 2 if rng() < 0.5 else 1

158def generate(me: Any, foe: Any, rounds: Any=ROUNDS_LOCAL, rng: Optional[Any]=None) -> Any:
159    """The whole fight: [(my_hit, my_dmg, foe_hit, foe_dmg), ...] plus final
160    HPs.
161
162    DEATH IS FINAL within a round (Joel 2026-07-22: "why are foes still
163    shooting after their energy bars are at 0?"): the theater plays the
164    player's volley first, and the engine now resolves in that same order
165    -- a side dropped to 0 by the round's earlier volley never returns
166    fire.  The old exchange was simultaneous, so a freshly-KO'd foe still
167    landed its shot AFTER its bar hit zero (and a double-KO needed a coin
168    flip on who whiffs -- both leave with it).  NOTE for the lobby: a
169    mixed-version peer computes the old fight from the same seed, so
170    reports disagree and the ladder simply doesn't credit until both
171    sides run this build."""
172    rng = rng or random.random
173    my_hp, foe_hp = HP, HP
174    p_me, p_foe = me.hit_chance(foe), foe.hit_chance(me)
175    seq = []
176    for _ in range(rounds):
177        if my_hp <= 0 or foe_hp <= 0:
178            break
179        my_dmg = me.roll_damage(rng)
180        my_hit = rng() < p_me
181        foe_dmg = foe.roll_damage(rng)
182        foe_hit = rng() < p_foe
183        if my_hit:
184            foe_hp -= my_dmg
185            if foe_hp <= 0:
186                foe_hit = False          # no posthumous return fire
187        if foe_hit:
188            my_hp -= foe_dmg
189        seq.append((my_hit, my_dmg, foe_hit, foe_dmg))
190    return seq, max(0, my_hp), max(0, foe_hp)

193def _drag_key(mine: Any, theirs: Any) -> Any:
194    """The ONE drag detector behind the coach line and the pre-fight
195    readiness line (Pen20 honesty 2026-07-23: two surfaces, one truth).
196    Precedence = fixability: condition anomalies first (fixable
197    tonight), then the rank gap, then the training ledger."""
198    if mine.base_weight > 0 and abs(mine.weight - mine.base_weight) >= 8:
199        return "weight"
200    if mine.hunger <= 1:
201        return "hunger"
202    if mine.energy <= mine.energy_max * 0.4:
203        return "energy"
204    if mine.strength <= 1:
205        return "effort"
206    if theirs._rank() > mine._rank():
207        return "rank"
208    if mine.trainings_total < 1000:
209        return "drills"
210    return ""

245class Battle:
246    """A playable fight the screen steps through round by round.
247
248    ALWAYS A LOCAL BOUT (the `source` parameter was CUT 2026-07-25 on
249    Joel's order, battle audit §5).  Nothing ever built a Battle any other
250    way: a lobby duel is fought on the relay and merely REPLAYED by a
251    presentation-only BattlePanel, and the lobby files its own result with
252    `record_battle(online=True)` -- the one live door to the online rules.
253    A fight that runs THIS engine on THIS pet is local, full stop, so it
254    records like one and always pays the +2."""
255
256    def __init__(self, pet: Any, enemy: Optional[Any]=None, rng: Optional[Any]=None, rounds: Any=ROUNDS_LOCAL) -> None:
257        self.pet = pet
258        rng = rng or random.random
259        if enemy is None:
260            enemy = pick_enemy(pet)
261        self.enemy = enemy                     # dict, kept for the HUD
262        # (the clone duck-typed on hunger_max, a clone-pet field; the classic
263        # pet has flat 4-heart gauges, so type on Side directly)
264        self.me = pet if isinstance(pet, Side) else Side.of_pet(pet)
265        self.foe = enemy["side"] if "side" in enemy else Side.wild(
266            enemy.get("num", 0), boss=bool(enemy.get("boss")))
267        self.seq, self._my_end, self._foe_end = generate(
268            self.me, self.foe, rounds=rounds, rng=rng)
269        self.pet_hp, self.enemy_hp = HP, HP
270        self.pet_max = self.enemy_max = HP
271        self.round = 0
272        self.over = False
273        self.won = False
274        self.drawn = False           # both stood at equal HP (counts as a loss)
275        self.reward = ""
276
277    def play_round(self, _choice: Optional[Any]=None) -> Any:
278        """Advance one precomputed round -> the round record for the stage
279        show: dict(pdmg, edmg, my_hit, foe_hit, ph, fh)."""
280        if self.over or self.round >= len(self.seq):
281            self._finish()
282            return None
283        my_hit, my_dmg, foe_hit, foe_dmg = self.seq[self.round]
284        self.round += 1
285        if my_hit:
286            self.enemy_hp = max(0, self.enemy_hp - my_dmg)
287        if foe_hit:
288            self.pet_hp = max(0, self.pet_hp - foe_dmg)
289        rec = {"pdmg": my_dmg if my_hit else 0,
290               "edmg": foe_dmg if foe_hit else 0,
291               "my_hit": my_hit, "foe_hit": foe_hit,
292               "ph": self.pet_hp, "fh": self.enemy_hp}
293        if self.pet_hp <= 0 or self.enemy_hp <= 0 \
294                or self.round >= len(self.seq):
295            self._finish()
296        return rec
297
298    def _finish(self) -> None:
299        if self.over:
300            return
301        self.over = True
302        # winner: the standing side; equal HP reads as a narrow win for the
303        # higher survivor, a draw counts as a loss for progression
304        if self.enemy_hp <= 0 and self.pet_hp > 0:
305            self.won = True
306        elif self.pet_hp <= 0:
307            self.won = False
308        else:
309            self.won = self.pet_hp > self.enemy_hp
310            # the silent rule made visible (gameplay polish #5): a true
311            # draw always read as an unexplained loss on the result card
312            self.drawn = self.pet_hp == self.enemy_hp
313        if hasattr(self.pet, "record_battle"):
314            self.pet.record_battle(self.won, self.enemy)
315        # record_battle grants the +2 on EVERY local bout, win or lose --
316        # gating the line on won left the losing grant silent under a
317        # DEFEAT card with an empty reward line
318        self.reward = "training +2"
319
320    def surrender(self) -> None:
321        if self.over:
322            return          # the bout already ended -- never a second
323            #                 record_battle on top of _finish's (audit 2026-07-19)
324        self.over = True
325        self.won = False
326        if hasattr(self.pet, "record_battle"):
327            self.pet.record_battle(False, self.enemy)
328        self.reward = "training +2"          # the surrendered bout still bills + trains

331class RaidBout:
332    """The raid attempt wearing the Battle interface, so the battlescreen
333    can replay it with the full volley show (tuipet raid adaptation: the
334    clone ran raids as lobby text -- Joel demanded sprites).  10 rounds
335    from RAID_PLAYER_HP; the boss NEVER falls (its bar holds at full) and
336    `dealt` accumulates for the relay report.  Writes NOTHING on the pet,
337    exactly like the clone's generate_raid."""
338
339    def __init__(self, pet: Any, boss: Any, rng: Optional[Any]=None) -> None:
340        self.pet = pet
341        self.enemy = boss                        # dict, kept for the HUD
342        me = Side.of_pet(pet)
343        foe = Side.wild(int(boss.get("num", 0)), boss=True)
344        self.seq, self.dealt, self._end_hp = generate_raid(me, foe, rng=rng)
345        self.pet_hp = self.pet_max = RAID_PLAYER_HP
346        self.enemy_hp = self.enemy_max = HP      # display only
347        self.round = 0
348        self.over = False
349        self.won = False
350        self.reward = ""
351
352    def play_round(self, _choice: Optional[Any]=None) -> Any:
353        if self.over or self.round >= len(self.seq):
354            self._finish()
355            return None
356        my_hit, my_dmg, boss_hit, boss_dmg = self.seq[self.round]
357        self.round += 1
358        if boss_hit:
359            self.pet_hp = max(0, self.pet_hp - boss_dmg)
360        rec = {"pdmg": my_dmg if my_hit else 0,
361               "edmg": boss_dmg if boss_hit else 0,
362               "my_hit": my_hit, "foe_hit": boss_hit,
363               "ph": self.pet_hp, "fh": self.enemy_hp}
364        if self.pet_hp <= 0 or self.round >= len(self.seq):
365            self._finish()
366        return rec
367
368    def _finish(self) -> None:
369        self.over = True                         # no record_battle: a report, not a bout
370        # presentation only: the ATTEMPT succeeded when the pet stood through
371        # its rounds -- won=False forever ended EVERY raid on the loser
372        # collapse frame, even a full-damage run (the boss still never falls)
373        self.won = self.pet_hp > 0
374
375    def surrender(self) -> None:
376        self.over = True

379def generate_raid(me: Any, boss: Any, rng: Optional[Any]=None) -> Any:
380    """The raid attempt: 10 rounds against an effectively-invincible boss.
381    The player fights from 10 HP; the boss never falls IN the fight -- the
382    raw damage landed here is what the shared pool eats (x5000 x stage-mult,
383    applied server-side).  Returns (rounds, raw_dealt, my_end_hp)."""
384    rng = rng or random.random
385    my_hp = RAID_PLAYER_HP
386    p_me, p_boss = me.hit_chance(boss), boss.hit_chance(me)
387    seq = []
388    dealt = 0
389    for _ in range(ROUNDS_RAID):
390        if my_hp <= 0:
391            break
392        my_dmg = me.roll_damage(rng)
393        my_hit = rng() < p_me
394        boss_dmg = boss.roll_damage(rng)
395        boss_hit = rng() < p_boss
396        if my_hit:
397            dealt += my_dmg
398        if boss_hit:
399            my_hp -= boss_dmg
400        seq.append((my_hit, my_dmg, boss_hit, boss_dmg))
401    return seq, dealt, max(0, my_hp)

404def pick_enemy(pet: Any, boss: bool=False) -> Any:
405    """A wild foe scaled to the pet: same stage bracket, one up for a boss."""
406    ladder = ["InTraining", "Rookie", "Champion", "Ultimate", "Mega"]
407    st = pet.stage if pet.stage in ladder else "Rookie"
408    if boss and ladder.index(st) < len(ladder) - 1:
409        st = ladder[ladder.index(st) + 1]
410    pool = [r for r in data.load_sprites()[0] if r["stage"] == st]
411    rec = random.choice(pool)
412    return {"num": rec["num"], "name": rec["name"], "boss": boss,
413            "stage": rec["stage"], "attribute": rec["attribute"]}

159def generate(me: Any, foe: Any, rounds: Any=ROUNDS_LOCAL, rng: Optional[Any]=None) -> Any:
160    """The whole fight: [(my_hit, my_dmg, foe_hit, foe_dmg), ...] plus final
161    HPs.
162
163    DEATH IS FINAL within a round (Joel 2026-07-22: "why are foes still
164    shooting after their energy bars are at 0?"): the theater plays the
165    player's volley first, and the engine now resolves in that same order
166    -- a side dropped to 0 by the round's earlier volley never returns
167    fire.  The old exchange was simultaneous, so a freshly-KO'd foe still
168    landed its shot AFTER its bar hit zero (and a double-KO needed a coin
169    flip on who whiffs -- both leave with it).  NOTE for the lobby: a
170    mixed-version peer computes the old fight from the same seed, so
171    reports disagree and the ladder simply doesn't credit until both
172    sides run this build."""
173    rng = rng or random.random
174    my_hp, foe_hp = HP, HP
175    p_me, p_foe = me.hit_chance(foe), foe.hit_chance(me)
176    seq = []
177    for _ in range(rounds):
178        if my_hp <= 0 or foe_hp <= 0:
179            break
180        my_dmg = me.roll_damage(rng)
181        my_hit = rng() < p_me
182        foe_dmg = foe.roll_damage(rng)
183        foe_hit = rng() < p_foe
184        if my_hit:
185            foe_hp -= my_dmg
186            if foe_hp <= 0:
187                foe_hit = False          # no posthumous return fire
188        if foe_hit:
189            my_hp -= foe_dmg
190        seq.append((my_hit, my_dmg, foe_hit, foe_dmg))
191    return seq, max(0, my_hp), max(0, foe_hp)

def generate_raid(me: Any, boss: Any, rng: Optional[Any] = None) -> Any: 
380def generate_raid(me: Any, boss: Any, rng: Optional[Any]=None) -> Any:
381    """The raid attempt: 10 rounds against an effectively-invincible boss.
382    The player fights from 10 HP; the boss never falls IN the fight -- the
383    raw damage landed here is what the shared pool eats (x5000 x stage-mult,
384    applied server-side).  Returns (rounds, raw_dealt, my_end_hp)."""
385    rng = rng or random.random
386    my_hp = RAID_PLAYER_HP
387    p_me, p_boss = me.hit_chance(boss), boss.hit_chance(me)
388    seq = []
389    dealt = 0
390    for _ in range(ROUNDS_RAID):
391        if my_hp <= 0:
392            break
393        my_dmg = me.roll_damage(rng)
394        my_hit = rng() < p_me
395        boss_dmg = boss.roll_damage(rng)
396        boss_hit = rng() < p_boss
397        if my_hit:
398            dealt += my_dmg
399        if boss_hit:
400            my_hp -= boss_dmg
401        seq.append((my_hit, my_dmg, boss_hit, boss_dmg))
402    return seq, dealt, max(0, my_hp)
