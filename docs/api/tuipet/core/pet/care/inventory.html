 18def take_item(pet: Any, key: str, n: int=1) -> None:
 19    """Spend n from the bag, dropping the key at zero -- add_item's mirror
 20    (this decrement lived in four hand-rolled copies; refactor 2026-07-05)."""

 24    else:
 25        pet.inventory[key] = left

 37def _compensate_attrs(pet: Any) -> Any:
 38    """compensateAttributes x3 rotations: each negative power borrows from
 39    the next two in canon's order.  (Canon's zero-all escape only fires
 40    when all THREE are negative -- with both banks empty its loop would
 41    spin forever; unreachable with the shipped symmetric trades, and the
 42    port floors the deficit at 0 instead of freezing.)"""
 43    def comp(main: Any, weak: Any, normal: Any) -> Any:
 44        while main < 0:
 45            if weak > 0:
 46                weak -= 1
 47                main += 1
 48            if main < 0 and normal > 0:
 49                normal -= 1
 50                main += 1
 51            if weak <= 0 and normal <= 0 and main < 0:
 52                return 0, weak, normal       # the safe floor (see docstring)
 53        return main, weak, normal
 54    v, d, vi = pet.vaccine, pet.data_power, pet.virus
 55    v, d, vi = comp(v, d, vi)
 56    d, vi, v = comp(d, vi, v)
 57    vi, v, d = comp(vi, v, d)
 58    pet.vaccine, pet.data_power, pet.virus = v, d, vi

185def _crest_egg(pet: Any, key: str) -> Any:
186    """A crest egg -> the classic Relic item-evolution flow."""
187    if pet.dead or pet.stage == "Egg" or pet.num < 0:
188        return Refused("")
189    item_id = pet._CREST_IDS.get(key, -1)
190    target = evolution.item_select(pet, item_id)
191    if target is None:
192        pet._set_anim("refuse", 1.0)
193        return Refused(f"{pet.name} can't use that yet.")
194    if pet.asleep:
195        pet._disturbed()
196    prev = pet.num
197    pet.evolve_to(target)
198    lines_mod.adopt_line(pet, prev=prev)     # a special jump re-anchors
199    pet.take_item(key)
200    pet._set_anim("happy", 1.6)
201    import tuipet.utils.persistence as _persist
202    _persist.armor_add(1)                 # the crest-wave shop gate counts it
203    return f"{pet.name} armor-evolved!"

216def _snack(pet: Any, hunger: int=0, energy: int=0, weight: int=0, obedience: int=0, powers: Optional[Any]=None,
217           strength: int=0) -> Any:
218    """The TUIPET food family (2018-07-18 -> grown 2026-07-26): plain
219    live-meter meals.  Positive-hunger food is refused at a full belly,
220    like every meal.  The expansion legs (obedience / VDV powers /
221    effort) land the authored columns of the new rows -- a pepper's +1
222    power rides the chip grammar, effort clamps to its 0-4 gauge."""
223    if hunger > 0 and pet.hunger >= FULL_HUNGER:
224        return Refused("Refused - belly's full.")
225    if hunger:
226        pet.hunger = _clamp(pet.hunger + hunger, 0, FULL_HUNGER)
227    if energy:
228        pet._set_energy(pet.energy + energy)
229    if weight:
230        pet._set_weight(max(1, pet.weight + weight))
231    if obedience:
232        pet._set_obedience(pet.obedience + obedience)
233    if strength:
234        pet.strength = _clamp(pet.strength + strength, 0, 4)  # noqa: F405
235    if powers:
236        v, d, vi = powers
237        pet.vaccine += v
238        pet.data_power += d
239        pet.virus += vi
240    return "Nham nham."

279def _caffeine(pet: Any) -> Any:
280    """Tonight's bedtime pushed later: a quarter of the night off the
281    clock the pet ACTUALLY sleeps by.  Line pets (every hatch) read the
282    wall-clock window, not sleep_lapse -- the old pressure-only nudge
283    made this a paid no-op for them (gameplay audit 2026-07-19); their
284    push rides the same grace channel a disturb uses.
285
286    THE NO-OP DOSE IS REFUSED (item sweep 2026-07-24).  Both branches
287    could spend a 300b pill and move nothing -- a second pill while the
288    grace already holds that push, or a pressure pet whose sleep_lapse
289    is still 0 (nowhere near bedtime) -- while saying "Wide awake for a
290    while yet."  Every care sibling refuses at full instead ("Energy is
291    already full", "already a model pupil"); this was the outlier."""
292    if pet.asleep:
293        return Refused("Too late - it's already down.")
294    if pet._in_sleep_window() is not None:
295        bt = lines_mod.bedtime_minutes(pet)
296        night = (pet.WAKE_MINUTE - bt) % DAY_MINUTES
297        push = night * 0.25
298        if getattr(pet, "_bed_postpone_t", 0.0) >= push:
299            return Refused("Bedtime's already pushed back.")  # noqa: F405
300        pet._bed_postpone_t = push
301    else:
302        if pet.sleep_lapse <= 0:
303            return Refused("It's nowhere near bedtime.")      # noqa: F405
304        pet.sleep_lapse = max(0.0, pet.sleep_lapse - pet.sleep_limit * 0.25)
305    return "Bem acordado por um tempo ainda."
