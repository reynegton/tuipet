tuipet.core.tournament

Tournament -- DVPet's HOME tournament (Tournament.java + tournies.csv).

The real thing is a SCHEDULED 8-entrant bracket, not an always-open menu:

  • Each game day, dailyChange rolls a 24-slot trophy schedule (one cup per game hour) from the season's pool -- randTrophyIDs interleaves the age tiers (open/Rookie/open/Champion/open/Ultimate/open/Mega) so tier cups pepper the day. Only the CURRENT hour's cup is enterable (checkTourneyClosed).
  • Entry runs Tournament.isEligible: not already fought today (SameDayRetry cups exempt), not too old for an age-limited cup, field/attribute restrictions, and a Prelim chain (its qualifier beaten this season). DVPet's fully-recovered gate has no tuipet analog (battle HP is per-fight).
  • The bracket is the player + 7 entrants drawn from the DEX (TournamentAble forms filtered by the cup's stage/field/attribute rules; the default tier follows the pet's AGE via TourneyRandom*Age). Each entrant rolls real stats: stage-banded HP and a power total split main/2 weak/6 rest/3 by its attribute. Pairs are (0,1)(2,3)(4,5)(6,7) with the player at a random slot; the OTHER matches auto-resolve between rounds.
  • Prizes: calcBits = sum over the 7 entrants of their stage's Tourney*Bits x the cup's BitModifier (a Mega entrant pays TourneyMaxBits once the pet is past TourneyRandomMegaAge). Losing still pays: nothing from the quarterfinal, a third from the semi, half from the final.

 73def featured_now(pet: Any, today: Optional[Any]=None) -> Any:
 74    """The day's FEATURED cup: one per real date, drawn from the season's
 75    pool by a date-seeded roll.  The weekend headliner draws from the PET's
 76    own bracket or the open tier (cup ruling 2026-07-18: the old top-tier-
 77    only weekend draw handed a Champion pressing F a near-unwinnable Mega
 78    bracket -- the marquee event should be winnable by whoever's playing).
 79    Open at ANY hour, once per real day, on top of the hourly board."""
 80    d = today or _today()
 81    pool = [t for t in data.load_tournies() if t["season"] == real_season(d)]
 82    if is_weekend(d):
 83        mine = pet_tier(pet)                # None = a Mega pet: the open field
 84        top = [t for t in pool
 85               if not t["age_limit"] or (mine and t["age_limit"] == mine)]
 86        pool = top or pool
 87    if not pool:
 88        return None
 89    rng = random.Random(d.toordinal())
 90    return rng.choice(pool)

194def entry_fee(pet: Any, t: Any) -> Any:
195    """THE STAKE (bit-sink design 2026-07-14): a quarter of the bracket's
196    EXPECTED purse (7 entrants x the tier's stage bits x BitModifier), put
197    down at entry.  The champion nets +75% of the purse, a final loss +25%,
198    a semi loss ~+8%, and a quarterfinal exit eats the stake whole -- cups
199    were the game's dominant faucet with zero risk.  An open cup stakes the
200    pet's own tier: a Mega pays the open-field MaxBits rate it also wins by."""
201    if t["age_limit"]:
202        base = TOURNEY_BITS.get(t["age_limit"], 0)
203    elif pet_tier(pet) is None:
204        base = TOURNEY_MAX_BITS
205    else:
206        base = TOURNEY_BITS.get(pet_tier(pet), TOURNEY_BITS["Rookie"])
207    return int(7 * base * t["bit_mod"]) // ENTRY_FEE_DIV

210def _rand_trophy_ids(pet: Any) -> Any:
211    """Tournament.randTrophyIDs: bucket the cups by age tier, then fill
212    the 24 hourly slots rotating open/Rookie/open/Champion/open/Ultimate/open/
213    Mega; the fill STOPS at the first empty bucket (canon quirk).  Every cup in
214    the classic data has Time=None, so the time-of-day match never gates.
215    The day's pool is the REAL season's cups (cadence layer 2026-07-17): the
216    CSV Season column is live again, keyed to the actual calendar."""
217    season = real_season()
218    buckets = {"free": [], "Rookie": [], "Champion": [], "Ultimate": [], "Mega": []}  # type: ignore
219    for t in data.load_tournies():
220        if t["season"] != season:
221            continue
222        buckets[t["age_limit"] if t["age_limit"] in buckets else "free"].append(t)
223    order = ["free", "Rookie", "free", "Champion", "free", "Ultimate", "free", "Mega"]
224    sched = []  # type: ignore
225    while len(sched) < HOME_LIMIT:
226        for name in order:
227            pool = buckets[name]
228            if not pool:
229                return sched + [-1] * (HOME_LIMIT - len(sched))
230            t = pool.pop(random.randrange(len(pool)))
231            sched.append(t["id"])
232            if len(sched) >= HOME_LIMIT:
233                break
234    return sched

237def schedule(pet: Any) -> Any:
238    """The day's hourly cup schedule (setTrophySchedule; dailyChange re-rolls
239    it and clears foughtTrophiesToday)."""
240    from tuipet.core.pet import DAY_LENGTH
241    day = int(pet.world_seconds // DAY_LENGTH)
242    real = _today().toordinal()
243    if pet.tourney_day != day or getattr(pet, "tourney_real", -1) != real \
244            or len(pet.tourney_schedule) != HOME_LIMIT:
245        pet.tourney_day = day
246        pet.tourney_real = real
247        pet.tourney_schedule = _rand_trophy_ids(pet)
248        pet.fought_today = []
249        pet.fought_hours = []                      # a new day, every cup-hour fresh
250        pet.tourney_alarm = -1                     # dailyChange: _tourneyAlarm = -1
251        pet.tourney_alert = False
252    return pet.tourney_schedule

265def eligibility(pet: Any, t: Any) -> Any:
266    """Tournament.isEligible, minus the fully-recovered gate (no persistent
267    battle HP in tuipet).  Returns a refusal reason or None.
268
269    THE CUP-HOUR GATE (Joel 2026-07-13, economy audit): every trophy in the
270    shipped data carries SameDayRetry=TRUE, so canon's foughtTrophiesToday
271    lock never fires -- the open cup could be re-entered without limit,
272    re-rolling its bracket for the full purse each time (~1,500b a minute,
273    an order of magnitude past an adventure).  tuipet's rule: **the cup RUNS
274    once per hour**.  Entering spends that hour's slot; the next hour brings
275    a fresh cup.  Canon's own per-trophy lock is kept underneath for any
276    future SameDayRetry=FALSE data.
277    """
278    if _hour(pet) in (getattr(pet, "fought_hours", None) or []):
279        return "That cup has run — the next one starts on the hour."
280    # (festival days route through eligibility_at -- any un-run slot enters)
281    # ONE gate: the same rest+stake chain the live paths (eligibility_at /
282    # eligibility_featured) run.  This body was a hand-copy of the same
283    # rules and the two could drift silently (cup review 2026-07-18).
284    return (_eligibility_rest(pet, t) or _stake_check(pet, t)
285            or pet.battle_condition())   # the ONE bout condition gate (audit 2026-07-19)

288def eligibility_at(pet: Any, t: Any, slot: Any) -> Any:
289    """Eligibility for entering the SLOT's cup right now.  Ordinarily only
290    the current hour's slot takes entries; on a FESTIVAL day (holiday())
291    every un-run slot is open -- each still runs exactly once, so the purse
292    can't be farmed."""
293    run = getattr(pet, "fought_hours", None) or []
294    if slot in run:
295        return "Essa copa já terminou."
296    if slot != _hour(pet) and not holiday():
297        return "That cup is closed — only the %02d:00 one runs now." % _hour(pet)
298    return (_eligibility_rest(pet, t) or _stake_check(pet, t)
299            or pet.battle_condition())   # the ONE bout condition gate (audit 2026-07-19)

317def next_winnable(pet: Any) -> Any:
318    """The next hour TODAY whose cup this pet can actually enter (eligibility
319    passes) -- scanning from the current hour forward through the schedule.
320    Returns (hour, trophy) or None when nothing enterable is left today."""
321    sched = schedule(pet)
322    run = getattr(pet, "fought_hours", None) or []
323    for i in range(_hour(pet), len(sched)):
324        tid = sched[i]
325        if tid < 0 or i in run:                    # that hour's cup has already run
326            continue
327        tr = trophy_by_id(tid)
328        # the hour gate only speaks for THIS hour; a FUTURE slot is judged on
329        # the rest of the rules (we have not reached it yet)
330        if tr and _eligibility_rest(pet, tr) is None:
331            return (i, tr)
332    return None

335def _eligibility_rest(pet: Any, t: Any) -> Any:
336    """Eligibility WITHOUT the cup-hour gate -- for judging a FUTURE slot."""
337    if t["id"] in (pet.fought_today or []) and not t["same_day_retry"]:
338        return "Já lutou nessa copa hoje."
339    if t["age_limit"] and _pet_tier_rank(pet) > _TIER_RANK.get(t["age_limit"], 3):
340        return "Too old for the %s bracket." % t["age_limit"]
341    if t["field_req"] and t["field_req"] != getattr(pet, "field", ""):
342        return "%s only." % data.pretty_field(t["field_req"])
343    if t["attr_req"] and t["attr_req"] != getattr(pet, "attribute", ""):
344        return "%s only." % t["attr_req"]
345    if t.get("prelim"):
346        won = getattr(pet, "trophies_won", {}) or {}
347        if t["prelim"] not in won:
348            q = trophy_by_id(t["prelim"])
349            label = trophy_label(q) if q else "classificatória"
350            # the grand chain crosses REAL seasons -- name the missing
351            # link's season so the year-long arc reads as a journey, not a
352            # mystery wall (cup ruling 2026-07-18)
353            if q and q.get("season") and q["season"] != real_season():
354                return "Win the %s first (a %s cup)." % (label, q["season"])
355            return "Win the %s first." % label
356    return None

376def _eligible_forms(pet: Any, trophy: Any) -> Any:
377    """randomEnemies' entrant pool: TournamentAble dex forms matching the cup's
378    enemy overrides (or, absent those, its restrictions / the pet's age tier)."""
379    reqs = data.load_requirements()
380    _, by_num = data.load_sprites()
381    tier = trophy["enemy_stage"] or trophy["age_limit"] or pet_tier(pet) or "Mega"
382    out = []
383    for num, rec in by_num.items():
384        if rec["stage"] in ("Egg", "Fresh", "InTraining") or rec["stage"] != tier:
385            continue
386        if not reqs.get(num, {}).get("tournament_able", True):
387            continue
388        if trophy["enemy_field"]:
389            if rec["field"] != trophy["enemy_field"]:
390                continue
391        elif trophy["field_req"] and rec["field"] != trophy["field_req"]:
392            continue
393        if trophy["enemy_attr"]:
394            if rec["attribute"] != trophy["enemy_attr"]:
395                continue
396        elif trophy["attr_req"] and rec["attribute"] != trophy["attr_req"]:
397            continue
398        # (the enemy_elem filter left with the element system (ELEMENT SYSTEM REMOVED 2026-07-18)
399        # -- zero shipped cups used it)
400        if data.is_placeholder(num):
401            continue
402        out.append(rec)
403    return out

416def _npc_winner(a: Any, b: Any) -> Any:
417    """An NPC match runs the REAL 0.5 engine (2026-07-17): two wild Sides,
418    full HP race.  Initiative is a COIN (death-is-final 2026-07-22): the
419    engine resolves the 'me' side's volley first now, so a fixed argument
420    order would hand every bracket's first-listed entrant the edge."""
421    import tuipet.core.battle as battle

423        a, b = b, a

446class Tournament:
447    def __init__(self, pet: Any, trophy: Any, slot: Optional[Any]=None, featured: bool=False) -> None:
448        self.pet = pet
449        self.featured = featured
450        self._slot = slot
451        self.trophy = trophy
452        self.name = trophy_label(trophy)
453        self.round = 0
454        self.over = False
455        self.champion = False
456        self.reward_bits = 0
457        # the stake is paid AT ENTRY like the hour slot: a forfeit or an
458        # abandoned bracket does not hand it back.  eligibility() vets
459        # affordability first; a direct construction that cannot pay (tests,
460        # rogue callers) stakes nothing rather than silently owing.
461        self.stake = entry_fee(pet, trophy)
462        if not pet.spend_bits(self.stake):
463            self.stake = 0
464        pool = _eligible_forms(pet, trophy)
465        open_mega = not trophy["age_limit"] and not trophy["enemy_stage"] \
