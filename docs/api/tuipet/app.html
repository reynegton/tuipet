273    async def _check_update(self) -> None:
274            """Background, once per launch: ask PyPI for a newer tuipet and INSTALL
275            it (Joel 2026-07-14: "make it so the game automatically checks and
276            updates itself... then they have to restart for it to be the new one").
277    
278            Never blocks the game: this runs off the UI thread and the player keeps
279            playing the version they launched.  Python already imported that code,
280            so a fresh install can only take effect on the NEXT launch -- the nudge
281            says exactly that, and never claims a live swap.
282            Honest about what it cannot do (the silent-failure law): where we cannot
283            run pip for the player -- iOS sandboxes subprocesses, a source checkout
284            has no release to install over -- we fall back to telling them the
285            command.  A failed install says so and hands the command over too; it
286            never pretends the update happened.
287            """
288            import asyncio
289            latest = await asyncio.to_thread(update_check.latest_if_newer)
290            if not latest:
291                return
292            if not persistence.get_auto_update():          # the player opted out
293                self._update_msg = f"⬆ tuipet {latest} disponível — {update_check.manual_command()}"  # type: ignore
294                return
295            if update_check.upgrade_argv() is None:        # iOS / source: cannot self-install
296                self._update_msg = f"⬆ tuipet {latest} disponível — {update_check.manual_command()}"  # type: ignore
297                return
298            self._update_msg = f"⬆ instalando tuipet {latest}…"  # type: ignore
299            ok, _msg = await asyncio.to_thread(update_check.run_upgrade)
300            if ok:
301                self._updated_to = latest
302                self._update_msg = f"✔ tuipet {latest} instalado — reinicie para jogar"  # type: ignore
303            else:
304                self._update_msg = f"⬆ tuipet {latest} disponível — {update_check.manual_command()}"  # type: ignore

311    def on_key(self, event: Any) -> None:
312            fx = getattr(getattr(self, "screen_w", None), "fx", None)
313            if fx is not None and fx.get("kind") == "dying":
314                # dying(): the pet is a BUTTON -- frantic taps can save it
315                # (numHits > HitsToSave x (savedFromDeath + 1))
316                self._revive_hits = getattr(self, "_revive_hits", 0) + 1  # type: ignore
317                self.beep("click", bell=False)
318                event.stop()
319                event.prevent_default()
320                return
321            if (self.mode is None and not self.pet.dead
322                    and getattr(self.screen_w, "fx", None) is not None
323                    and event.key != "q"):
324                # canon disableMainMenu: the WHOLE menu locks while an animation
325                # plays (Joel 2026-07-06).  The 8 mutating care actions always
326                # guarded; the browse menus could still open mid-ceremony -- now
327                # every binding waits for the show.  q (quit) stays live; the
328                # dying-fx revive mash is handled above this gate.
329                # the swallowed key is ACKNOWLEDGED (QOL 2026-07-23): a silent
330                # eat read as a frozen app and invited a mash -- same soft
331                # click the dying-fx taps get
332                self.beep("click", bell=False)
333                event.stop()
334                event.prevent_default()
335                return
336            if self.mode is None and self.pet.dead:
337                # A departed pet can DO nothing (the device shows only the grave).
338                # ONE chokepoint ahead of every global binding -- the per-action
339                # can_*() gates kept slipping (a dead mon could still adventure,
340                # Joel 2026-07-05).  Any care key leads back to the memorial;
341                # quit, options and the LOBBY stay live beside the grave (Joel
342                # 2026-07-19: the social room is not a care action -- chat, DMs,
343                # ladder and rooms all work; battles/jogress stay refused by
344                # can_battle/can_jogress and the server's session gate).
345                if event.key not in ("q", "g", "l"):
346                    event.stop()
347                    event.prevent_default()
348                    if event.key == "n" and self.pet.death_banked:
349                        # the bare grave card promises "press N for a new egg"
350                        # -- deliver the CAROUSEL, not another lap through the
351                        # memorial (QOL 2026-07-23).  Pre-bank, N still runs
352                        # the ceremony below: the etch comes first.
353                        self.action_new()
354                        return
355                    # every care key leads to the memorial (the dead-gate law) --
356                    # and if this death's etch/seed ceremony hasn't run yet (a
357                    # relaunch mid-dying-beat), it runs HERE, so the inheritance
358                    # can never be lost to a quit (gameplay audit 2026-07-19).
359                    # An untouched relaunch still gets the dying beat + mash
360                    # window from the tick's state check instead.
361                    self._death_ceremony()
362                return
363            if self.mode is not None:
364                event.stop()
365                event.prevent_default()      # a panel owns the keyboard: don't fire global BINDINGS
366                # Textual names punctuation keys ("." -> "full_stop", "!" ->
367                # "exclamation_mark"), so a panel's `len(k) == 1 and k.isprintable()`
368                # text test silently dropped every non-alphanumeric.  For a
369                # text-capturing panel, forward the actual typed character instead of
370                # the key NAME (nav keys carry no printable character, so they still
371                # arrive by name; space stays "space" via its explicit handling).
372                k = event.key
373                ch = getattr(event, "character", None)
374                if (getattr(self.mode, "captures_text", False)
375                        and ch is not None and len(ch) == 1
376                        and ch.isprintable() and not ch.isspace()):
377                    k = ch
378                result = self.mode.key(k)
379                snd = getattr(self.mode, "sfx", None)
380                if snd:
381                    self.beep(snd, bell=False)
382                    self.mode.sfx = None
383                elif getattr(self.mode, "captures_text", False):
384                    pass                                # typing: no nav/confirm blips (audit 2026-07)
385                elif event.key in _NAV_KEYS:
386                    self.beep("scroll", bell=False)     # cursor-move blip for every list screen
387                elif event.key == "enter":
388                    self.beep("confirm", bell=False)    # menu confirm (a screen's own sfx wins above)
389                elif event.key == "escape":
390                    self.beep("cancel", bell=False)     # back/cancel
391                if result is not None and result[0] == "done":
392                    self._close_mode(result[1])
393                elif result is not None and result[0] == "quit":
394                    self.action_quit()                  # a screen asked to quit the app (e.g. q on the title)
395                elif event.key == "q" and not getattr(self.mode, "captures_text", False):
396                    self.action_quit()                  # QoL: q quits from any non-text screen, not just the main view
397                else:
398                    self.repaint()
