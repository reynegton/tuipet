tuipet.ui.screens.raidscreen

RAID — the community boss fight, on its own action key (BASIC VPET 2026-07-16; ported from the DSprite rebuild and given the sprite-rich screen Joel asked for: the BOSS stands on the arena, not in a text menu).

One boss stands per rotation with a SHARED HP pool on the relay (energy_max x 5.5M). Every tamer gets 3 attempts a day: an attempt is the clone's generate_raid (0.5 BATTLE 2026-07-17) — a 10-round volley from RAID_PLAYER_HP against a boss that never falls locally, replayed through the real battlescreen (the sprites Joel demanded). The raw damage landed is reported and the relay multiplies it against the pool (x5000 x stage-mult, bound server-side to the card's num). The volley writes NOTHING on the pet's record. When the community breaks the pool, the board archives and pays on claim — and a felled boss is a Mega down: the claim counts KO6 and the raids egg-unlock channel.

 32def _fmt(n: Any) -> Any:
 33    """Board damage is raw x5000 x mult — millions.  Keep it readable."""
 34    n = int(n)
 35    if n >= 10_000_000:
 36        return f"{n // 1_000_000}M"
 37    if n >= 1_000_000:
 38        return f"{n / 1_000_000:.1f}M"
 39    if n >= 1_000:
 40        return f"{n // 1_000}k"
 41    return str(n)

103                    # gate the press runs, read-only -- NEVER _disturbed()

107                            if self.pet.asleep else self.pet.battle_condition())

131                why = hit.get("why") or ""
132                default_refusal = t("raid_msg_refused", "O portal recusou o relatório.")

134                self.sfx = "error"  # type: ignore

156    def _apply_reward(self, reward: Any) -> None:
157        if not reward.get("ok"):
158            self.msg = t("raid_msg_nothing", "Nada a coletar.")
159            self.sfx = "error"  # type: ignore
160            return
161        bits = int(reward.get("bits", 0))
162        self.pet.bits += bits
163        got = list(reward.get("items") or [])
164        for key in got:
165            self.pet.add_item(key)
166        if reward.get("defeated"):
167            # the felled boss is a Mega down for everyone on the board: the
168            # claim pays the KO6 counter and the raids egg-unlock channel
169            self.pet.mega_kills += 1
170            persistence.mega_kills_add(1)
171            persistence.raid_add()
172            # the prize names come from shop.entry -- THE resolver for a
173            # catalog key (item sweep 2026-07-24).  This read vitems.json,
174            # which since the TUIPET catalog (2026-07-18) holds only the
175            # Relics plus the RETIRED consumable rows: the reward line
176            # for the game's biggest moment printed "Energy.D" (a name that
177            # left the shelf) and raw keys -- "Rank 1: 12000b + Energy.D,
178            # vitamin, dna_crystal".
179            names = ", ".join((shop.entry(k) or {}).get("name", k)
180                              for k in got) if got else ""
181            boss_name = reward.get('boss', t("raid_msg_the_boss", "The boss"))
182            rank = reward.get('rank', '?')
183            base_msg = t("raid_msg_fell_full", "{boss} fell! Rank {rank}: {bits}b").format(boss=boss_name, rank=rank, bits=bits)
184            self.msg = base_msg + (f" + {names}" if names else "")
185            self.sfx = "champion"  # type: ignore
186        else:
187            self.msg = t("raid_msg_escaped", "The boss escaped… {bits}b consolation.").format(bits=bits)
188            self.sfx = "confirm"  # type: ignore

208    def _report(self, b: Any) -> None:
209        if b is None:
210            # ESC before the bell: no volley rolled, no report, no attempt
211            # spent -- the old "Not a scratch" called the walk-away a whiff
212            # (raid round 2026-07-19)
213            self.msg = t("raid_msg_back_off", "You back off. The attempt keeps.")
214            return
215        # THE VOLLEY BILLS THE BODY (Joel 2026-07-28: "bill the body only").
216        # A raid was the ONE fight door that spent nothing -- RaidBout writes
217        # no record by design, so the pet threw itself at a Mega for free
218        # while every sibling bout (home key, cups, even L17 lobby duels)
219        # paid energy+weight.  Same single source as the lobby:
220        # record_battle(online=True) bills the body and touches no
221        # progression counter, no injury roll, no rolling log.  This seam
222        # fires exactly once per THROWN volley -- the walk-away above hands
223        # back None, and an ESC mid-show fast-runs the bout to completion.
224        self.pet.record_battle(bool(getattr(b, "won", False)), online=True)
225        dealt = int(getattr(b, "dealt", 0) or 0)
226        if dealt > 0 and self.view:
227            self.client.raid_hit(dealt)
228            self._dealt += dealt
229            # NEUTRAL until the ack lands: "reported!" used to stand even
230            # when the gate rejected it or the socket was down (raid review
231            # 2026-07-18); the ack path speaks the credit or the refusal
232            self.msg = t("raid_msg_landed", "Landed {dealt} — reporting to the gate…").format(dealt=dealt)
233        else:
234            self.msg = t("raid_msg_scratch", "Not a scratch. Rest and try again.")

236    def key(self, k: Any) -> Any:
237        if self.sub is not None:
238            r = self.sub.key(k)
239            if r is not None and r[0] == "done":
240                self.sub = None
241                self._report(r[1])
242            return None
243        if k in ("space", "enter"):
244            v = self.view
245            if not v:
246                self.msg = t("raid_msg_gate_no_ans", "The gate hasn't answered yet…")
247                self.client.raid_get()
248                return None
249            if not self._standing():
250                self.msg = t("raid_msg_not_stand", "O chefe não está ativo.")
251                self.sfx = "error"  # type: ignore
252                return None
253            if int(v.get("attempts", 0)) <= 0:
254                self.msg = t("raid_msg_no_att", "Nenhuma tentativa restante hoje.")
255                self.sfx = "error"  # type: ignore
256                return None
257            # THE SLEEPER ANSWERS FIRST (sleep audit 2026-07-25, S1).  The
258            # gate below took the BODY half of the house rule and left the
259            # SLEEP half behind: measured, a sleeping pet threw a full raid
260            # volley with `asleep=True, disturb=0` -- the only fight door in
261            # the game that neither woke it nor refused.  A volley is a
262            # player POKE, so it costs what every sibling poke costs: the
263            # home key (can_battle), both cups (can_enter), training,
264            # adventure and the care menu all wake the sleeper, bill the
265            # disturb and refuse THIS press.  (The lobby is the deliberate
266            # opposite -- it refuses without touching the pet, because a
267            # STRANGER's invite is not the player's finger.)
268            if self.pet.asleep:
269                self.msg = self.pet._disturbed()
270                self.sfx = "error"  # type: ignore
271                return None
272            # THE DEVICE'S GATE (battle audit ruling 2026-07-25, Joel: "as
273            # close to bandai vpet as much as possible"): a volley is a
274            # fight you CHOOSE, so the body answers first -- exactly as the
275            # home key, both cups and the lobby have always done.  The raid
276            # board asked only the relay's attempt count, so a pet that the
277            # house key refuses ("Too hurt to fight.") could still throw
278            # itself at a Mega three times a day.
279            if (cond := self.pet.battle_condition()) is not None:
280                self.msg = cond
281                self.sfx = "error"  # type: ignore
282                return None
283            # the clone raid bout: RaidBout precomputes generate_raid and
284            # the battlescreen replays it (records nothing on the pet)
285            self.sub = BattlePanel(self.pet, self._boss_enemy(), raid=True)  # type: ignore
286            return None
287        if k in ("c", "C"):        # both cases, like the lobby's letter keys
288            award = (self.view or {}).get("award")
289            if award:
290                self.client.raid_claim(award["id"])
291                self.msg = t("raid_msg_claim_sent", "Claim sent…")
292            else:
293                self.msg = t("raid_msg_nothing_yet", "Nada a coletar ainda.")
294            return None
295        if k in ("escape", "r"):
296            # the exit line speaks the GATE's number (board damage), not the
297            # raw accumulator -- three magnitudes described one session
298            # (raid review 2026-07-18)
299            if self._credited:
300                done = t("raid_msg_credited_done", "Raid: the gate credited {dmg:,} damage.").format(dmg=self._credited)
301            elif self._dealt:
302                done = t("raid_msg_landed_done", "Raid: {dmg} raw landed — no gate credit yet.").format(dmg=self._dealt)
303            else:
304                done = None
305            return ("done", done)
306        return None

309    def _context_line(self, v: Any, b: Any) -> Any:
310        """The raid page's one context line: the status message alternates
311        with the waiting purse / weekly cadence on the shop-tease beat (40
312        ticks), so neither starves the other (menu.note marquees any of
313        them when over-wide)."""
314        award = v.get("award")
315        if award:
316            alt = t("raid_msg_purse", "purse waiting: {name} — press C").format(name=award.get('boss', '?')[:12])
317        elif not self._standing():
318            alt = ""
319        else:
320            import time as _time
321            import tuipet.core.tournament as _cad
322            left = max(0, int(b.get("end", 0) - v.get("now", 0)))
323            days, hrs = left // 86400, left % 86400 // 3600
324            fest = _cad.holiday()
325            # the relay pays x1.5 on WEEKENDS only -- "2x" and a festival
326            # bonus were promises the server never paid (raid review
327            # 2026-07-18); the festival stays as pure calendar flavor.
328            # The weekend note keys off the SERVER'S clock (its own `now`,
329            # UTC weekday) -- borrowing the cups' LOCAL is_weekend() lied
330            # both ways at week edges: a US Friday evening is already a
331            # paying UTC Saturday, a US Sunday evening a 1x UTC Monday
332            # (cup audit 2026-07-19).  Cups stay local: their purse pays
333            # client-side, so the player's own weekend IS their truth.
334            srv_wknd = _time.gmtime(v.get("now", _time.time())).tm_wday >= 5
335            alt = (t("raid_msg_weekly", "weekly boss · {d}d {h}h left").format(d=days, h=hrs)
336                   + (f" · {fest}" if fest
337                      else t("raid_msg_weekend", " · weekend claims pay 1.5x") if srv_wknd else ""))
338        if self.msg and (not alt or (self.frame_i // 40) % 2 == 0):
339            return self.msg
340        return alt or self.msg

342    def text(self) -> Any:
343        if self.sub is not None:
344            return self.sub.text()
345        v = self.view
346        b = self._boss()

348            out = menu.header(t("raid_hdr_raid", "RAID"), "…")
349            out.append_text(menu.blanks(5))
350            out.append_text(menu.note(self.msg, tick=self.frame_i))
351            # keys ride the STRIP here like every other state of this
352            # screen -- the in-LCD footer was the family's one stray
353            # (raid round 2026-07-19)
354            return out
355        num = int(b.get("num", -1))
356        # THE LCD IS PURE SCENE (raid uncramp 2026-07-23, Joel: "its still
357        # a cramped up mess, it looks like hes in the sky").  The old page
358        # stacked POOL bar + stats + cadence UNDER an 8-row scene -- every
359        # one of those numbers already lives on the STATUS card, and the
360        # boss paid for the duplication: a 16px sprite crammed edge-to-edge
361        # in a 16px band, head touching the header, feet flush on text.
362        # Now: header + a TALL 10-row stage + one context line.  The boss
363        # gets 4px of sky and the standard 2px floor margin, and the
364        # backdrop's BOTTOM 20px keeps the arena floor under its feet
365        # (_paint_cells indexes bgimg by absolute row -- an unsliced 24px
366        # backdrop painted this reduced scene with its sky band).
367        rows = data.bob_frame(num, self.frame_i,
368                              "attack" if (self.frame_i // 9) % 4 == 3 else "idle")
369        sc_rows = ROWS - 2
370        boss = grid.prep(rows, ph=sc_rows * 2) if rows else None
371        placements = [(boss, (COLS - grid.width(boss)) // 2, False)] if boss else []
372        bgimg = self.pet.background(file="tourneyBack")
373        if bgimg and len(bgimg) > sc_rows * 2:
374            bgimg = bgimg[-sc_rows * 2:]
375        scene = render_scene(placements, COLS, sc_rows, menu.scene_ink(bgimg),
376                              LCD_BG, bgimg=bgimg)
377        stage = data.record_for(num).get("stage", "Mega")
378        out = menu.bar(t("raid_hdr_raid_name", "RAID · {name}").format(name=b.get('name', '?')[:14]), stage[:8])
379        out.append_text(scene)
380        out.append("\n")               # terminate the scene's last row
381        if not self._standing():
382            left = max(0, int(b.get("start", 0) - v.get("now", 0)))
383            note = t("raid_msg_inc_boss", "INCOMING BOSS — {h}h {m}m").format(h=left // 3600, m=left % 3600 // 60)
384        else:
385            # priority msg > waiting purse > the weekly cadence; every
386            # number (pool, standing, tries, top) lives on the CARD
387            note = self._context_line(v, b)
388        out.append_text(menu.note(note, tick=self.frame_i))
389        out.right_crop(1)          # 12 rows exactly, no trailing 13th
390        return out
