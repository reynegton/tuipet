tuipet.ui.screens.datacorescreen

datacore — DVPet's EvolutionState page + tuipet's paged data book.

The CORE page is the canon Datacore (SpriteAnim setupDatacore): the field- flavoured core backdrop (ViewUtil.getDatacoreBackground picks it by the pet's HIGHEST DNA field, falling back to its own), the core badge (a special core from datacoreMenuConfig.csv, else the X-antibody state badge), and the core NUMBER — with a pending evolution it counts DOWN from DatacoreBaseRate 14 as the growth period elapses; past growth (or a final form) it counts UP through the days toward the elder line (age-based since the DSprite mortality port 2026-07-22). Pressing the core plays the silhouette teaser (EvolSilhouetteTransition): the next natural evolution as a blacked-out shape. The data-book pages after it are tuipet's own readout (kept adaptation).

 78                self.teaser = False           # EvolSilhouetteBack: dark blink out

 83            # onDatacore -> EvolSilhouetteTransition (the fade sound has no rip;
 84            # the dna-wash sweep substitutes, like heal's click/confirm cues)
 85            self.teaser = True
 86            self.teaser_t = 0                 # the badge zooms in first (datacoreExpand)
 87            self.sfx = "wash"
 88            return None
 89        if k == "m" and self.i == 0:
 90            # the tModeChange button on the EvolutionState page
 91            if not self.pet.can_mode_change():
 92                return None
 93            old_num, msg = self.pet.mode_change()
 94            if old_num is not None:
 95                return ("done", ("evolve", old_num, msg))
 96            self.note = msg
 97            self.sfx = "error"
 98            return None
 99        if self.detail is not None:                    # the requirement checklist
100            if k in ("up", "k"):
101                self.det_off = max(0, self.det_off - 1)
102            elif k in ("down", "j"):
103                self.det_off += 1                      # text() clamps to the list
104            elif k == "pageup":                        # page jumps, lobby-chat style
105                self.det_off = max(0, self.det_off - (DET_VIS - 1))
106            elif k == "pagedown":
107                self.det_off += DET_VIS - 1            # text() clamps to the list
108            elif k in ("escape", "enter", "space", "d"):
109                self.detail = None
110                self.det_off = 0

112        on_evo = self.pages[self.i][0] == "EVOLVES"
113        rows = self.pages[self.i][1] if on_evo else None

116                self.evo_sel = (self.evo_sel - 1) % len(rows)

168    def _core_scene(self) -> Any:
169        """The CORE landing page -- a data-menu page like every other datacore
170        page (Joel 2026-07-05: one layout language for the whole data book;
171        the scene experiment read as inconsistent).  SPACE opens the core
172        gaze, where the visuals live."""
173        p = self.pet
174        n = core_number(p)
175        growth = p.STAGE_DURATION.get(p.stage)
176        # ONE predicate for the whole page (datacore audit 2026-07-18):
177        # the Meter row and core_number must never disagree again
178        pending = (growth is not None and p.stage_seconds < growth
179                   and core.has_next(p))
180        x = getattr(p, "x_antibody", "None")
181        # the number wears its own meaning (gameplay polish #16, 2026-07-22):
182        # a bare "◆ 7" was a countdown-to-evolve on a growing pet and an
183        # age-count on a final form -- same glyph, opposite directions,
184        # with only the Meter row to disambiguate
185        if pending:
186            core_val = t("datacore_to_evolve", "{chr} {n} to evolve").format(chr=chr(0x25C6), n=n)
187        elif p.is_geriatric:
188            core_val = t("datacore_elder", "{chr} {n} — elder").format(chr=chr(0x25C6), n=n)
189        else:
190            core_val = t("datacore_to_elder_full", "{chr} {n} of {base} to elder").format(chr=chr(0x25C6), n=n, base=DATACORE_BASE_RATE)
191        rows = [
192            (t("datacore_lbl_core", "Core"), core_val),
193            (t("datacore_lbl_meter", "Meter"), t("datacore_meter_near", "evolution nears at 1") if pending else t("datacore_meter_days", "counts the days")),
194            (t("datacore_lbl_field", "Field"), data.pretty_field(getattr(p, "field", "") or "None")),
195            (t("datacore_lbl_xstate", "X-State"), t("datacore_val_none_lower", "none") if x == "None" else x.lower()),
196            (t("datacore_lbl_mode", "Mode"), (t("datacore_mode_ready", "ready — press M") if p.can_mode_change() else chr(0x2014))),
197        ]
198        out = menu.header(t("datacore_hdr_core", "DATACORE  CORE"), self._dots())
199        for label, val in rows:
200            out.append(f" {label:<9}", style=DIM)
201            out.append(f"{val}\n", style=INK_B)
202        out.append_text(menu.blanks(9 - len(rows) - 3))
203        out.append(t("datacore_hint_gaze_1", " gaze into the core to glimpse\n"), style=DIM)
204        out.append(t("datacore_hint_gaze_2", " what stirs within…\n"), style=DIM)
205        # the door wears its key in BOLD (menu polish 2026-07-21: the gaze
206        # was easy to look over in dim prose) -- the EVOLVES "ENTER: what it
207        # takes" / TROPHIES "ENTER: the album" teaching line, gaze verdicts
208        # still take the slot when one is pending
209        out.append_text(menu.note(self.note if self.note != t("datacore_stirs", "the core stirs…")
210                                  else t("datacore_hint_space_gaze", "SPACE: gaze into the core"),
211                                  tick=self.frame_i))
212        out.append_text(menu.footer(t("datacore_hint_footer", "SPACE gaze  ←→ page  ESC out")))
213        return out

215    def _teaser_scene(self) -> Any:
216        """EvolSilhouetteTransition: the core badge ZOOMS IN (datacoreExpand,
217        canon beats 6-14 grow it 1.5x each), then the next natural evolution
218        holds as a STATIC blacked-out shape (canon draws frame 0 -- the old
219        pose-flicker here was the 10Hz flutter class)."""
220        """The gaze (Joel 2026-07-05 round 3): the WHOLE LCD, nothing else --
221        background, the current mon, the circle animation, the silhouette.
222        Boom boom boom.  The narration rides the message box (it IS a
223        message); no key hints anywhere."""
224        p = self.pet
225        bgimg = core_background(p)
226        on = menu.scene_ink(bgimg)
227        t = self.teaser_t
228        if t < MON_T:                                     # beat one: the mon itself
229            rows = self._pet_rows(p.num)
230            placements = [self._core_place(rows)] if rows else []
231            return render_scene(placements, 40, SCENE_ROWS, on, LCD_BG, bgimg=bgimg, clip=grid.WINDOW)
232        if t < MON_T + EXPAND_T:                          # beat two: the core turns
233            badge = data.load_effects().get(core_badge_key(p) or "core_xnone", [None])[0]
234            overlay = []
235            if badge:
236                k = 1 + (t - MON_T) // (EXPAND_T // 2)    # 1x -> 2x zoom
237                k = min(k, 2)
238                big = ["".join(ch * k for ch in r) for r in badge for _ in range(k)]
239                bw, bh = max(len(r) for r in big), len(big)
240                ox, oy = (40 - bw) // 2, (SCENE_ROWS * 2 - bh) // 2
241                overlay = [(ox + x, oy + y) for y, row in enumerate(big)
242                           for x, c in enumerate(row) if c == "1" and oy + y >= 0]
243            return render_scene([], 40, SCENE_ROWS, on, LCD_BG,
244                                overlay=overlay, bgimg=bgimg)
245        nxt = next_evolution(p)                           # beat three: what stirs
246        if nxt is None:
247            rows = self._pet_rows(p.num, idx=0)
248            placements = [self._core_place(rows)] if rows else []
249            return render_scene(placements, 40, SCENE_ROWS, on, LCD_BG, bgimg=bgimg, clip=grid.WINDOW)
250        sil = ghost(silhouette(self._pet_rows(nxt, idx=0) or []),
251                    phase=(self.frame_i // 5) % 2)
252        placements = [self._core_place(sil)] if sil else []
253        return render_scene(placements, 40, SCENE_ROWS, on, LCD_BG, bgimg=bgimg, clip=grid.WINDOW)

46def core_number(pet: Any) -> int:
47    """setupDatacore's meter: an evolution countdown while a normal evolution is
48    pending, otherwise an AGE meter counting up toward the elder line (the
49    lifespan denominator left with the clock -- DSprite mortality 2026-07-22:
50    age is what kills now, so age is what the core counts).  Floors at 1."""
51    base = DATACORE_BASE_RATE
52    growth = pet.STAGE_DURATION.get(pet.stage)
53    # Mega's 9e9 is the "never auto-evolves" sentinel, not a growth clock:
54    # counting it as pending froze the meter at its top for any Mega with
55    # onward (jogress/X) rows -- the count-up this docstring promises a
56    # final form never appeared (gameplay audit 2026-07-19)
57    pending = (growth is not None and growth < 9e8
58               and pet.stage_seconds < growth and has_next(pet))
59    if pending:
60        denom = round(growth / base) or 1
61        n = int(base - pet.stage_seconds / denom)
62    else:
63        from tuipet.core.petbase import AGE_DAY, GERIATRIC_AGE_DAYS
64        denom = round(GERIATRIC_AGE_DAYS * AGE_DAY / base) or 1
65        n = int(pet.age_seconds / denom)
66    return max(1, n)
