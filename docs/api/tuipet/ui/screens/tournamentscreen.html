243    def _render_tree(self) -> Any:
244        """The bracket page: the field of eight and the tree filling in round
245        by round (the entrants always existed in the engine; the player just
246        never SAW the tournament)."""
247        tourn = self.tourney
248        tree = tourn.tree  # type: ignore
249
250        def nm(e: Any, w: Any) -> Any:
251            if e == "YOU":
252                s = self.pet.name or "YOU"
253            else:                              # the rival wears its grudge mark
254                s = ("!" if e.get("rival") else "") + e["name"]
255            return s[:w]
256
257        out = menu.bar(tourn.name, t("cup_bracket_title", "BRACKET"))  # type: ignore
258        champ = tree[3][0] if len(tree) > 3 else None
259        for i in range(8):
260            c1 = nm(tree[0][i], 10)
261            c2 = c3 = ""
262            if i % 2 == 0 and len(tree) > 1:
263                c2 = nm(tree[1][i // 2], 10)
264            if i in (0, 4) and len(tree) > 2:
265                c3 = nm(tree[2][i // 4], 10)
266                if champ is not None and tree[2][i // 4] == champ:
267                    c3 += " ★"
268            you = (tree[0][i] == "YOU" or c2 and tree[1][i // 2] == "YOU"
269                   or c3 and tree[2][i // 4] == "YOU")
270            style = INK_B if you else INK
271            out.append(" %-11s%-11s%s\n" % (c1, c2, c3),
272                       style=style if you else (INK if c1 else DIM))
273        out.append_text(menu.note(tourn.last, tick=self.frame_i))  # type: ignore
274        if tourn.over:  # type: ignore
275            out.append_text(menu.footer(t("cup_hint_result", "SPACE result   ESC leave")))
276        else:
277            # two-space gap: "quarterfinal" runs the line to exactly 38 --
278            # three spaces clipped "ESC forfeit" to "ESC forfei" (menu audit
279            # 2026-07-21; menu.footer hard-cuts at W)
280            out.append_text(menu.footer(t("cup_hint_next", "SPACE to the {round}  ESC forfeit").format(round=tourn.round_name.lower())))  # type: ignore
281        return out

307    def _advance_frame(self) -> Any:
308        """THE FIELD ADVANCES: the other winners cross the arena one at a
309        time (the parade idiom on the cup stage) before the bracket page
310        lands -- the tournament happening AROUND you, visible at last."""
311        a = self._advance
312        i = min(a["t"] // NPC_T, len(a["nums"]) - 1)  # type: ignore
313        frame_t = a["t"] % NPC_T  # type: ignore
314        fr = data.frames_for(a["nums"][i])  # type: ignore
315        wi = data.ROLES["walk"][(frame_t // 3) % 2]
316        rows = grid.prep((fr[wi] if wi < len(fr) else None) or fr[0],
317                         ph=FIGHT_ROWS * 2)
318        # walk the winner clean ACROSS: in from off-screen right (X1) and out
319        # past off-screen left (X0-w), the window clipping the partial sprite
320        # at each edge -- the old roam_bounds sweep (X1-w -> X0) popped it in at
321        # the right interior edge and vanished it at the left, never walking
322        # off (parade fix 2026-07-24, Joel "just appears to the right, and
323        # disappears when it hits the left").  Mirrors the intro's own walk-in.
324        w = grid.width(rows)
325        x = round(grid.X1 + (grid.X0 - w - grid.X1) * (frame_t / max(1, NPC_T - 1)))

327        scene = render_scene([(rows, x, False)], COLS, FIGHT_ROWS,
328                             menu.scene_ink(bgimg), LCD_BG, bgimg=bgimg,
329                             clip=grid.WINDOW)

331              if i < len(self.tourney.results) else "")  # type: ignore
332        # pure scene; who advanced rides the STRIP (cup audit 2026-07-25)
333        self._say = (t("cup_advances_name", "{name} advances").replace("{name}", nm)) if nm else t("cup_advances_field", "the field advances")

336    def _intro_frame(self) -> Any:
337        """MATCH INTRODUCTIONS: the challenger strides in from the RIGHT and
338        is announced, your mon answers from the LEFT, both hold the
339        stare-down -- then the bell (anim opens the fight).  The corners are
340        grid.faceoff's own, so the entrance lands exactly where the fight
341        stands."""
342        frame_t = self._intro["t"]  # type: ignore
343        opp = self.tourney.current_opponent()  # type: ignore
344        pet_rows = self._frames(self.pet.num, "walk" if frame_t >= INTRO_OPP_T else "idle")
345        opp_rows = self._frames(opp["num"], "walk" if frame_t < INTRO_OPP_T else "idle")
346        left, right = grid.faceoff(pet_rows, opp_rows, left_mirror=True,
347                                   right_mirror=False, ph=FIGHT_ROWS * 2)
348        lrows, lx, lm = left
349        rrows, rx, rm = right
350        placements = []
351        if frame_t < INTRO_OPP_T:                    # the challenger walks in
352            p = frame_t / max(1, INTRO_OPP_T - 1)
353            placements = [(rrows, round(grid.X1 + (rx - grid.X1) * p), rm)]
354            note = (t("cup_rival_enters", "{name} — your RIVAL!").replace("{name}", opp["name"]) if opp.get("rival") else
355                    t("cup_foe_enters", "{name} [{attr}] enters!").replace("{name}", opp["name"]).replace("{attr}", opp["attribute"][:2]))
356        elif frame_t < INTRO_OPP_T + INTRO_PET_T:    # your mon answers
357            p = (frame_t - INTRO_OPP_T) / max(1, INTRO_PET_T - 1)
358            lw = grid.width(lrows)
359            placements = [(lrows, round((grid.X0 - lw) + (lx - (grid.X0 - lw)) * p), lm),
360                          (rrows, rx, rm)]
361            note = (t("cup_pet_answers", "{name} answers!").replace("{name}", self.pet.name) if self.pet.name
362                    else t("cup_you_answer", "You answer!"))           # unnamed: "YOU answers!" was bad grammar
363        else:                                  # the held stare-down
364            placements = [(lrows, lx, lm), (rrows, rx, rm)]
365            note = t("cup_round_fight", "{round} — FIGHT!").replace("{round}", self.tourney.round_name)  # type: ignore
366        bgimg = self.pet.background(file="tourneyBack")
367        scene = render_scene(placements, COLS, FIGHT_ROWS,
368                             menu.scene_ink(bgimg), LCD_BG, bgimg=bgimg,
369                             clip=grid.WINDOW)
370        # pure scene; the announcement rides the STRIP (cup audit
371        # 2026-07-25), where it is actually visible -- and "vs <foe>" +
372        # "your ★N" ride the card, which already carries the trophy count
373        self._say = note
374        return scene
