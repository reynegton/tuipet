 96    def key(self, k: Any) -> Any:
 97        row = self._ROWS[self.cursor]
 98        if k in ("up", "k"):
 99            self.cursor = (self.cursor - 1) % len(self._ROWS)
100            self.msg = ""
101        elif k in ("down", "j"):
102            self.cursor = (self.cursor + 1) % len(self._ROWS)
103            self.msg = ""

105            if not sound.available():
106                self.msg = t("opt_snd_bell_no_vol", "the terminal bell has no volume")

111            if self.sound_get():
112                self.sfx = "confirm"   # type: ignore
113        elif k in ("enter", "space"):
114            if row == "sound":
115                self.sound_toggle()
116                self.msg = t("opt_snd_res", "sound: {val}").format(val=_sound_value(self.sound_get()))
117                self.sfx = "confirm" if self.sound_get() else None  # type: ignore
118            elif not sound.available():
119                self.msg = t("opt_snd_bell_no_vol", "the terminal bell has no volume")
120            elif not self.sound_get():
121                self.msg = t("opt_snd_off_msg", "sound is off — nothing to hear")
122            else:
123                self.msg = t("opt_snd_vol", "volume: {v}%").format(v=sound.volume())
124                self.sfx = "confirm"  # type: ignore
125        elif k in ("escape", "g"):
126            return ("done", None)
127        return None

149class KeysPanel:
150    """Every home-screen binding on one scrollable page (no cursor — there is
151    nothing to pick, ↑↓ just slide the window)."""
152
153    VISIBLE = 8
154
155    def anim(self) -> None:
156        pass          # frame heartbeat: over-wide notes marquee (sweep 2026-07-15)
157
158    def __init__(self, bindings: Any) -> None:
159        # Textual binds a couple of keys by identifier, not glyph -- show the
160        # glyph so the page reads "?  Help" / "Enter  Accept gift" instead of
161        # leaking "question_mark" (which also overran the 6-col key column).
162        keyname = {"question_mark": "?", "enter": "ENTER",
163                   "enter,space": "ENTER"}   # space is a silent gift alias
164        self.rows = [f"{keyname.get(k, k):<6} {label}"
165                     for k, _action, label in bindings]
166        self.top = 0
167
168    def strip(self) -> Any:
169        return menu.hints(("↑↓", t("opt_key_hint_scroll", "scroll")), ("ESC", t("opt_snd_hint_back", "back")))
170
171    def key(self, k: Any) -> Any:
172        last = max(0, len(self.rows) - self.VISIBLE)
173        if k in ("up", "k"):
174            self.top = max(0, self.top - 1)
175        elif k in ("down", "j"):
176            self.top = min(last, self.top + 1)
177        elif k == "pageup":                  # page jumps, lobby-chat style
178            self.top = max(0, self.top - (self.VISIBLE - 1))
179        elif k == "pagedown":
180            self.top = min(last, self.top + (self.VISIBLE - 1))
181        elif k in ("escape", "enter", "space", "g"):
182            return ("done", None)
183        return None
184
185    def text(self) -> Any:
186        n = len(self.rows)
187        lo, hi = self.top + 1, min(self.top + self.VISIBLE, n)
188        out = menu.header(t("opt_key_hdr", "KEYS"), f"{lo}-{hi}/{n}")
189        shown = self.rows[self.top:self.top + self.VISIBLE]
190        for r in shown:
191            out.append_text(menu.row(r))
192        out.append_text(menu.blanks(self.VISIBLE - len(shown) + 1))
193        out.append_text(menu.footer(t("opt_key_footer", "↑↓ scroll  ESC back")))
194        return out

311    def _install_update(self) -> None:
312        """Install the newer release, off the UI thread (pip takes seconds).
313
314        The running process keeps executing the OLD code -- Python imported it
315        at launch -- so a success always ends in "restart tuipet".  We never
316        pretend the swap happened live.
317        """
318        if self._installing:
319            return
320        self._installing = True
321        self.msg = t("opt_msg_updating", "updating… (this takes a moment)")
322
323        def run() -> None:
324            ok, msg = update_check.run_upgrade()
325            self.msg = msg
326            self._installing = False
327            if ok:
328                self._upd = ""                # nothing left to offer
329                self._updated = True
330                # the restart OFFER (Joel 2026-07-18: "make it so the update
331                # option asks to restart after update"): ENTER relaunches
332                self.confirm_restart = True
333                self.msg = t("opt_msg_updated_restart", "Updated! Restart now?  ENTER restarts · ESC later")
334            # the completion ALSO rides the app's verdict channel (options
335            # audit 2026-07-19, swallow class #4): pip takes seconds -- if
336            # the player closed options meanwhile, the offer above lands on
337            # a dead panel and they learn nothing.  The parked verdict
338            # flashes wherever they are next home; a redundant note beside
339            # the live offer is honest, a swallowed one is not.
340            self.verdict(t("opt_msg_verdict_ok", "tuipet updated — restart to play the new version.")
341                         if ok else t("opt_msg_verdict_err", "update: {msg}").format(msg=msg))
342        threading.Thread(target=run, daemon=True).start()
