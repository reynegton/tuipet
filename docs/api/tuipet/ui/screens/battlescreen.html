tuipet.ui.screens.battlescreen

Battle — the alternating-view volley show over the precomputed HP race (the 0.5 battlescreen, ported 2026-07-17; DSprite is the ultimate truth for animations and mechanics).

The fight itself is decided by battle.generate() (care, training, stage and the attribute triangle feed each side's hit chance; the trained hit-type sets damage). This screen plays it back one round at a time: whoever fires is shown alone, the orb flies off-screen, the defender dodges or eats the blast. Before the fight a TIMING BAR sets your hit-type for the bout — good condition widens the mega window (care widens skill).

 33    renderer strobes it at 100ms).  The skull-and-crossbones that strobed

 78def mega_window(pet: Any) -> Any:
 79    """Care widens the skill ceiling: condition score -> the mega zone
 80    [12-c, 12+c] on the 0..24 bar (width 1/3/5/7); ±5 around it = normal.
 81    (Classic gauges: flat 4-heart meters, age off the world clock.)"""
 82    from tuipet.core.pet import DAY_LENGTH
 83    wr = pet.wins / pet.battles if pet.battles > 0 else 0
 84    hu = pet.hunger / 4.0
 85    st = pet.strength / 4.0
 86    en = max(0, pet.energy) / pet.max_energy if pet.max_energy else 0
 87    ag = min((pet.age_seconds / DAY_LENGTH) / 5, 1)
 88    o = min(1.0, wr * 0.2 + hu * 0.2 + st * 0.2 + ag * 0.2 + en * 0.2)
 89    # FLOORED at 3px (timing rework 2026-07-23): the marker steps every
 90    # 100ms, so the old 1px floor was a 100ms target -- physically
 91    # impossible in a terminal.  3px = 300ms, the honest minimum; care
 92    # still pays (3/5/7).
 93    w = max(3, 1 + int(o * 3) * 2)
 94    c = w // 2
 95    return 12 - c, 12 + c

 98def round_timeline(ph0: Any, fh0: Any, pdmg: Any, edmg: Any, player_first: Any, effect: Optional[Any]=None,
 99                   hold_foe_bar: bool=False) -> Any:
100    """One round's alternating-view volley timeline, from PURE round data --
101    shared by the PvE panel (which reads it off its Battle) and the lobby's
102    PvP replay (which reads it off the relayed result; lobby audit 2026-07-04:
103    PvP rounds were a text log while PvE plays the full animation)."""
104    # strike order: initiative first.  A KO'd side's strike is hidden when
105    # it MISSED; a LANDED one always animates (audit 2026-07-19: hiding it
106    # showed HP dropping with no animation).  Since death-is-final
107    # (2026-07-22) the LOCAL engine never records posthumous landed hits,
108    # but this timeline also replays RELAYED lobby rounds -- it stays
109    # faithful to whatever the record says was applied.
110    if player_first:
111        seq = [("pet", "foe", pdmg)]
112        if fh0 - max(0, pdmg) > 0 or edmg > 0:
113            seq.append(("foe", "pet", edmg))
114    else:
115        seq = [("foe", "pet", edmg)]
116        if ph0 - max(0, edmg) > 0 or pdmg > 0:
117            seq.append(("pet", "foe", pdmg))
118    tl = []
119    ph, fh = ph0, fh0
120    tl += [{"m": "faceoff", "view": seq[0][0], "ph": ph, "fh": fh}] * FACEOFF_T
121    for atk, dfn, dmg in seq:
122        other = edmg if atk == "pet" else pdmg
123        dbl = dmg >= 2 and dmg >= other                  # the classic V-pet doubleAttack: strong & out/matching power
124        fxn = effect if atk == "pet" else None           # only the player carries chip effects (PvE)
125        for s in range(WINDUP_T):
126            tl.append({"m": "windup", "view": atk, "atk": atk, "wu": s, "ph": ph, "fh": fh})
127        for s in range(FIRE_T):                          # attacker shown: orb leaves off-screen
128            tl.append({"m": "fire_out", "view": atk, "atk": atk, "double": dbl, "fx": fxn,
129                       "prog": (s + 1) / FIRE_T, "ph": ph, "fh": fh})
130        for s in range(FIRE_T):                          # defender shown: orb arrives off-screen
131            tl.append({"m": "fire_in", "view": dfn, "atk": atk, "def": dfn, "double": dbl,
132                       "prog": (s + 1) / FIRE_T, "ph": ph, "fh": fh})
133        if dmg > 0:                                      # HIT: fullscreen flash, then flinch
134            if dfn == "foe":
135                if not hold_foe_bar:      # a raid boss's bar NEVER falls: the old
136                    #  in-round dip snapped back to full at the next round's frames
137                    fh = max(0, fh - dmg)
138            else:
139                ph = max(0, ph - dmg)
140            # device-exact (GML 2026-07-14): the hit STING is skipped on the
141            # final winning blow -- the KO presentation carries the audio
142            final = dfn == "foe" and fh == 0
143            for s in range(EXPLODE_FRAMES):
144                tl.append({"m": "hit", "f": (s // EXPLODE_HOLD) % 2, "def": dfn,
145                           "double": dbl, "final": final, "ph": ph, "fh": fh})
146            tl += [{"m": "flinch", "view": dfn, "def": dfn, "ph": ph, "fh": fh}] * FLINCH_T
147        else:                                            # DODGE: defender weaves, orb whiffs past
148            for s in range(DODGE_T):
149                tl.append({"m": "dodge", "view": dfn, "atk": atk, "def": dfn,
150                           "prog": (s + 1) / DODGE_T, "ph": ph, "fh": fh})
151    return tl

159def boss_death_timeline(ph: Any) -> Any:
160    """SpriteAnim.zoneBossDeath: a beaten ZONE BOSS doesn't just explode -- it
161    blinks out (three lights-flicker cycles over a shaken hurt pose, bossDying
162    stings) and then SQUASHES into the ground in three steps (canon sizeY
163    48->24->12->0 with the feet planted, bossDeath stings)."""
164    tl = []
165    for c in range(BOSSDIE_FLICKERS):
166        for s in range(BOSSDIE_ON):
167            tl.append({"m": "bossdie", "stage": "on", "jit": (c + s) % 2, "ph": ph, "fh": 0})
168        tl += [{"m": "bossdie", "stage": "off", "ph": ph, "fh": 0}] * BOSSDIE_OFF
169    for keep in (8, 4, 2):
170        tl += [{"m": "bossdie", "stage": "squash", "keep": keep, "ph": ph, "fh": 0}] * BOSSDIE_STEP_T
171    tl += [{"m": "bossdie", "stage": "off", "ph": ph, "fh": 0}] * 4
172    return tl

175def _squash_rows(rows: Any, keep: Any) -> Any:
176    """Vertical squash with the feet planted: sample `keep` rows across the
177    sprite's height and pad the removed height with blank rows on top."""
178    h = len(rows)
179    if keep >= h:
180        return rows
181    idx = [round(i * (h - 1) / (keep - 1)) for i in range(keep)] if keep > 1 else [h - 1]
182    w = max(len(r) for r in rows)
183    blank = "0" * w if isinstance(rows[0], str) else [None] * w
184    return [blank] * (h - keep) + [rows[i] for i in idx]

187def _full(frame: Any) -> Any:
188    # window-law: the 32x16 banner/flash fills the PLAY WINDOW exactly (like
189    # training's explosion), not the whole LCD -- LCD-centring put its top two
190    # rows in the bezel sky at y4-5 (audit 2026-07-13)
191    w = len(frame[0]) if frame and frame[0] else 0
192    ox = grid.X0 + max(0, (grid.W - w) // 2)
193    oy = grid.TOP + max(0, (grid.BAND - len(frame)) // 2)
194    return [(ox + x, oy + y) for y, row in enumerate(frame)
195            for x, c in enumerate(row) if c == "1"]

308        """A one-shot beep at timeline marker edges."""
309        entry = self.timeline[self.i]
310        m = entry.get("m")
311        if m == "bossdie":
312            prev = self.timeline[self.i - 1] if self.i else {}

314            if st == "on" and prev.get("stage") != "on":

316            elif st == "squash" and prev.get("keep") != entry.get("keep"):
317                self.sfx = "attackHit"
318        elif m != self._last_m:
319            s = (None if m == "hit" and entry.get("final")
320                 else strikefx.beat_sfx(m, entry.get("double")))
321            if s:

323            elif m == "reveal":

327    def anim(self) -> None:
328        self.frame_i += 1
329        if self.phase == "ready":
330            self._bar_hist = (self._bar_hist + [self.bar])[-strikefx.LOCK_GRACE:]
331            self.bar += self.bar_dir
332            if self.bar >= BAR_MAX or self.bar <= 0:
333                self.bar_dir = -self.bar_dir
334                self.bar = max(0, min(BAR_MAX, self.bar))
335            return
336        if self.phase == "result":
337            return
338        if self.i < len(self.timeline) - 1:
339            self.i += 1
340            self._emit_sfx()
341        elif self.phase == "intro":
342            self.phase = "ready"
343            self._ready_frame = self.frame_i
344        else:
345            if self.battle is None or self.battle.over:
346                self._enter_result()
347            else:
348                self._next_round()

350    def strip(self) -> Any:
351        """The message-box hint line."""
352        if self.phase == "ready":
353            return menu.hints(("SPACE", "lock the bar"),
354                              ("ESC", "flee" if self.wild else "back out"))
355        if self.phase == "intro":

363    def _lock_bar(self) -> None:
364        # ONE grading source with the drill (strikefx.grade_lock: the
365        # latency grace, the 2px marker, the veteran rule)
366        t = strikefx.grade_lock(self._bar_hist + [self.bar],
367                                self.mega_lo, self.mega_hi,
368                                veteran=self.pet.battles >= 999)
369        self.sfx = "confirm" if t != "miss" else "refuse"
370        self._start_fight(t)

372    def key(self, k: Any) -> Any:
373        if self.phase == "intro":
374            if k in ("space", "enter", "escape"):
375                self.i = len(self.timeline) - 1
376                self.phase = "ready"
377                self._ready_frame = self.frame_i
378            return None
379        if self.phase == "ready":
380            if k in ("space", "enter"):
381                if self.frame_i - self._ready_frame < LOCK_ARM_T:
382                    return None            # the intro mash bleeding in -- not a timed lock
383                self._lock_bar()
384            elif k == "escape":
385                self.ran_away = True
386                return ("done", None)          # walked away before the bell
387            return None
388        if self.phase == "anim":
389            # the debounce anchors on the LOCK frame, not the round frame:
390            # gating on self.i (reset every round) ate the first 0.6s of
391            # skip presses at EVERY round boundary (QOL sweep 2026-07-23)
392            if self.frame_i - self._lock_frame < SKIP_DEBOUNCE:
393                return None
394            if k == "escape" and self.battle and not self.battle.over:
395                # end it: fast-run the precomputed rounds to the DECIDING one
396                # and let that single round play out -- the KO (and a boss's
397                # death beat) still shows, but a lopsided 20-round bout no
398                # longer needs ~20 timed presses (QOL sweep 2026-07-23)
399                while self.phase == "anim" and not self.battle.over:
400                    self._next_round()
401                return None
402            if k in ("space", "enter", "escape"):
403                last = next((j for j in range(len(self.timeline) - 1, -1, -1)
404                             if self.timeline[j]["m"] in ("hit", "dodge")), None)
405                if last is None or self.i >= last - (EXPLODE_FRAMES + 2):
406                    self.i = len(self.timeline) - 1   # already at the impact
407                else:
408                    first = last
409                    while first > 0 and self.timeline[first - 1]["m"] == self.timeline[last]["m"]:
410                        first -= 1
411                    # ⚠ NEVER BACKWARD (battle audit 2026-07-25).  `first` is
412                    # the START of the closing impact run, so a press made
413                    # after the playhead had already entered that run SNAPPED
414                    # IT BACK -- and a mashed key (a phone tap-tap-tap) reset
415                    # it every frame, so the round replayed forever and the
416                    # fight sat frozen: 4000 frames of pressing every frame
417                    # never left round 1, where sparse presses finish in ~80.
418                    # A hurry key may only ever hurry.
419                    self.i = max(self.i, first)
420            return None
421        if k in ("space", "enter", "escape"):
422            return ("done", self.battle)
423        return None

373    def key(self, k: Any) -> Any:
374        if self.phase == "intro":
375            if k in ("space", "enter", "escape"):
376                self.i = len(self.timeline) - 1
377                self.phase = "ready"
378                self._ready_frame = self.frame_i
379            return None
380        if self.phase == "ready":
381            if k in ("space", "enter"):
382                if self.frame_i - self._ready_frame < LOCK_ARM_T:
383                    return None            # the intro mash bleeding in -- not a timed lock
384                self._lock_bar()
385            elif k == "escape":
386                self.ran_away = True
387                return ("done", None)          # walked away before the bell
388            return None
389        if self.phase == "anim":
390            # the debounce anchors on the LOCK frame, not the round frame:
391            # gating on self.i (reset every round) ate the first 0.6s of
392            # skip presses at EVERY round boundary (QOL sweep 2026-07-23)
393            if self.frame_i - self._lock_frame < SKIP_DEBOUNCE:

395            if k == "escape" and self.battle and not self.battle.over:
396                # end it: fast-run the precomputed rounds to the DECIDING one
397                # and let that single round play out -- the KO (and a boss's
398                # death beat) still shows, but a lopsided 20-round bout no
399                # longer needs ~20 timed presses (QOL sweep 2026-07-23)
400                while self.phase == "anim" and not self.battle.over:
401                    self._next_round()
402                return None
403            if k in ("space", "enter", "escape"):
404                last = next((j for j in range(len(self.timeline) - 1, -1, -1)
405                             if self.timeline[j]["m"] in ("hit", "dodge")), None)
406                if last is None or self.i >= last - (EXPLODE_FRAMES + 2):
407                    self.i = len(self.timeline) - 1   # already at the impact
408                else:
409                    first = last
410                    while first > 0 and self.timeline[first - 1]["m"] == self.timeline[last]["m"]:
411                        first -= 1
412                    # ⚠ NEVER BACKWARD (battle audit 2026-07-25).  `first` is
413                    # the START of the closing impact run, so a press made
414                    # after the playhead had already entered that run SNAPPED
415                    # IT BACK -- and a mashed key (a phone tap-tap-tap) reset
416                    # it every frame, so the round replayed forever and the
417                    # fight sat frozen: 4000 frames of pressing every frame
418                    # never left round 1, where sparse presses finish in ~80.
419                    # A hurry key may only ever hurry.
420                    self.i = max(self.i, first)
421            return None
422        if k in ("space", "enter", "escape"):
423            return ("done", self.battle)
424        return None
