 10def display_len(s: str) -> int:
 11    """Calculates the physical rendering width of a string in the terminal.
 12    Uses wcwidth to account for double-width characters (e.g. CJK)."""
 13    w = wcswidth(s)
 14    return len(s) if w < 0 else w
 15
 16def slice_by_width(s: str, max_w: int) -> str:
 17    """Slices a string such that its rendered width does not exceed max_w."""
 18    w = 0
 19    for i, c in enumerate(s):
 20        cw = wcswidth(c)
 21        w += 1 if cw < 0 else cw
 22        if w > max_w:
 23            return s[:i]
 24    return s
 25
 26def ljust_width(s: str, max_w: int, fillchar: str = " ") -> str:
 27    """Left-justifies a string based on its physical rendering width, padding
 28    with fillchar until it reaches max_w."""
 29    w = display_len(s)
 30    if w >= max_w:
 31        return s
 32    return s + fillchar * (max_w - w)

 37def header(title: Any, right: str="") -> Any:
 38    """Title (left, bold) + optional right-aligned info, then a divider rule."""
 39    title = slice_by_width(title, W)
 40    t = Text()
 41    if right:
 42        gap = max(1, W - display_len(title) - display_len(right))

 75NOTE_HOLD = 16   # marquee ticks held on the head each pass (~1.6s at the 10Hz clock)
 76NOTE_STEP = 2    # advance one character every N ticks (~5 chars/s, the HUD cadence)
 77NOTE_GAP = "      "
 78
 79# the shared marquee clock: app.on_frame advances it at 10 Hz, so EVERY note
 80# scrolls when over-wide -- no screen can silently clip a message again (Joel
 81# 2026-07-15: "Too early to lay out the Futon" lost its tail in the bag).
 82# Panels may still pass their own frame counter; same cadence either way.
 83TICK = 0
 84
 85
 86def _scrolled(msg: str, tick: Any) -> Any:
 87    """The marquee window over an over-wide line (note()'s cadence: hold on
 88    the head, then one character per NOTE_STEP ticks, wrap through NOTE_GAP)."""
 89    if tick is None:
 90        tick = TICK
 91    loop = msg + NOTE_GAP
 92    cycle = len(loop) + NOTE_HOLD                 # hold on the head again each wrap
 93    pos = (tick // NOTE_STEP) % cycle
 94    off = max(0, pos - NOTE_HOLD)
 95    return slice_by_width((loop + loop)[off:], W)

102    marquee now: panels pass their frame counter or inherit the module TICK."""
103    if display_len(msg) <= W:
104        return Text(msg + "\n", style=INK_B)

160def icon_cell(src: Any) -> Any:
161    """Rasterise one sprite bitmap into the IC_W x IC_ROWS icon cell,
162    auto-downsampled to fit both dimensions so it never clips."""
163    blank = [" " * IC_W] * IC_ROWS
164    sh = len(src) if src else 0
165    sw = max((len(r) for r in src), default=0) if src else 0
166    if not sw:
167        return blank
168    from tuipet.utils.render import downsample, bitmap_text
169    from tuipet.utils.theme import LCD_ON, LCD_BG    # read late: theme.apply retints
170    factor = max(1, -(-sw // IC_W), -(-sh // (2 * IC_ROWS)))
171    bm = downsample(src, factor)
172    if not max((len(r) for r in bm), default=0):
173        return blank
174    lines = [t.plain.ljust(IC_W)                # w <= IC_W (factor guarantees it)
175             for t in bitmap_text(bm, LCD_ON, LCD_BG)]
176    # short art anchors to the BASELINE, not the ceiling: foods sit on
177    # plates, props sit on the ground -- a top-floated icon over a dead
178    # bottom row reads as clipped (Joel's report 2026-07-19: "giga meal
179    # sprite might be getting cut off in shop?" -- the 24x18 rip is
180    # complete; the float was the lie)
181    return (blank + lines)[-IC_ROWS:]

184def item_icon(e: Any) -> Any:
185    """A consumable/egg entry's icon as IC_ROWS cell lines.  ONE lookup for
186    every icon view -- the shop, the bag, the feed menu and the town shops
187    (refactor 2026-07-05); shop eggs ride their real egg frames."""
188    fr = None
189    if e and e.get("egg_idx") is not None:
190        import tuipet.core.egg as egg_mod
191        fr = egg_mod.frames(e["egg_idx"])
192    elif e:
193        import tuipet.data.loaders.data as data
194        fr = data.load_icons().get(e.get("key"))
195    import tuipet.core.shop as shop
196    art = shop.icon_art(e.get("key")) if e else None
197    if art:                       # a substitute rip outranks the sheet frame
198        return icon_cell(art)     # (and works even for a key with no sheet)
199    if not fr:
200        return [" " * IC_W] * IC_ROWS
201    return icon_cell(fr[shop.icon_frame(e.get("key")) % len(fr)])

213def list_window(out: Any, rows: Any, cursor: Any, vis: Any, fmt: Any, empty: Optional[Any]=None) -> Any:
214    """The shared scrolling list body: a vis-row window centred on the cursor,
215    each row through fmt(item, index), padded with blanks.  Retires seven
216    hand-rolled copies (audit 2026-07).  fmt returns a plain label (rendered
217    via row() with the ▸ cursor) OR a styled Text owning its whole line incl.
218    the newline -- for lists with their own row grammar (datacore EVOLVES).
219    Returns the clamped cursor so callers can keep theirs in range."""
220    n = len(rows)
221    cursor = min(cursor, max(0, n - 1))
222    if not n:
223        if empty:
224            out.append_text(row(empty))
225        out.append_text(blanks(vis - (1 if empty else 0)))
226        return cursor
227    lo = max(0, min(cursor - vis // 2, n - vis))
228    shown = 0
229    for i in range(lo, min(lo + vis, n)):
230        lbl = fmt(rows[i], i)
231        out.append_text(lbl if isinstance(lbl, Text) else row(lbl, i == cursor))
232        shown += 1
233    out.append_text(blanks(vis - shown))
234    return cursor

237def page_step(cursor: Any, n: Any, vis: Any, k: Any) -> Any:
238    """PgUp/PgDn for a CURSOR list: a vis-1 leap, clamped at both ends (never
239    wrapped -- a page key is "get me across this list", and wrapping past the
240    end reads as a bug).  Returns the new cursor, or None when k isn't a page
241    key so the caller's elif chain falls through untouched.
242
243    Help and the README have promised "PgUp/PgDn leap through long lists"
244    since 0.5.64, but only the top/window scrollers (help, egg guide, datacore,
245    options, lobby) ever implemented it -- the CURSOR lists, including the
246    31-row scene picker and the 24-slot cup board, silently ate the key
247    (help audit 2026-07-21).  One helper so the claim stays true everywhere."""
248    if k not in ("pageup", "pagedown"):
249        return None
250    if n <= 0:
251        return 0
252    step = max(1, vis - 1)
253    delta = -step if k == "pageup" else step
254    return max(0, min(n - 1, cursor + delta))

257def scroll_window(out: Any, rows: Any, off: Any, vis: Any, fmt: Any) -> Any:
258    """list_window's cursor-less cousin: a vis-row window at a raw scroll
259    OFFSET (requirement checklists, logs) -- no selection, no centring.  Same
260    fmt contract (plain label or a styled whole-line Text).  Returns the
261    clamped offset."""
262    off = max(0, min(off, max(0, len(rows) - vis)))
263    shown = 0
264    for i in range(off, min(off + vis, len(rows))):
265        lbl = fmt(rows[i], i)
266        out.append_text(lbl if isinstance(lbl, Text) else row(lbl))
267        shown += 1
268    out.append_text(blanks(vis - shown))
269    return off

272class SubHost:
273    """Mixin for panels that host a child panel in `self.sub` (battle inside
274    adventure, town inside adventure, battle inside town/cup).  One home for
275    the anim-delegate + sfx-bubble seam that was hand-rolled three ways --
276    the flee-boss bug lived in exactly this kind of drift (audit 2026-07)."""
277
278    sub = None
279
280    def sub_anim(self) -> Any:
281        """Delegate a frame to the child; bubble its sfx up.  True if handled.
282        A child with no anim() (ShopPanel) is simply held -- the host's own
283        clock pauses either way (adventure road-keys 2026-07-07)."""
284        if self.sub is None:
285            return False
286        if hasattr(self.sub, "anim"):
287            self.sub.anim()
288        self.sfx = getattr(self.sub, "sfx", None) or getattr(self, "sfx", None)
289        if getattr(self.sub, "sfx", None):
290            self.sub.sfx = None
291        return True
292
293    def sub_key(self, k: Any, on_done: Any) -> Any:
294        """Route a key to the child.  When the child finishes (('done', r)),
295        clear it and hand r to on_done.  Returns True if the child had the key."""
296        if self.sub is None:
297            return False
298        r = self.sub.key(k)
299        if r is not None and r[0] == "done":
300            self.sub = None
301            on_done(r[1])
302        return True
